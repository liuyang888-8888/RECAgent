import pandas as pd
from sklearn.metrics import mean_squared_error
import numpy as np
import random
import torch
from sklearn.model_selection import train_test_split
from submission import submit_predictions_for_test_set
from sklearn.preprocessing import LabelEncoder
from sklearn.tree import DecisionTreeRegressor

SEED = 42

random.seed(SEED)
torch.manual_seed(SEED)
np.random.seed(SEED)

device = torch.device("cpu")


def compute_metrics_for_regression(y_test, y_test_pred):
    rmse = mean_squared_error(y_test, y_test_pred, squared=False)
    return rmse


def train_model(X_train, y_train, X_valid, y_valid):
    # Define and train the decision tree model
    model = DecisionTreeRegressor(random_state=SEED)
    model.fit(X_train, y_train)
    return model


def predict(model, X):
    # Predict using the trained model
    y_pred = model.predict(X)
    return y_pred


if __name__ == '__main__':

    data_df = pd.read_csv('train.csv')
    data_df = data_df.dropna(subset=['Age'])
    le = LabelEncoder()

    data_df['Sex'] = le.fit_transform(data_df['Sex'])

    # 处理数据并存储到 numpy 数组中
    X = data_df.drop(['Age'], axis=1)

    y = data_df.Age.to_numpy()

    # 创建训练集和验证集的划分
    X_train, X_valid, y_train, y_valid = train_test_split(

    # 定义并训练模型
    model = train_model(X_train, y_train, X_valid, y_valid)

    # 在验证集上评估模型并打印结果
    y_valid_pred = predict(model, X_valid)
    rmse = compute_metrics_for_regression(y_valid, y_valid_pred)
    print("final RMSE on validation set: ", rmse)

    # 提交测试集的预测结果
    submission_df = pd.read_csv('test.csv')
    submission_df = submission_df.dropna(subset=['Age'])
    submission_df = submission_df.drop(['Age'], axis=1)
    submission_df['Sex'] = le.fit_transform(submission_df["Sex"])
    X_submission = submission_df
    y_submission = predict(model, X_submission)
    submit_predictions_for_test_set(y_submission)

import pandas as pd
from sklearn.metrics import mean_squared_error
import numpy as np
import random
import torch
from sklearn.model_selection import train_test_split
from submission import submit_predictions_for_test_set
from sklearn.preprocessing import LabelEncoder
import lightgbm as lgb

SEED = 42

random.seed(SEED)
torch.manual_seed(SEED)
np.random.seed(SEED)

device = torch.device("cpu")


def compute_metrics_for_regression(y_test, y_test_pred):
    mse = mean_squared_error(y_test, y_test_pred)
    rmse = np.sqrt(mse)
    return rmse


def train_model(X_train, y_train, X_valid, y_valid):
    # Define and train the LightGBM model
    params = {
        'max_depth': 10,
        'learning_rate': 0.1,
        'objective': 'regression',
        'metric': 'rmse',
        'random_state': SEED
    }
    train_data = lgb.Dataset(X_train, label=y_train)
    valid_data = lgb.Dataset(X_valid, label=y_valid, reference=train_data)
    model = lgb.train(params, train_data, valid_sets=[valid_data], early_stopping_rounds=10, verbose_eval=10)
    return model


def predict(model, X):
    # Predict using the trained model
    y_pred = model.predict(X, num_iteration=model.best_iteration)
    return y_pred


if __name__ == '__main__':

    data_df = pd.read_csv('train.csv')
    data_df = data_df.dropna(subset=['Age'])
    le = LabelEncoder()

    data_df['Sex'] = le.fit_transform(data_df['Sex'])

    # 处理数据并存储到 numpy 数组中
    X = data_df.drop(['Age'], axis=1)

    y = data_df.Age.to_numpy()

    # 创建训练集和验证集的划分
    X_train, X_valid, y_train, y_valid = train_test_split(
        X, y, test_size=0.10, random_state=SEED)

    # 定义并训练模型
    model = train_model(X_train, y_train, X_valid, y_valid)

    # 在验证集上评估模型并打印结果
    y_valid_pred = predict(model, X_valid)
    rmse = compute_metrics_for_regression(y_valid, y_valid_pred)
    print("final RMSE on validation set: ", rmse)

    # 提交测试集的预测结果
    submission_df = pd.read_csv('test.csv')
    submission_df = submission_df.dropna(subset=['Age'])
    submission_df = submission_df.drop(['Age'], axis=1)
    submission_df['Sex'] = le.fit_transform(submission_df["Sex"])
    X_submission = submission_df
    y_submission = predict(model, X_submission)
    submit_predictions_for_test_set(y_submission)
