import pandas as pd
from sklearn.metrics import f1_score
import numpy as np
import random
import torch
from sklearn.model_selection import train_test_split
from submission import submit_predictions_for_test_set
from sklearn.preprocessing import LabelEncoder
from sklearn.ensemble import RandomForestClassifier
import lightgbm as lgb

SEED = 42

random.seed(SEED)
torch.manual_seed(SEED)
np.random.seed(SEED)

device = torch.device("cpu")


def compute_metrics_for_classification(y_test, y_test_pred):
    f1 = f1_score(y_test, y_test_pred, average='weighted')
    return f1


def train_model(X_train, y_train, X_valid, y_valid, model_type='rf'):
    if model_type == 'rf':
        model = RandomForestClassifier(random_state=SEED)
    elif model_type == 'lgbm':
        model = lgb.LGBMClassifier(random_state=SEED)
    model.fit(X_train, y_train)
    return model


def predict(model, X):
    y_pred = model.predict(X)
    return y_pred


if __name__ == '__main__':

    data_df = pd.read_csv('train.csv')
    data_df = data_df.dropna(subset=['booking_status'])
    for col in data_df.columns:
        if pd.api.types.is_object_dtype(data_df[col]):
            le = LabelEncoder()
            data_df[col] = le.fit_transform(data_df[col])

    X = data_df.drop(['booking_status'], axis=1)
    y = data_df['booking_status'].to_numpy()

    X_train, X_valid, y_train, y_valid = train_test_split(
        X, y, test_size=0.10, random_state=SEED)

    # Train and evaluate the LGBM model
    lgbm_model = train_model(X_train, y_train, X_valid, y_valid, model_type='lgbm')
    y_valid_pred = predict(lgbm_model, X_valid)
    f1 = compute_metrics_for_classification(y_valid, y_valid_pred)
    print("Final F1 Score on validation set (LGBM): ", f1)

    # Submit test set predictions using the LGBM model
    submission_df = pd.read_csv('test.csv')
    submission_df = submission_df.dropna(subset=['booking_status'])
    for col in submission_df.columns:
        if pd.api.types.is_object_dtype(submission_df[col]):
            le = LabelEncoder()
            submission_df[col] = le.fit_transform(submission_df[col])
    submission_df = submission_df.drop(['booking_status'], axis=1)
    X_submission = submission_df
    y_submission = predict(lgbm_model, X_submission)
    submit_predictions_for_test_set(y_submission)
