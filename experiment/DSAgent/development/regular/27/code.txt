import pandas as pd
from sklearn.metrics import mean_squared_error
import numpy as np
import random
import torch
from sklearn.model_selection import train_test_split
from submission import submit_predictions_for_test_set
from sklearn.preprocessing import StandardScaler
from statsmodels.tsa.vector_ar.var_model import VAR
import warnings

warnings.filterwarnings("ignore")

SEED = 42

random.seed(SEED)
torch.manual_seed(SEED)
np.random.seed(SEED)

device = torch.device("cpu")


def compute_metrics_for_timeseries(y_fits, y):
    mse = mean_squared_error(y, y_fits)
    return mse


def train_model(X_train):
    # Normalize the data
    scaler = StandardScaler()
    X_train_scaled = scaler.fit_transform(X_train.values.reshape(-1, 1))
    
    # Reshape the data for VAR model
    X_train_scaled = X_train_scaled.reshape(-1, 1)
    
    # Define and train the model
    model = VAR(X_train_scaled)
    model_fit = model.fit()
    return model_fit, scaler


def predict(model, scaler, forecast_steps):
    # Predict the model
    forecast = model.forecast(model.y, steps=forecast_steps)
    forecast = scaler.inverse_transform(forecast)
    forecast_mean = forecast.flatten()
    return forecast_mean

import pandas as pd
from sklearn.metrics import mean_squared_error
import numpy as np
import random
import torch
from sklearn.model_selection import train_test_split
from sklearn.preprocessing import MinMaxScaler
from statsmodels.tsa.vector_ar.var_model import VAR
from statsmodels.tsa.arima.model import ARIMA
from submission import submit_predictions_for_test_set
SEED = 42

random.seed(SEED)
torch.manual_seed(SEED)
np.random.seed(SEED)

device = torch.device("cpu")

def compute_metrics_for_timeseries(y_fits, y):
    mse = mean_squared_error(y, y_fits)
    return mse

def train_model(X_train):
    # Normalize the input data
    scaler = MinMaxScaler()
    X_train_scaled = scaler.fit_transform(X_train)
    
    # Check if the data is univariate
    if X_train.shape[1] == 1:
        # Define and train the ARIMA model
        model = ARIMA(X_train_scaled.flatten(), order=(5, 1, 0))
        model_fit = model.fit()
    else:
        # Define and train the VAR model
        model = VAR(X_train_scaled)
        model_fit = model.fit()
    
    return model_fit, scaler

def predict(model, scaler, forecast_steps):
    if isinstance(model, ARIMA):
        # Predict the model for ARIMA
        forecast = model.forecast(steps=forecast_steps)
        forecast_scaled = scaler.inverse_transform(forecast.reshape(-1, 1)).flatten()
    else:
        # Predict the model for VAR
        forecast = model.forecast(model.y, steps=forecast_steps)
        forecast_scaled = scaler.inverse_transform(forecast)
    
    return forecast_scaled

if __name__ == '__main__':

    data_df = pd.read_csv('train.csv')
    data_df['Date'] = pd.to_datetime(data_df['Date'])
    data_df.set_index('Date', inplace=True)

    # 对数据按日期排序
    data_df.sort_index(inplace=True)

    # 假设数据集中有其他相关列，例如 "High", "Low", "Close", "Volume"
    # 如果没有其他列，可以使用单变量模型如ARIMA
    relevant_columns = ["Open", "High", "Low", "Close", "Volume"]
    if not all(col in data_df.columns for col in relevant_columns):
        relevant_columns = ["Open"]

    # 模型训练
    model, scaler = train_model(data_df[relevant_columns])

    # 打印模型评估结果
    if isinstance(model, ARIMA):
        fitted_values = scaler.inverse_transform(model.fittedvalues.values.reshape(-1, 1))
    else:
        fitted_values = scaler.inverse_transform(model.fittedvalues)
    
    mse = compute_metrics_for_timeseries(fitted_values[:, 0], data_df["Open"])
    print("final MSE on trained set: ", mse)

    # 提交预测结果
    # 往后预测 3 步
    forecast_steps = 3
    y_submission = predict(model, scaler, forecast_steps=forecast_steps)
    submit_predictions_for_test_set(forecast_steps, y_submission)
