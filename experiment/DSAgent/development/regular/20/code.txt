import pandas as pd
from sklearn.metrics import mean_squared_error
import numpy as np
import random
import torch
from sklearn.model_selection import train_test_split
from submission import submit_predictions_for_test_set
from sklearn.preprocessing import LabelEncoder
import xgboost as xgb

SEED = 42

torch.manual_seed(SEED)
np.random.seed(SEED)

device = torch.device("cpu")


def compute_metrics_for_regression(y_test, y_test_pred):
    rmse = mean_squared_error(y_test, y_test_pred, squared=False)
    return rmse


def train_model(X_train, y_train, X_valid, y_valid):
    # Define and train the model
    model = xgb.XGBRegressor(random_state=SEED)
    model.fit(X_train, y_train, eval_set=[(X_valid, y_valid)], early_stopping_rounds=10, verbose=False)
    return model


def predict(model, X):
    # Predict using the model
    y_pred = model.predict(X)
    return y_pred

    submission_df = submission_df.dropna(subset=['cnt'])

        if pd.api.types.is_object_dtype(submission_df[col]):
            le = LabelEncoder()
            submission_df[col] = le.fit_transform(submission_df[col])
    submission_df = submission_df.drop(['cnt'], axis=1)
    X_submission = submission_df

    submit_predictions_for_test_set(y_submission)

import pandas as pd
from sklearn.metrics import mean_squared_error
import numpy as np
import random
import torch
from sklearn.model_selection import train_test_split
from submission import submit_predictions_for_test_set
from sklearn.preprocessing import LabelEncoder
import xgboost as xgb

SEED = 42

random.seed(SEED)
torch.manual_seed(SEED)
np.random.seed(SEED)

device = torch.device("cpu")


def compute_metrics_for_regression(y_test, y_test_pred):
    mse = mean_squared_error(y_test, y_test_pred)
    rmse = np.sqrt(mse)
    return rmse, mse


def train_model(X_train, y_train, X_valid, y_valid, max_depth, learning_rate):
    # Define and train the model
    model = xgb.XGBRegressor(max_depth=max_depth, learning_rate=learning_rate, random_state=SEED)
    model.fit(X_train, y_train, eval_set=[(X_valid, y_valid)], verbose=False)
    return model


def predict(model, X):
    # Predict using the model
    y_pred = model.predict(X)
    return y_pred


if __name__ == '__main__':

    data_df = pd.read_csv('train.csv')
    data_df = data_df.dropna(subset=['cnt'])
    for col in data_df.columns:
        if pd.api.types.is_object_dtype(data_df[col]):
            le = LabelEncoder()
            data_df[col] = le.fit_transform(data_df[col])

    # Process data and store in numpy arrays
    X = data_df.drop(['cnt'], axis=1)
    y = data_df['cnt'].to_numpy()

    # Create train and validation set split
    X_train, X_valid, y_train, y_valid = train_test_split(
        X, y, test_size=0.10, random_state=SEED)

    # Define hyperparameter grid
    max_depths = [3, 5, 7, 9]
    learning_rates = [0.01, 0.05, 0.1]

    best_rmse = float('inf')
    best_params = None

    # Grid search over hyperparameters
    for max_depth in max_depths:
        for learning_rate in learning_rates:
            model = train_model(X_train, y_train, X_valid, y_valid, max_depth, learning_rate)
            y_valid_pred = predict(model, X_valid)
            rmse = compute_metrics_for_regression(y_valid, y_valid_pred)[0]
            print(f"Max Depth: {max_depth}, Learning Rate: {learning_rate}, RMSE: {rmse}")
            if rmse < best_rmse:
                best_rmse = rmse
                best_params = {'max_depth': max_depth, 'learning_rate': learning_rate}

    print(f"Best Hyperparameters: {best_params}, Best RMSE: {best_rmse}")

    # Train the final model with the best hyperparameters
    final_model = train_model(X_train, y_train, X_valid, y_valid, best_params['max_depth'], best_params['learning_rate'])

    # Evaluate the final model on the validation set
    y_valid_pred = predict(final_model, X_valid)
    rmse, mse = compute_metrics_for_regression(y_valid, y_valid_pred)
    print(f"Final RMSE on validation set: {rmse}, Final MSE on validation set: {mse}")

    # Submit predictions for the test set
    submission_df = pd.read_csv('test.csv')
    submission_df = submission_df.dropna(subset=['cnt'])
    for col in submission_df.columns:
        if pd.api.types.is_object_dtype(submission_df[col]):
            le = LabelEncoder()
            submission_df[col] = le.fit_transform(submission_df[col])
    submission_df = submission_df.drop(['cnt'], axis=1)
    X_submission = submission_df
    y_submission = predict(final_model, X_submission)
    submit_predictions_for_test_set(y_submission)
