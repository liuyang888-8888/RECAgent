import pandas as pd
from sklearn.metrics import silhouette_score
import numpy as np
import random
import torch
from sklearn.model_selection import train_test_split
from submission import submit_predictions_for_test_set
from sklearn.preprocessing import LabelEncoder
from sklearn.cluster import Birch
SEED = 42

random.seed(SEED)
torch.manual_seed(SEED)
np.random.seed(SEED)

device = torch.device("cpu")

def compute_metrics_for_clustering(X, labels):
    # 计算轮廓系数
    silhouette_avg = silhouette_score(X, labels)
    return silhouette_avg

def train_model(X_train):
    # 定义并训练 BIRCH 模型
    best_silhouette = -1
    best_model = None
    best_params = {}
    
    # Hyperparameter tuning
    hyperparameters = [
        {'threshold': 0.4, 'branching_factor': 50},
        {'threshold': 0.6, 'branching_factor': 50},
        {'threshold': 0.5, 'branching_factor': 60},
        {'threshold': 0.5, 'branching_factor': 40}
    ]
    
    for params in hyperparameters:
        model = Birch(threshold=params['threshold'], branching_factor=params['branching_factor'], n_clusters=5)
        model.fit(X_train)
        silhouette_avg = compute_metrics_for_clustering(X_train, model.labels_)
        
        if silhouette_avg > best_silhouette:
            best_silhouette = silhouette_avg
            best_model = model
            best_params = params
    
    print(f"Best hyperparameters: {best_params}")
    print(f"Best silhouette score: {best_silhouette}")
    
    return best_model

if __name__ == '__main__':

    data_df = pd.read_csv('train.csv')

    for col in data_df.columns:
        if pd.api.types.is_object_dtype(data_df[col]):
            le = LabelEncoder()
            data_df[col] = le.fit_transform(data_df[col])

    # 处理数据并存储到 numpy 数组中
    X = data_df.values

    # 定义并训练模型
    model = train_model(X)

    # 在训练集上评估模型并打印结果
    silhouette_avg = compute_metrics_for_clustering(X, model.labels_)
    print("final silhouette_avg on trained set: ", silhouette_avg)

    # 提交结果
    submit_predictions_for_test_set(model.labels_)
