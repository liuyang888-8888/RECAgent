import pandas as pd

def train_model(X_train, y_train, X_valid, y_valid):
    best_rmse = float('inf')
    best_model = None
    best_params = {}
    
    n_estimators_list = [50, 100, 200]
    learning_rate_list = [0.01, 0.1, 1.0]
    
    for n_estimators in n_estimators_list:

                best_rmse = rmse
                best_model = model
