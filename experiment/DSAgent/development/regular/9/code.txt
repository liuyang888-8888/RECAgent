import pandas as pd
from sklearn.metrics import f1_score
import numpy as np
import random
import torch
from sklearn.model_selection import train_test_split
from sklearn.ensemble import RandomForestClassifier
from sklearn.preprocessing import LabelEncoder
SEED = 42

random.seed(SEED)
torch.manual_seed(SEED)
np.random.seed(SEED)

device = torch.device("cpu")


def compute_metrics_for_classification(y_test, y_test_pred):
    f1 = f1_score(y_test, y_test_pred, average='weighted')
    return f1


def train_model(X_train, y_train, X_valid, y_valid):
    # Define the parameter grid for grid search
    param_grid = {
        'n_estimators': [50, 100, 200],
        'max_depth': [None, 10, 20, 30]
    }

    best_f1 = 0
    best_model = None
    best_params = {}

    for n_estimators in param_grid['n_estimators']:
        for max_depth in param_grid['max_depth']:
            # Define and train the model with current parameters
            model = RandomForestClassifier(n_estimators=n_estimators, max_depth=max_depth, random_state=SEED)
            model.fit(X_train, y_train)
            
            # Evaluate the model on the validation set
            y_valid_pred = model.predict(X_valid)
            f1 = compute_metrics_for_classification(y_valid, y_valid_pred)
            
            # Update the best model and parameters if this one is better
            if f1 > best_f1:
                best_f1 = f1
                best_model = model
                best_params = {'n_estimators': n_estimators, 'max_depth': max_depth}
    
    print(f"Best F1 Score: {best_f1}")
    print(f"Best Parameters: {best_params}")
    return best_model


def predict(model, X):
    # Use the trained model to make predictions
    y_pred = model.predict(X)
    return y_pred
