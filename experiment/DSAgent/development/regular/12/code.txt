import pandas as pd
from sklearn.metrics import f1_score
import numpy as np
import random
import torch
from sklearn.model_selection import train_test_split
from submission import submit_predictions_for_test_set
from sklearn.preprocessing import LabelEncoder
from xgboost import XGBClassifier
SEED = 42

random.seed(SEED)
torch.manual_seed(SEED)
np.random.seed(SEED)

device = torch.device("cpu")


def compute_metrics_for_classification(y_test, y_test_pred):
    f1 = f1_score(y_test, y_test_pred, average='weighted')
    return f1


def train_model(X_train, y_train, X_valid, y_valid, max_depth, learning_rate):
    # Define and train the XGBoost model with specified hyperparameters
    model = XGBClassifier(use_label_encoder=False, eval_metric='logloss', max_depth=max_depth, learning_rate=learning_rate)
    model.fit(X_train, y_train)
    return model


def predict(model, X):
    # Predict using the trained XGBoost model
    y_pred = model.predict(X)
    return y_pred


if __name__ == '__main__':

    data_df = pd.read_csv('train.csv')
    data_df = data_df.dropna(subset=['class'])
    for col in data_df.columns:
        if pd.api.types.is_object_dtype(data_df[col]):
            le = LabelEncoder()
            data_df[col] = le.fit_transform(data_df[col])

    # Process data and store in numpy arrays
    X = data_df.drop(['class'], axis=1)
    y = data_df['class'].to_numpy()

    # Create train and validation set split
    X_train, X_valid, y_train, y_valid = train_test_split(
        X, y, test_size=0.10, random_state=SEED)

    # Define the hyperparameter grid
    max_depths = [3, 5, 7, 9]
    learning_rates = [0.01, 0.05, 0.1]

    best_f1 = -1
    best_params = None

    # Perform grid search
    for max_depth in max_depths:
        for learning_rate in learning_rates:
            # Train the model with the current hyperparameters
            model = train_model(X_train, y_train, X_valid, y_valid, max_depth, learning_rate)

            # Evaluate the model on the validation set
            y_valid_pred = predict(model, X_valid)
            f1 = compute_metrics_for_classification(y_valid, y_valid_pred)

            # Check if this is the best model so far
            if f1 > best_f1:
                best_f1 = f1
                best_params = {'max_depth': max_depth, 'learning_rate': learning_rate}

            print(f"Max Depth: {max_depth}, Learning Rate: {learning_rate}, F1 Score: {f1}")

    print(f"Best Hyperparameters: {best_params}, Best F1 Score: {best_f1}")

    # Train the final model with the best hyperparameters
    final_model = train_model(X_train, y_train, X_valid, y_valid, best_params['max_depth'], best_params['learning_rate'])

    # Submit predictions for the test set
    submission_df = pd.read_csv('test.csv')
    submission_df = submission_df.dropna(subset=['class'])
    for col in submission_df.columns:
        if pd.api.types.is_object_dtype(data_df[col]):
            le = LabelEncoder()
            submission_df[col] = le.fit_transform(submission_df[col])
    submission_df = submission_df.drop(['class'], axis=1)
    X_submission = submission_df
    y_submission = predict(final_model, X_submission)
    submit_predictions_for_test_set(y_submission)
