import pandas as pd
from sklearn.metrics import mean_squared_error
import numpy as np
import random
import torch
from sklearn.model_selection import train_test_split
from sklearn.preprocessing import StandardScaler
from sklearn.svm import SVR
from submission import submit_predictions_for_test_set
from sklearn.preprocessing import LabelEncoder
SEED = 42

random.seed(SEED)
torch.manual_seed(SEED)
np.random.seed(SEED)

device = torch.device("cpu")


def compute_metrics_for_regression(y_test, y_test_pred):
    rmse = mean_squared_error(y_test, y_test_pred, squared=False)
    return rmse


def train_model(X_train, y_train, X_valid, y_valid):
    # Normalize the input features
    scaler = StandardScaler()
    X_train_scaled = scaler.fit_transform(X_train)
    X_valid_scaled = scaler.transform(X_valid)

    # Define and train the SVM model with RBF kernel
    model = SVR(kernel='rbf')
    model.fit(X_train_scaled, y_train)
    
    return model, scaler


def predict(model, X, scaler):
    # Normalize the input features
    X_scaled = scaler.transform(X)
    
    # Predict using the model
    y_pred = model.predict(X_scaled)
    return y_pred


if __name__ == '__main__':

    data_df = pd.read_csv('train.csv')
    data_df = data_df.dropna(subset=['yield'])

    # Process data and store in numpy arrays
    X = data_df.drop(['yield'], axis=1)
    y = data_df['yield'].to_numpy()

    # Create train and validation set split
    X_train, X_valid, y_train, y_valid = train_test_split(
        X, y, test_size=0.10, random_state=SEED)

    # Define and train the model
    model, scaler = train_model(X_train, y_train, X_valid, y_valid)

    # Evaluate the model on the validation set and print the result
    y_valid_pred = predict(model, X_valid, scaler)
    rmse = compute_metrics_for_regression(y_valid, y_valid_pred)
    print("Final RMSE on validation set: ", rmse)

    # Submit predictions for the test set
    submission_df = pd.read_csv('test.csv')
    submission_df = submission_df.dropna(subset=['yield'])
    submission_df = submission_df.drop(['yield'], axis=1)
    X_submission = submission_df
    y_submission = predict(model, X_submission, scaler)
    submit_predictions_for_test_set(y_submission)
