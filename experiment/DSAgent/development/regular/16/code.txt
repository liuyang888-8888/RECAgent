import pandas as pd
from sklearn.metrics import mean_squared_error
import numpy as np
import random
import torch
from sklearn.model_selection import train_test_split
from submission import submit_predictions_for_test_set
from sklearn.preprocessing import LabelEncoder, StandardScaler
from sklearn.linear_model import SGDRegressor
from sklearn.model_selection import GridSearchCV
SEED = 42

random.seed(SEED)
torch.manual_seed(SEED)
np.random.seed(SEED)

device = torch.device("cuda:0")

def encode_categorical_features(X):
    le = LabelEncoder()
    for column in X.select_dtypes(include=['object']).columns:
        X[column] = le.fit_transform(X[column].astype(str))
    return X

def scale_numerical_features(X):
    scaler = StandardScaler()
    numerical_features = X.select_dtypes(include=['float64', 'int64']).columns
    X[numerical_features] = scaler.fit_transform(X[numerical_features])
    return X

def compute_metrics_for_regression(y_test, y_test_pred):
    mse = mean_squared_error(y_test, y_test_pred)
    rmse = np.sqrt(mse)
    return rmse

def train_model(X_train, y_train, X_valid, y_valid):
    # Encode categorical features
    X_train = encode_categorical_features(X_train)
    X_valid = encode_categorical_features(X_valid)
    
    # Scale numerical features
    X_train = scale_numerical_features(X_train)
    X_valid = scale_numerical_features(X_valid)
    
    # Define the model and hyperparameter grid
    model = SGDRegressor(random_state=SEED)
    param_grid = {
        'learning_rate': ['constant', 'optimal', 'invscaling'],
        'alpha': [0.0001, 0.001, 0.01]
    }
    
    # Perform grid search to find the best hyperparameters
    grid_search = GridSearchCV(model, param_grid, cv=3, scoring='neg_mean_squared_error', n_jobs=-1)
    grid_search.fit(X_train, y_train)
    
    # Get the best model
    best_model = grid_search.best_estimator_
    
    # Evaluate the best model on the validation set
    y_valid_pred = best_model.predict(X_valid)
    rmse = compute_metrics_for_regression(y_valid, y_valid_pred)
    print("Best RMSE on validation set: ", rmse)
    
    return best_model

def predict(model, X):
    # Encode categorical features
    X = encode_categorical_features(X)
    
    # Scale numerical features
    X = scale_numerical_features(X)
    
    # Predict using the trained model
    y_pred = model.predict(X)
    return y_pred

if __name__ == '__main__':

    data_df = pd.read_csv('train.csv')
    data_df = data_df.dropna(subset=['cost'])

    # Process data and store in numpy arrays
    X = data_df.drop(['cost'], axis=1)
    y = data_df.cost.to_numpy()

    # Create train and validation splits
    X_train, X_valid, y_train, y_valid = train_test_split(
        X, y, test_size=0.10, random_state=SEED)

    # Define and train the model
    model = train_model(X_train, y_train, X_valid, y_valid)

    # Inference on the validation set
    y_valid_pred = predict(model, X_valid)
    rmse = compute_metrics_for_regression(y_valid, y_valid_pred)
    print("Final RMSE on validation set: ", rmse)

    # Submit predictions for the test set

    submission_df = submission_df.drop(['cost'], axis=1)
    X_submission = submission_df
    y_submission = predict(model, X_submission)
    submit_predictions_for_test_set(y_submission)
