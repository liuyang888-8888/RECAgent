import pandas as pd
from sklearn.metrics import f1_score
import numpy as np
import random
import torch
from sklearn.model_selection import train_test_split
from submission import submit_predictions_for_test_set
from sklearn.preprocessing import LabelEncoder
from sklearn.ensemble import AdaBoostClassifier

SEED = 42

random.seed(SEED)
torch.manual_seed(SEED)
np.random.seed(SEED)

device = torch.device("cpu")


def compute_metrics_for_classification(y_test, y_test_pred):
    f1 = f1_score(y_test, y_test_pred, average='weighted')
    return f1


def train_model(X_train, y_train, X_valid, y_valid, n_estimators, learning_rate):
    # Define and train the AdaBoost classifier
    model = AdaBoostClassifier(n_estimators=n_estimators, learning_rate=learning_rate, random_state=SEED)
    model.fit(X_train, y_train)
    return model


def predict(model, X):
    # Use the trained model to make predictions
    y_pred = model.predict(X)
    return y_pred


if __name__ == '__main__':

    data_df = pd.read_csv('train.csv')
    data_df = data_df.dropna(subset=['class'])
    for col in data_df.columns:
        if pd.api.types.is_object_dtype(data_df[col]):
            le = LabelEncoder()
            data_df[col] = le.fit_transform(data_df[col])

    # 处理数据并存储到 numpy 数组中
    X = data_df.drop(['class'], axis=1)
    y = data_df['class'].to_numpy()

    # 创建训练集和验证集的划分
    X_train, X_valid, y_train, y_valid = train_test_split(
        X, y, test_size=0.10, random_state=SEED)

    # Define the hyperparameter grid
    n_estimators_list = [10, 20, 30, 40, 50]
    learning_rate_list = [0.01, 0.1, 0.5]

    best_f1 = -1
    best_n_estimators = None
    best_learning_rate = None

    # Perform grid search
    for n_estimators in n_estimators_list:
        for learning_rate in learning_rate_list:
            model = train_model(X_train, y_train, X_valid, y_valid, n_estimators, learning_rate)
            y_valid_pred = predict(model, X_valid)
            f1 = compute_metrics_for_classification(y_valid, y_valid_pred)
            print(f"n_estimators: {n_estimators}, learning_rate: {learning_rate}, F1 Score: {f1}")
            if f1 > best_f1:
                best_f1 = f1
                best_n_estimators = n_estimators
                best_learning_rate = learning_rate

    print(f"Best hyperparameters: n_estimators={best_n_estimators}, learning_rate={best_learning_rate}, F1 Score: {best_f1}")

    # Train the final model with the best hyperparameters
    final_model = train_model(X_train, y_train, X_valid, y_valid, best_n_estimators, best_learning_rate)

    # Inference on the test set
    submission_df = pd.read_csv('test.csv')
    submission_df = submission_df.dropna(subset=['class'])
    for col in submission_df.columns:
        if pd.api.types.is_object_dtype(submission_df[col]):
            le = LabelEncoder()
            submission_df[col] = le.fit_transform(submission_df[col])
    submission_df = submission_df.drop(['class'], axis=1)
    X_submission = submission_df
    y_submission = predict(final_model, X_submission)
    submit_predictions_for_test_set(y_submission)
