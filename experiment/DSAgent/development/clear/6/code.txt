import pandas as pd
from sklearn.metrics import f1_score
import numpy as np
import random
import torch
from sklearn.model_selection import train_test_split
from submission import submit_predictions_for_test_set
from sklearn.preprocessing import LabelEncoder
import lightgbm as lgb

SEED = 42

random.seed(SEED)
torch.manual_seed(SEED)
np.random.seed(SEED)

device = torch.device("cpu")


def compute_metrics_for_classification(y_test, y_test_pred):
    f1 = f1_score(y_test, y_test_pred, average='weighted')
    return f1


def train_model(X_train, y_train, X_valid, y_valid):
    # Define and train the LightGBM model with default hyperparameters
    train_data = lgb.Dataset(X_train, label=y_train)
    valid_data = lgb.Dataset(X_valid, label=y_valid, reference=train_data)
    
    params = {
        'objective': 'multiclass',
        'num_class': len(np.unique(y_train)),
        'metric': 'multi_logloss',
        'boosting_type': 'gbdt',
        'verbose': 0,
        'seed': SEED
    }
    
    model = lgb.train(params, train_data, valid_sets=[valid_data], verbose_eval=False)
    return model


def predict(model, X):
    # Use the trained LightGBM model to make predictions
    y_pred = model.predict(X, num_iteration=model.best_iteration)
    y_pred = np.argmax(y_pred, axis=1)
    return y_pred


if __name__ == '__main__':

    data_df = pd.read_csv('train.csv')
    data_df = data_df.dropna(subset=['target'])
    for col in data_df.columns:
        if pd.api.types.is_object_dtype(data_df[col]):
            le = LabelEncoder()
            data_df[col] = le.fit_transform(data_df[col])

    # Process data and store in numpy arrays
    X = data_df.drop(['target'], axis=1)
    y = data_df['target'].to_numpy()

    # Create training and validation set split
    X_train, X_valid, y_train, y_valid = train_test_split(
        X, y, test_size=0.10, random_state=SEED)

    # Define and train model
    model = train_model(X_train, y_train, X_valid, y_valid)

    # Evaluate model on validation set and print results
    y_valid_pred = predict(model, X_valid)
    f1 = compute_metrics_for_classification(y_valid, y_valid_pred)
    print("Final F1 Score on validation set: ", f1)

    # Submit predictions for test set
    submission_df = pd.read_csv('test.csv')
    submission_df = submission_df.dropna(subset=['target'])
    for col in submission_df.columns:
        if pd.api.types.is_object_dtype(submission_df[col]):
            le = LabelEncoder()
            submission_df[col] = le.fit_transform(submission_df[col])
    submission_df = submission_df.drop(['target'], axis=1)
    X_submission = submission_df
    y_submission = predict(model, X_submission)
    submit_predictions_for_test_set(y_submission)
