import pandas as pd
from sklearn.metrics import f1_score
import numpy as np
import random
import torch
from sklearn.model_selection import train_test_split
from sklearn.linear_model import LogisticRegression
from sklearn.preprocessing import OneHotEncoder, LabelEncoder
import joblib

SEED = 42

random.seed(SEED)
torch.manual_seed(SEED)
np.random.seed(SEED)

device = torch.device("cpu")

def compute_metrics_for_classification(y_test, y_test_pred):
    f1 = f1_score(y_test, y_test_pred, average='weighted')
    return f1

def train_model(X_train, y_train, X_valid, y_valid):
    # One-hot encode the 'pdDistrict' field
    ohe = OneHotEncoder(sparse=False, handle_unknown='ignore')
    pdDistrict_encoded = ohe.fit_transform(X_train[['pdDistrict']])
    X_train_encoded = np.hstack([X_train.drop('pdDistrict', axis=1).values, pdDistrict_encoded])
    
    # Train the logistic regression model with L2 regularization (Ridge)
    model = LogisticRegression(penalty='l2', C=1.5, solver='lbfgs', max_iter=1000, random_state=SEED)
    model.fit(X_train_encoded, y_train)
    
    # Save the model and the one-hot encoder
    joblib.dump(model, 'logistic_regression_model.pkl')
    joblib.dump(ohe, 'one_hot_encoder.pkl')
    
    return model, ohe

def predict(model, ohe, X):
    # One-hot encode the 'pdDistrict' field
    pdDistrict_encoded = ohe.transform(X[['pdDistrict']])
    X_encoded = np.hstack([X.drop('pdDistrict', axis=1).values, pdDistrict_encoded])
    
    # Predict using the model
    y_pred = model.predict(X_encoded)
    return y_pred

    data_df = pd.read_csv('train.csv')

        if pd.api.types.is_object_dtype(data_df[col]):

    # Process data and store in numpy arrays
    X = data_df.drop(['category'], axis=1)
    y = data_df['category'].to_numpy()

    # Create train and validation splits
    X_train, X_valid, y_train, y_valid = train_test_split(
        X, y, test_size=0.10, random_state=SEED)

    # Define and train the model
    model, ohe = train_model(X_train, y_train, X_valid, y_valid)

    # Evaluate the model on the validation set
    y_valid_pred = predict(model, ohe, X_valid)
    f1 = compute_metrics_for_classification(y_valid, y_valid_pred)
    print("Final F1 Score on validation set: ", f1)

    # Submit predictions for the test set
    submission_df = pd.read_csv('test.csv')
    submission_df = submission_df.dropna(subset=['category'])

import pandas as pd
from sklearn.metrics import f1_score
import numpy as np
import random
import torch
from sklearn.model_selection import train_test_split, GridSearchCV
from sklearn.linear_model import LogisticRegression
from sklearn.preprocessing import OneHotEncoder, LabelEncoder
import joblib

SEED = 42

random.seed(SEED)
torch.manual_seed(SEED)
np.random.seed(SEED)

device = torch.device("cpu")

def compute_metrics_for_classification(y_test, y_test_pred):
    f1 = f1_score(y_test, y_test_pred, average='weighted')
    return f1

def train_model_with_cv(X_train, y_train):
    # One-hot encode the 'pdDistrict' field
    ohe = OneHotEncoder(sparse_output=False, handle_unknown='ignore')
    pdDistrict_encoded = ohe.fit_transform(X_train[['pdDistrict']])
    X_train_encoded = np.hstack([X_train.drop('pdDistrict', axis=1).values, pdDistrict_encoded])
    
    # Define the parameter grid for grid search
    param_grid = {'C': [0.5, 1.0, 1.5, 2.0]}
    
    # Initialize the logistic regression model
    model = LogisticRegression(penalty='l2', solver='lbfgs', max_iter=2000, random_state=SEED)
    
    # Perform grid search with cross-validation
    grid_search = GridSearchCV(model, param_grid, cv=5, scoring='f1_weighted', n_jobs=-1)
    grid_search.fit(X_train_encoded, y_train)
    
    # Get the best model and its parameters
    best_model = grid_search.best_estimator_
    best_params = grid_search.best_params_
    
    # Save the best model and the one-hot encoder
    joblib.dump(best_model, 'best_logistic_regression_model.pkl')
    joblib.dump(ohe, 'one_hot_encoder.pkl')
    
    return best_model, ohe, best_params

def predict(model, ohe, X):
    # One-hot encode the 'pdDistrict' field
    pdDistrict_encoded = ohe.transform(X[['pdDistrict']])
    X_encoded = np.hstack([X.drop('pdDistrict', axis=1).values, pdDistrict_encoded])
    
    # Predict using the model
    y_pred = model.predict(X_encoded)
    return y_pred

def submit_predictions_for_test_set(y_submission):
    # Save the predictions to a CSV file
    submission_df = pd.DataFrame({'category': y_submission})
    submission_df.to_csv('submission.csv', index=False)

if __name__ == '__main__':

    data_df = pd.read_csv('train.csv')
    data_df = data_df.dropna(subset=['category'])
    for col in data_df.columns:
        if pd.api.types.is_object_dtype(data_df[col]):
            le = LabelEncoder()
            data_df[col] = le.fit_transform(data_df[col])

    # Process data and store in numpy arrays
    X = data_df.drop(['category'], axis=1)
    y = data_df['category'].to_numpy()

    # Create train and validation splits
    X_train, X_valid, y_train, y_valid = train_test_split(
        X, y, test_size=0.10, random_state=SEED)

    # Define and train the model with cross-validation
    best_model, ohe, best_params = train_model_with_cv(X_train, y_train)

    # Evaluate the model on the validation set
    y_valid_pred = predict(best_model, ohe, X_valid)
    f1 = compute_metrics_for_classification(y_valid, y_valid_pred)
    print("Best C value:", best_params['C'])
    print("Best F1 Score on validation set: ", f1)

    # Submit predictions for the test set
    submission_df = pd.read_csv('test.csv')
    submission_df = submission_df.dropna(subset=['category'])
    for col in submission_df.columns:
        if pd.api.types.is_object_dtype(submission_df[col]):
            le = LabelEncoder()
            submission_df[col] = le.fit_transform(submission_df[col])
    submission_df = submission_df.drop(['category'], axis=1)
    X_submission = submission_df
    y_submission = predict(best_model, ohe, X_submission)
    submit_predictions_for_test_set(y_submission)
