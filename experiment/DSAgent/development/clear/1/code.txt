import pandas as pd
from sklearn.metrics import f1_score
import numpy as np
import random
import torch
from sklearn.model_selection import train_test_split
from sklearn.preprocessing import OneHotEncoder, LabelEncoder
from sklearn.ensemble import RandomForestClassifier
from submission import submit_predictions_for_test_set
SEED = 42

random.seed(SEED)
torch.manual_seed(SEED)
np.random.seed(SEED)

device = torch.device("cpu")

def compute_metrics_for_classification(y_test, y_test_pred):
    f1 = f1_score(y_test, y_test_pred, average='weighted')
    return f1

def train_model(X_train, y_train, X_valid, y_valid):
    model = RandomForestClassifier(n_estimators=10, random_state=SEED)
    model.fit(X_train, y_train)
    return model

def predict(model, X):
    y_pred = model.predict(X)
    return y_pred

if __name__ == '__main__':
    data_df = pd.read_csv('train.csv')
    data_df = data_df.dropna(subset=['Exited'])

    # One-hot encode 'Geography' and 'Gender'
    encoder = OneHotEncoder(sparse=False, handle_unknown='ignore')
    encoded_features = encoder.fit_transform(data_df[['Geography', 'Gender']])
    encoded_df = pd.DataFrame(encoded_features, columns=encoder.get_feature_names_out(['Geography', 'Gender']))
    data_df = pd.concat([data_df, encoded_df], axis=1).drop(['Geography', 'Gender'], axis=1)

    # Handle other categorical variables
    for col in data_df.columns:
        if pd.api.types.is_object_dtype(data_df[col]):
            le = LabelEncoder()
            data_df[col] = le.fit_transform(data_df[col])

    # Process data and store in numpy arrays
    X = data_df.drop(['Exited'], axis=1)
    y = data_df['Exited'].to_numpy()

    # Create train and validation splits
    X_train, X_valid, y_train, y_valid = train_test_split(X, y, test_size=0.10, random_state=SEED)

    # Define and train the model
    model = train_model(X_train, y_train, X_valid, y_valid)

    # Evaluate the model on the validation set
    y_valid_pred = predict(model, X_valid)
    f1 = compute_metrics_for_classification(y_valid, y_valid_pred)
    print("Final F1 Score on validation set: ", f1)

    # Submit test set predictions
    submission_df = pd.read_csv('test.csv')
    submission_df = submission_df.dropna(subset=['Exited'])

    # One-hot encode 'Geography' and 'Gender' in the test set
    encoded_features_test = encoder.transform(submission_df[['Geography', 'Gender']])
    encoded_df_test = pd.DataFrame(encoded_features_test, columns=encoder.get_feature_names_out(['Geography', 'Gender']))
    submission_df = pd.concat([submission_df, encoded_df_test], axis=1).drop(['Geography', 'Gender'], axis=1)

    # Handle other categorical variables in the test set
    for col in submission_df.columns:
        if pd.api.types.is_object_dtype(submission_df[col]):
            le = LabelEncoder()
            submission_df[col] = le.fit_transform(submission_df[col])

    # Drop 'Exited' column if it exists in the test set
    submission_df = submission_df.drop(['Exited'], axis=1, errors='ignore')
    X_submission = submission_df
    y_submission = predict(model, X_submission)
    submit_predictions_for_test_set(y_submission)
