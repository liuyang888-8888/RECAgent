import pandas as pd
from sklearn.metrics import mean_squared_error
import numpy as np
import random
import torch
from sklearn.model_selection import train_test_split
from submission import submit_predictions_for_test_set
from sklearn.preprocessing import LabelEncoder
import lightgbm as lgb

SEED = 42

random.seed(SEED)
torch.manual_seed(SEED)
np.random.seed(SEED)

device = torch.device("cpu")


def compute_metrics_for_regression(y_test, y_test_pred):
    rmse = mean_squared_error(y_test, y_test_pred, squared=False)
    return rmse


def train_model(X_train, y_train, X_valid, y_valid):
    # Define the LightGBM model with the specified hyperparameters
    params = {
        'boosting_type': 'rf',
        'n_estimators': 20,
        'objective': 'regression',
        'metric': 'rmse',
        'random_state': SEED
    }
    model = lgb.LGBMRegressor(**params)
    model.fit(X_train, y_train, eval_set=[(X_valid, y_valid)], early_stopping_rounds=10, verbose=False)
    return model


def predict(model, X):
    # Use the trained model to make predictions
    y_pred = model.predict(X)
    return y_pred


if __name__ == '__main__':

    data_df = pd.read_csv('train.csv')
    data_df = data_df.dropna(subset=['rings'])
    le = LabelEncoder()

    data_df['sex'] = le.fit_transform(data_df['sex'])

    # 处理数据并存储到 numpy 数组中
    X = data_df.drop(['rings'], axis=1)

    y = data_df.rings.to_numpy()

    # 创建训练集和验证集的划分
    X_train, X_valid, y_train, y_valid = train_test_split(
        X, y, test_size=0.10, random_state=SEED)

    # 定义并训练模型
    model = train_model(X_train, y_train, X_valid, y_valid)

    # 在验证集上评估模型并打印结果
    y_valid_pred = predict(model, X_valid)
    rmse = compute_metrics_for_regression(y_valid, y_valid_pred)
    print("Final RMSE on validation set: ", rmse)

    # 提交测试集的预测结果
    submission_df = pd.read_csv('test.csv')
    submission_df = submission_df.dropna(subset=['rings'])
    submission_df = submission_df.drop(['rings'], axis=1)
    submission_df['sex'] = le.transform(submission_df["sex"])
    X_submission = submission_df
    y_submission = predict(model, X_submission)
    submit_predictions_for_test_set(y_submission)
