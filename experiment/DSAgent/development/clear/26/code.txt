import pandas as pd
from sklearn.metrics import mean_squared_error
import numpy as np
import random
import torch
from statsmodels.tsa.arima.model import ARIMA
from sklearn.model_selection import train_test_split
from submission import submit_predictions_for_test_set
from sklearn.preprocessing import LabelEncoder
SEED = 42

random.seed(SEED)
torch.manual_seed(SEED)
np.random.seed(SEED)

device = torch.device("cpu")


def compute_metrics_for_timeseries(y_fits, y):
    mse = mean_squared_error(y, y_fits)
    return mse


def train_model(X_train):
    # Define and train the ARIMA model with p=2, d=1, q=1
    model = ARIMA(X_train, order=(2, 1, 1))
    model_fit = model.fit()
    return model_fit


def predict(model, forecast_steps):
    # Predict the model
    forecast = model.get_forecast(steps=forecast_steps)
    forecast_mean = forecast.predicted_mean
    return forecast_mean


if __name__ == '__main__':

    data_df = pd.read_csv('train.csv')
    data_df['date'] = pd.to_datetime(data_df['date'])
    data_df.set_index('date', inplace=True)

    # 对数据按日期排序
    data_df.sort_index(inplace=True)

    # Normalize the CO2 time series data
    mean_value = data_df["Average"].mean()
    normalized_data = data_df["Average"] / mean_value

    # Compute the rolling mean over a 12-month window and subtract it from the original time series
    rolling_mean = normalized_data.rolling(window=12).mean()
    normalized_data = normalized_data - rolling_mean

    # Drop NaN values resulting from the rolling mean calculation
    normalized_data.dropna(inplace=True)

    # 模型训练
    model = train_model(normalized_data)

    # 打印模型评估结果
    mse = compute_metrics_for_timeseries(model.fittedvalues, normalized_data)
    print("final MSE on trained set: ", mse)

    # 提交预测结果
    # 往后预测 3 步
    forecast_steps = 3
    y_submission = predict(model, forecast_steps=forecast_steps)
    submit_predictions_for_test_set(forecast_steps, y_submission)
