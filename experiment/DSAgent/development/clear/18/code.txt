import pandas as pd
from sklearn.metrics import mean_squared_error
import numpy as np
import random
import torch
from sklearn.model_selection import train_test_split
from submission import submit_predictions_for_test_set
from sklearn.preprocessing import LabelEncoder
from sklearn.svm import SVR
from joblib import dump

SEED = 42

random.seed(SEED)
torch.manual_seed(SEED)
np.random.seed(SEED)

device = torch.device("cpu")


def compute_metrics_for_regression(y_test, y_test_pred):
    mse = mean_squared_error(y_test, y_test_pred)
    rmse = np.sqrt(mse)
    return rmse


def train_model(X_train, y_train, X_valid, y_valid, C, epsilon):
    # Define and train the SVM model
    model = SVR(kernel='linear', C=C, epsilon=epsilon)
    model.fit(X_train, y_train)
    return model


def predict(model, X):
    # Predict using the trained model
    y_pred = model.predict(X)
    return y_pred


if __name__ == '__main__':

    data_df = pd.read_csv('train.csv')
    data_df = data_df.dropna(subset=['yield'])

    # 处理数据并存储到 numpy 数组中
    X = data_df.drop(['yield'], axis=1)

    y = data_df['yield'].to_numpy()

    # 创建训练集和验证集的划分
    X_train, X_valid, y_train, y_valid = train_test_split(
        X, y, test_size=0.10, random_state=SEED)

    # Define the hyperparameter grid
    Cs = [0.1, 0.5, 1.0, 2.0]
    epsilons = [0.01, 0.1, 0.5]

    best_rmse = float('inf')
    best_model = None
    best_params = {}

    # Perform grid search
    for C in Cs:
        for epsilon in epsilons:
            # Train the model with the current hyperparameters
            model = train_model(X_train, y_train, X_valid, y_valid, C, epsilon)
            
            # Evaluate the model on the validation set
            y_valid_pred = predict(model, X_valid)
            rmse = compute_metrics_for_regression(y_valid, y_valid_pred)
            print(f"RMSE for C={C}, epsilon={epsilon}: {rmse}")
            
            # Update the best model if this one is better
            if rmse < best_rmse:
                best_rmse = rmse
                best_model = model
                best_params = {'C': C, 'epsilon': epsilon}

    # Save the best model
    dump(best_model, 'best_svm_model.joblib')

    # Print the best parameters and RMSE
    print(f"Best parameters: {best_params}")
    print(f"Best RMSE on validation set: {best_rmse}")

    # Submit the test set predictions using the best model
    submission_df = pd.read_csv('test.csv')
    submission_df = submission_df.dropna(subset=['yield'])
    submission_df = submission_df.drop(['yield'], axis=1)
    X_submission = submission_df
    y_submission = predict(best_model, X_submission)
    submit_predictions_for_test_set(y_submission)
