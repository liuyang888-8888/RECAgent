import pandas as pd
from sklearn.metrics import f1_score
import numpy as np
import random
import torch
from sklearn.model_selection import train_test_split
from sklearn.ensemble import RandomForestClassifier
from sklearn.preprocessing import LabelEncoder
SEED = 42

random.seed(SEED)
torch.manual_seed(SEED)
np.random.seed(SEED)

device = torch.device("cpu")


def compute_metrics_for_classification(y_test, y_test_pred):
    f1 = f1_score(y_test, y_test_pred, average='weighted')
    return f1


def train_model(X_train, y_train, X_valid, y_valid):
    max_depths = [5, 10, 15]
    min_samples_splits = [2, 5, 10]
    best_f1 = 0
    best_model = None
    best_params = {}

    for depth in max_depths:
        for split in min_samples_splits:
            model = RandomForestClassifier(n_estimators=10, max_depth=depth, min_samples_split=split, random_state=SEED)
            model.fit(X_train, y_train)
            y_valid_pred = model.predict(X_valid)
            f1 = compute_metrics_for_classification(y_valid, y_valid_pred)
            print(f"Max Depth: {depth}, Min Samples Split: {split}, F1 Score: {f1}")
            if f1 > best_f1:
                best_f1 = f1
                best_model = model
                best_params = {'max_depth': depth, 'min_samples_split': split}
    
    print(f"Best Hyperparameters: {best_params}, Best F1 Score: {best_f1}")
    return best_model


def predict(model, X):
    # Generate predictions using the trained model
    y_pred = model.predict(X)
    return y_pred


def submit_predictions_for_test_set(y_submission):
    # Create a DataFrame for the predictions
    submission = pd.DataFrame({'id': range(len(y_submission)), 'target': y_submission})
    # Save the predictions to a CSV file

    # Process data and store in numpy arrays

    # Create train and validation splits
