import pandas as pd
from sklearn.metrics import mean_squared_error
import numpy as np
import random
import torch
from sklearn.model_selection import train_test_split
from submission import submit_predictions_for_test_set
from sklearn.preprocessing import LabelEncoder
from sklearn.ensemble import AdaBoostRegressor

SEED = 42

random.seed(SEED)
torch.manual_seed(SEED)
np.random.seed(SEED)

device = torch.device("cpu")


def compute_metrics_for_regression(y_test, y_test_pred):
    mse = mean_squared_error(y_test, y_test_pred)
    rmse = np.sqrt(mse)
    return rmse


def train_model(X_train, y_train, X_valid, y_valid, n_estimators, learning_rate):
    # Define and train the AdaBoost regressor with specified parameters
    model = AdaBoostRegressor(n_estimators=n_estimators, learning_rate=learning_rate, random_state=SEED)
    model.fit(X_train, y_train)
    return model


def predict(model, X):
    # Predict using the trained model
    y_pred = model.predict(X)
    return y_pred


if __name__ == '__main__':

    data_df = pd.read_csv('train.csv')
    data_df = data_df.dropna(subset=['G3'])
    for col in data_df.columns:
        if pd.api.types.is_object_dtype(data_df[col]):
            le = LabelEncoder()
            data_df[col] = le.fit_transform(data_df[col])

    # 处理数据并存储到 numpy 数组中
    X = data_df.drop(['G3'], axis=1)
    y = data_df['G3'].to_numpy()

    # 创建训练集和验证集的划分
    X_train, X_valid, y_train, y_valid = train_test_split(
        X, y, test_size=0.10, random_state=SEED)

    # Define the hyperparameter grid
    n_estimators_values = [50, 100, 200]
    learning_rate_values = [0.01, 0.1, 1.0]

    best_rmse = float('inf')
    best_n_estimators = None
    best_learning_rate = None
    best_model = None

    # Perform grid search
    for n_estimators in n_estimators_values:
        for learning_rate in learning_rate_values:
            model = train_model(X_train, y_train, X_valid, y_valid, n_estimators, learning_rate)
            y_valid_pred = predict(model, X_valid)
            rmse = compute_metrics_for_regression(y_valid, y_valid_pred)
            print(f"RMSE for n_estimators={n_estimators}, learning_rate={learning_rate}: {rmse}")
            if rmse < best_rmse:
                best_rmse = rmse
                best_n_estimators = n_estimators
                best_learning_rate = learning_rate
                best_model = model

    print(f"Best configuration: n_estimators={best_n_estimators}, learning_rate={best_learning_rate}, RMSE={best_rmse}")

    # Evaluate the best model on the test set
    submission_df = pd.read_csv('test.csv')
    submission_df = submission_df.dropna(subset=['G3'])
    for col in submission_df.columns:
        if pd.api.types.is_object_dtype(submission_df[col]):
            le = LabelEncoder()
            submission_df[col] = le.fit_transform(submission_df[col])
    submission_df = submission_df.drop(['G3'], axis=1)
    X_submission = submission_df
    y_submission = predict(best_model, X_submission)
    submit_predictions_for_test_set(y_submission)
