import pandas as pd
from sklearn.metrics import silhouette_score
import numpy as np
import random
import torch
from sklearn.model_selection import train_test_split
from submission import submit_predictions_for_test_set
from sklearn.preprocessing import LabelEncoder
from sklearn.cluster import KMeans
from sklearn.impute import SimpleImputer

SEED = 42

random.seed(SEED)
torch.manual_seed(SEED)
np.random.seed(SEED)

device = torch.device("cpu")

def compute_metrics_for_clustering(X, labels):
    # 计算轮廓系数
    silhouette_avg = silhouette_score(X, labels)
    return silhouette_avg

def train_model(X_train):
    # Handle missing values
    imputer_num = SimpleImputer(strategy='mean')
    imputer_cat = SimpleImputer(strategy='most_frequent')
    
    # Apply imputation
    X_train_num = X_train.select_dtypes(include=[np.number])
    X_train_cat = X_train.select_dtypes(exclude=[np.number])
    
    X_train_num_imputed = imputer_num.fit_transform(X_train_num)
    X_train_cat_imputed = imputer_cat.fit_transform(X_train_cat)
    
    # Combine imputed data
    X_train_imputed = np.hstack((X_train_num_imputed, X_train_cat_imputed))
    
    # Define and train the k-Means model
    model = KMeans(n_clusters=5, random_state=SEED)
    model.fit(X_train_imputed)
    
    return model

if __name__ == '__main__':
    data_df = pd.read_csv('train.csv')

    for col in data_df.columns:
        if pd.api.types.is_object_dtype(data_df[col]):
            le = LabelEncoder()
            data_df[col] = le.fit_transform(data_df[col])

    # Preprocess data
    X = data_df

    # Define and train model
    model = train_model(X)

    # Evaluate model and print silhouette coefficient
    silhouette_avg = compute_metrics_for_clustering(X, model.labels_)
    print("Final silhouette_avg on trained set: ", silhouette_avg)

    # Submit results
    submit_predictions_for_test_set(model.labels_)
