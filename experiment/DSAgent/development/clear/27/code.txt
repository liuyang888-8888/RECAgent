import pandas as pd
from sklearn.metrics import mean_squared_error
import numpy as np
import random
import torch
from sklearn.model_selection import train_test_split
from submission import submit_predictions_for_test_set
from sklearn.preprocessing import LabelEncoder
from statsmodels.tsa.vector_ar.var_model import VAR

SEED = 42

random.seed(SEED)
torch.manual_seed(SEED)
np.random.seed(SEED)

device = torch.device("cpu")


def compute_metrics_for_timeseries(y_fits, y):
    mse = mean_squared_error(y, y_fits)
    return mse


def train_model(X_train):
    # Define and fit the VAR model
    model = VAR(X_train)
    fitted_model = model.fit()
    return fitted_model


def predict(model, forecast_steps):
    # Predict the model
    forecast = model.forecast(model.y, steps=forecast_steps)
    forecast_mean = pd.Series(forecast[:, 0], index=pd.date_range(start=model.data.dates[-1], periods=forecast_steps+1, closed='right'))
    return forecast_mean


if __name__ == '__main__':

    data_df = pd.read_csv('train.csv')
    data_df['Date'] = pd.to_datetime(data_df['Date'])
    data_df.set_index('Date', inplace=True)

    # 对数据按日期排序
    data_df.sort_index(inplace=True)

    # 模型训练
    model = train_model(data_df[["Open"]])

    # 打印模型评估结果
    mse = compute_metrics_for_timeseries(model.fittedvalues, data_df["Open"])

    # 提交预测结果
    # 往后预测 3 步
    forecast_steps = 3
    y_submission = predict(model, forecast_steps=forecast_steps)
