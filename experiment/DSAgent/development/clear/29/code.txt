import pandas as pd
from sklearn.metrics import mean_squared_error
import numpy as np
import random
import torch
from statsmodels.tsa.statespace.sarimax import SARIMAX
from sklearn.model_selection import train_test_split
from submission import submit_predictions_for_test_set
from sklearn.preprocessing import StandardScaler
SEED = 42

random.seed(SEED)
torch.manual_seed(SEED)
np.random.seed(SEED)

device = torch.device("cpu")


def compute_metrics_for_timeseries(y_fits, y):
    mse = mean_squared_error(y, y_fits)
    return mse


def train_model(X_train):
    # Normalize the data
    scaler = StandardScaler()
    X_train_scaled = scaler.fit_transform(X_train.values.reshape(-1, 1)).flatten()
    
    # Fit the ARIMA model
    model = SARIMAX(X_train_scaled, order=(1, 1, 1))
    model_fit = model.fit(disp=False)
    
    return model_fit, scaler


def predict(model, forecast_steps, scaler):
    # Predict the model
    forecast = model.get_forecast(steps=forecast_steps)
    forecast_mean = forecast.predicted_mean
    forecast_mean = scaler.inverse_transform(forecast_mean.values.reshape(-1, 1)).flatten()
    return forecast_mean


if __name__ == '__main__':

    data_df = pd.read_csv('train.csv')
    data_df['Date'] = pd.to_datetime(data_df['Date'])
    data_df.set_index('Date', inplace=True)

    # 对数据按日期排序
    data_df.sort_index(inplace=True)

    # 模型训练
    model, scaler = train_model(data_df["Open"])

    # 打印模型评估结果
    y_fits = scaler.inverse_transform(model.fittedvalues.values.reshape(-1, 1)).flatten()
    mse = compute_metrics_for_timeseries(y_fits, data_df["Open"])
    print("final MSE on trained set: ", mse)

    # 提交预测结果
    # 往后预测 3 步
    forecast_steps = 3
    y_submission = predict(model, forecast_steps, scaler)
    submit_predictions_for_test_set(forecast_steps, y_submission)

import pandas as pd
from sklearn.metrics import mean_squared_error
import numpy as np
import random
import torch
from statsmodels.tsa.statespace.sarimax import SARIMAX
from sklearn.model_selection import train_test_split
from submission import submit_predictions_for_test_set
from sklearn.preprocessing import StandardScaler
SEED = 42

random.seed(SEED)
torch.manual_seed(SEED)
np.random.seed(SEED)

device = torch.device("cpu")


def compute_metrics_for_timeseries(y_fits, y):
    mse = mean_squared_error(y, y_fits)
    return mse


def train_model(X_train):
    # Normalize the data
    scaler = StandardScaler()
    X_train_scaled = scaler.fit_transform(X_train.values.reshape(-1, 1)).flatten()
    
    # Define the ARIMA orders to test
    arima_orders = [(1, 1, 0), (1, 1, 1), (1, 1, 2), (2, 1, 1)]
    best_mse = float('inf')
    best_model = None
    
    for order in arima_orders:
        # Fit the ARIMA model
        model = SARIMAX(X_train_scaled, order=order)
        model_fit = model.fit(disp=False)
        
        # Compute the MSE on the training set
        y_fits = scaler.inverse_transform(model_fit.fittedvalues.reshape(-1, 1)).flatten()
        mse = compute_metrics_for_timeseries(y_fits, X_train)
        
        print(f"Order {order} - MSE: {mse}")
        
        if mse < best_mse:
            best_mse = mse
            best_model = model_fit
    
    return best_model, scaler


def predict(model, forecast_steps, scaler):
    # Predict the model
    forecast = model.get_forecast(steps=forecast_steps)
    forecast_mean = forecast.predicted_mean
    forecast_mean = scaler.inverse_transform(forecast_mean.reshape(-1, 1)).flatten()
    return forecast_mean

    # 对数据按日期排序
