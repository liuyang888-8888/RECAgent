import pandas as pd
from sklearn.metrics import mean_squared_error
import numpy as np
import random
import torch
from statsmodels.tsa.arima.model import ARIMA
from sklearn.model_selection import train_test_split
from submission import submit_predictions_for_test_set
from sklearn.preprocessing import LabelEncoder
SEED = 42

random.seed(SEED)
torch.manual_seed(SEED)
np.random.seed(SEED)

device = torch.device("cpu")


def compute_metrics_for_timeseries(y_fits, y):
    mse = mean_squared_error(y, y_fits)
    return mse


def train_model(X_train):
    # Define and train the ARIMA model
    model = ARIMA(X_train, order=(5, 1, 0))
    model_fit = model.fit()
    return model_fit


def predict(model, forecast_steps):
    # Generate forecasts using the fitted ARIMA model
    forecast = model.get_forecast(steps=forecast_steps)
    forecast_mean = forecast.predicted_mean
    return forecast_mean


def add_lag_features(df, column, lags):
    for lag in lags:
        df[f'{column}_lag_{lag}'] = df[column].shift(lag)
    return df


def add_rolling_window_features(df, column, window):
    df[f'{column}_rolling_mean_{window}'] = df[column].rolling(window=window).mean()
    df[f'{column}_rolling_std_{window}'] = df[column].rolling(window=window).std()
    return df


if __name__ == '__main__':

    data_df = pd.read_csv('train.csv')
    data_df['Date'] = pd.to_datetime(data_df['Date'])
    data_df.set_index('Date', inplace=True)

    # 对数据按日期排序
    data_df.sort_index(inplace=True)

    # 添加滞后特征
    data_df = add_lag_features(data_df, 'Open', [1, 2, 3])

    # 添加滚动窗口统计特征
    data_df = add_rolling_window_features(data_df, 'Open', 5)

    # 去除含有 NaN 的行
    data_df.dropna(inplace=True)

    # 模型训练
    model = train_model(data_df["Open"])

    # 打印模型评估结果
    mse = compute_metrics_for_timeseries(model.fittedvalues, data_df["Open"])
    print("final MSE on trained set: ", mse)

    # 提交预测结果

    submit_predictions_for_test_set(forecast_steps, y_submission)
