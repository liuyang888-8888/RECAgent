    rmse = mean_squared_error(y_test, y_test_pred, squared=False)
    return rmse

    return model


def predict(model, X):

import pandas as pd
from sklearn.metrics import mean_squared_error
import numpy as np
import random
import torch
from sklearn.model_selection import train_test_split
from submission import submit_predictions_for_test_set
from sklearn.preprocessing import LabelEncoder
import lightgbm as lgb
from hyperopt import fmin, tpe, hp, Trials, STATUS_OK

SEED = 42

random.seed(SEED)
torch.manual_seed(SEED)
np.random.seed(SEED)

device = torch.device("cpu")


def compute_metrics_for_regression(y_test, y_test_pred):
    mse = mean_squared_error(y_test, y_test_pred)
    rmse = np.sqrt(mse)
    return rmse


def train_model(X_train, y_train, X_valid, y_valid, params):
    # Define and train the LightGBM model with specified hyperparameters
    train_data = lgb.Dataset(X_train, label=y_train)
    valid_data = lgb.Dataset(X_valid, label=y_valid, reference=train_data)
    
    model = lgb.train(params, train_data, valid_sets=[valid_data], early_stopping_rounds=10, verbose_eval=False)
    return model


def predict(model, X):
    # Generate predictions using the trained LightGBM model
    y_pred = model.predict(X, num_iteration=model.best_iteration)
    return y_pred


def objective(params, X_train, y_train, X_valid, y_valid):
    params = {
        'objective': 'regression',
        'metric': 'rmse',
        'boosting_type': 'gbdt',
        'verbose': 0,
        'seed': SEED,
        'learning_rate': params['learning_rate'],
        'n_estimators': int(params['n_estimators'])
    }
    
    model = train_model(X_train, y_train, X_valid, y_valid, params)
    y_valid_pred = predict(model, X_valid)
    rmse = compute_metrics_for_regression(y_valid, y_valid_pred)
    return {'loss': rmse, 'status': STATUS_OK}


if __name__ == '__main__':

    data_df = pd.read_csv('train.csv')
    data_df = data_df.dropna(subset=['area'])
    for col in data_df.columns:
        if pd.api.types.is_object_dtype(data_df[col]):
            le = LabelEncoder()
            data_df[col] = le.fit_transform(data_df[col])

    # Process data and store in numpy arrays
    X = data_df.drop(['area'], axis=1)
    y = data_df['area'].to_numpy()

    # Create train and validation set split
    X_train, X_valid, y_train, y_valid = train_test_split(
        X, y, test_size=0.10, random_state=SEED)

    # Define the search space for hyperparameters
    space = {
        'learning_rate': hp.loguniform('learning_rate', -5, 0),
        'n_estimators': hp.quniform('n_estimators', 50, 500, 1)
    }

    # Perform hyperparameter tuning
    trials = Trials()
    best = fmin(fn=lambda params: objective(params, X_train, y_train, X_valid, y_valid),
                space=space,
                algo=tpe.suggest,
                max_evals=50,
                trials=trials)

    # Print the best hyperparameters found
    print("Best hyperparameters found: ", best)

    # Train the model with the best hyperparameters
    best_params = {
        'objective': 'regression',
        'metric': 'rmse',
        'boosting_type': 'gbdt',
        'verbose': 0,
        'seed': SEED,
        'learning_rate': best['learning_rate'],
        'n_estimators': int(best['n_estimators'])
    }
    model = train_model(X_train, y_train, X_valid, y_valid, best_params)

    # Evaluate the model on the validation set and print the result
    y_valid_pred = predict(model, X_valid)
    rmse = compute_metrics_for_regression(y_valid, y_valid_pred)
    print("Final RMSE on validation set after tuning: ", rmse)

    # Submit predictions for the test set
    submission_df = pd.read_csv('test.csv')
    submission_df = submission_df.dropna(subset=['area'])
    for col in submission_df.columns:
        if pd.api.types.is_object_dtype(submission_df[col]):
            le = LabelEncoder()
            submission_df[col] = le.fit_transform(submission_df[col])
    submission_df = submission_df.drop(['area'], axis=1)
    X_submission = submission_df
    y_submission = predict(model, X_submission)
    submit_predictions_for_test_set(y_submission)
