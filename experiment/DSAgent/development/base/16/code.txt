import pandas as pd
from sklearn.metrics import mean_squared_error
import numpy as np
import random
import torch
from sklearn.model_selection import train_test_split
from xgboost import XGBRegressor
from submission import submit_predictions_for_test_set
from sklearn.preprocessing import LabelEncoder
SEED = 42

random.seed(SEED)
torch.manual_seed(SEED)
np.random.seed(SEED)

device = torch.device("cuda:0")


def compute_metrics_for_regression(y_test, y_test_pred):
    rmse = mean_squared_error(y_test, y_test_pred, squared=False)
    return rmse


def train_model(X_train, y_train, X_valid, y_valid):
    # Define and train the XGBoost model with default hyperparameters
    model = XGBRegressor()
    model.fit(X_train, y_train, eval_set=[(X_valid, y_valid)], early_stopping_rounds=10, verbose=False)
    return model


def predict(model, X):
    # Predict using the trained model
    y_pred = model.predict(X)
    return y_pred


if __name__ == '__main__':

    data_df = pd.read_csv('train.csv')
    data_df = data_df.dropna(subset=['cost'])

    # Process data and store in numpy arrays
    X = data_df.drop(['cost'], axis=1)

    y = data_df.cost.to_numpy()

    # Create train and validation set split
    X_train, X_valid, y_train, y_valid = train_test_split(
        X, y, test_size=0.10, random_state=SEED)

    # Define and train the model
    model = train_model(X_train, y_train, X_valid, y_valid)

    # Evaluate the model on the validation set and print the result
    y_valid_pred = predict(model, X_valid)
    rmse = compute_metrics_for_regression(y_valid, y_valid_pred)
    print("Final RMSE on validation set: ", rmse)

    # Submit predictions for the test set
    submission_df = pd.read_csv('test.csv')
    submission_df = submission_df.dropna(subset=['cost'])
    submission_df = submission_df.drop(['cost'], axis=1)
    X_submission = submission_df
    y_submission = predict(model, X_submission)
    submit_predictions_for_test_set(y_submission)

import pandas as pd
from sklearn.metrics import mean_squared_error
import numpy as np
import random
import torch
from sklearn.model_selection import train_test_split
from xgboost import XGBRegressor
from submission import submit_predictions_for_test_set
from sklearn.preprocessing import LabelEncoder
SEED = 42

random.seed(SEED)
torch.manual_seed(SEED)
np.random.seed(SEED)

device = torch.device("cuda:0")

def encode_categorical_features(df):
    le = LabelEncoder()
    for column in df.select_dtypes(include=['object']).columns:
        df[column] = le.fit_transform(df[column].astype(str))
    return df

def compute_metrics_for_regression(y_test, y_test_pred):
    mse = mean_squared_error(y_test, y_test_pred)
    rmse = np.sqrt(mse)
    return rmse

def train_model(X_train, y_train, X_valid, y_valid):
    # Define and train the XGBoost model with tuned hyperparameters
    model = XGBRegressor(max_depth=5, learning_rate=0.1, n_estimators=100)
    model.fit(X_train, y_train)
    return model

def predict(model, X):
    # Predict using the trained model
    y_pred = model.predict(X)
    return y_pred

if __name__ == '__main__':

    data_df = pd.read_csv('train.csv')
    data_df = data_df.dropna(subset=['cost'])

    # Encode categorical features
    data_df = encode_categorical_features(data_df)

    # Process data and store in numpy arrays
    X = data_df.drop(['cost'], axis=1)
    y = data_df.cost.to_numpy()

    # Create train and validation set split
    X_train, X_valid, y_train, y_valid = train_test_split(
        X, y, test_size=0.10, random_state=SEED)

    # Define and train the model
    model = train_model(X_train, y_train, X_valid, y_valid)

    # Evaluate the model on the validation set and print the result
    y_valid_pred = predict(model, X_valid)
    rmse = compute_metrics_for_regression(y_valid, y_valid_pred)
    print("Final RMSE on validation set: ", rmse)

    # Submit predictions for the test set
    submission_df = pd.read_csv('test.csv')
    submission_df = submission_df.dropna(subset=['cost'])
    submission_df = submission_df.drop(['cost'], axis=1)
    submission_df = encode_categorical_features(submission_df)
    X_submission = submission_df
    y_submission = predict(model, X_submission)
    submit_predictions_for_test_set(y_submission)
