import pandas as pd
from sklearn.metrics import f1_score
import numpy as np
import random
import torch
from sklearn.model_selection import train_test_split
from sklearn.linear_model import LogisticRegression
from sklearn.preprocessing import LabelEncoder
import lightgbm as lgb

SEED = 42

random.seed(SEED)
torch.manual_seed(SEED)
np.random.seed(SEED)

device = torch.device("cpu")


def compute_metrics_for_classification(y_test, y_test_pred):
    f1 = f1_score(y_test, y_test_pred, average='weighted')
    return f1


def train_logistic_regression(X_train, y_train, X_valid, y_valid):
    # Define and train the Logistic Regression model
    model = LogisticRegression(random_state=SEED, max_iter=1000)
    model.fit(X_train, y_train)
    return model


def train_lightgbm(X_train, y_train, X_valid, y_valid):
    # Define and train the LightGBM model
    train_data = lgb.Dataset(X_train, label=y_train)
    valid_data = lgb.Dataset(X_valid, label=y_valid, reference=train_data)
    
    params = {
        'objective': 'multiclass',
        'num_class': len(np.unique(y_train)),
        'metric': 'multi_logloss',
        'boosting_type': 'gbdt',
        'learning_rate': 0.1,
        'num_leaves': 31,
        'max_depth': -1,
        'min_data_in_leaf': 20,
        'feature_fraction': 0.9,
        'bagging_fraction': 0.8,
        'bagging_freq': 5,
        'verbose': 0,
        'seed': SEED
    }
    
    model = lgb.train(params, train_data, valid_sets=[valid_data], num_boost_round=1000, early_stopping_rounds=100, verbose_eval=100)
    return model


def predict(model, X):
    # Predict using the trained model
    if isinstance(model, lgb.Booster):
        y_pred = model.predict(X, num_iteration=model.best_iteration)
        y_pred = np.argmax(y_pred, axis=1)
    else:
        y_pred = model.predict(X)
    return y_pred


def submit_predictions_for_test_set(y_submission):
    # Save the predictions to a CSV file
    submission = pd.DataFrame({'id': range(len(y_submission)), 'target': y_submission})
    submission.to_csv('submission.csv', index=False)
    print("Predictions saved to submission.csv")


if __name__ == '__main__':

    data_df = pd.read_csv('train.csv')
    data_df = data_df.dropna(subset=['target'])
    for col in data_df.columns:
        if pd.api.types.is_object_dtype(data_df[col]):
            le = LabelEncoder()
            data_df[col] = le.fit_transform(data_df[col])

    # Process data and store in numpy arrays
    X = data_df.drop(['target'], axis=1)
    y = data_df['target'].to_numpy()

    # Create training and validation set split
    X_train, X_valid, y_train, y_valid = train_test_split(
        X, y, test_size=0.10, random_state=SEED)

    # Train and evaluate the Logistic Regression model
    lr_model = train_logistic_regression(X_train, y_train, X_valid, y_valid)
    y_valid_pred_lr = predict(lr_model, X_valid)
    f1_lr = compute_metrics_for_classification(y_valid, y_valid_pred_lr)
    print("F1 Score (Logistic Regression) on validation set: ", f1_lr)

    # Train and evaluate the LightGBM model
    lgb_model = train_lightgbm(X_train, y_train, X_valid, y_valid)
    y_valid_pred_lgb = predict(lgb_model, X_valid)
    f1_lgb = compute_metrics_for_classification(y_valid, y_valid_pred_lgb)
    print("F1 Score (LightGBM) on validation set: ", f1_lgb)

    # Compare the performance of the two models
    if f1_lgb > f1_lr:
        print("LightGBM model outperforms Logistic Regression.")
        best_model = lgb_model
    else:
        print("Logistic Regression model outperforms LightGBM.")
        best_model = lr_model

    # Submit predictions for the test set
    submission_df = pd.read_csv('test.csv')
    submission_df = submission_df.dropna(subset=['target'])
    for col in submission_df.columns:
        if pd.api.types.is_object_dtype(submission_df[col]):
            le = LabelEncoder()
            submission_df[col] = le.fit_transform(submission_df[col])
    submission_df = submission_df.drop(['target'], axis=1)
    X_submission = submission_df
    y_submission = predict(best_model, X_submission)
    submit_predictions_for_test_set(y_submission)
