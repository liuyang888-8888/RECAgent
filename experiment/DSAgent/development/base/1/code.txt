import pandas as pd
from sklearn.metrics import f1_score
import numpy as np
import random
import torch
from sklearn.model_selection import train_test_split
from submission import submit_predictions_for_test_set
from sklearn.preprocessing import LabelEncoder
import lightgbm as lgb
from hyperopt import fmin, tpe, hp, STATUS_OK, Trials

SEED = 42

random.seed(SEED)
torch.manual_seed(SEED)
np.random.seed(SEED)

device = torch.device("cpu")


def compute_metrics_for_classification(y_test, y_test_pred):
    f1 = f1_score(y_test, y_test_pred, average='weighted')
    return f1


def train_model(X_train, y_train, X_valid, y_valid, params):
    # Define and train the model
    model = lgb.LGBMClassifier(**params, random_state=SEED)
    model.fit(X_train, y_train, eval_set=[(X_valid, y_valid)], early_stopping_rounds=10, verbose=False)
    return model


def predict(model, X):
    # Predict the model
    y_pred = model.predict(X)
    return y_pred


def objective(params):
    model = train_model(X_train, y_train, X_valid, y_valid, params)
    y_valid_pred = predict(model, X_valid)
    f1 = compute_metrics_for_classification(y_valid, y_valid_pred)
    return {'loss': -f1, 'status': STATUS_OK}


if __name__ == '__main__':

    data_df = pd.read_csv('train.csv')
    data_df = data_df.dropna(subset=['Exited'])
    for col in data_df.columns:
        if pd.api.types.is_object_dtype(data_df[col]):
            le = LabelEncoder()
            data_df[col] = le.fit_transform(data_df[col])

    # Process data and store in numpy arrays
    X = data_df.drop(['Exited'], axis=1)
    y = data_df['Exited'].to_numpy()

    # Create train and validation set split
    X_train, X_valid, y_train, y_valid = train_test_split(
        X, y, test_size=0.10, random_state=SEED)

    # Define the search space for hyperparameters
    space = {
        'n_estimators': hp.choice('n_estimators', [100, 200, 300, 400, 500]),
        'learning_rate': hp.loguniform('learning_rate', -5, 0)
    }

    # Perform hyperparameter tuning
    trials = Trials()
    best = fmin(fn=objective, space=space, algo=tpe.suggest, max_evals=50, trials=trials)

    # Train the final model with the best hyperparameters
    best_params = {
        'n_estimators': best['n_estimators'],
        'learning_rate': best['learning_rate']
    }
    model = train_model(X_train, y_train, X_valid, y_valid, best_params)

    # Evaluate the model on the validation set and print the result
    y_valid_pred = predict(model, X_valid)
    f1 = compute_metrics_for_classification(y_valid, y_valid_pred)
    print("Final F1 Score on validation set: ", f1)

    # Submit predictions for the test set
    submission_df = pd.read_csv('test.csv')
    submission_df = submission_df.dropna(subset=['Exited'])
    for col in submission_df.columns:
        if pd.api.types.is_object_dtype(submission_df[col]):
            le = LabelEncoder()
            submission_df[col] = le.fit_transform(submission_df[col])
    submission_df = submission_df.drop(['Exited'], axis=1)
    X_submission = submission_df
    y_submission = predict(model, X_submission)
    submit_predictions_for_test_set(y_submission)
