import pandas as pd
from sklearn.metrics import f1_score
import numpy as np
import random
import torch
from sklearn.model_selection import train_test_split
from sklearn.preprocessing import LabelEncoder, StandardScaler
from sklearn.linear_model import LogisticRegression
import optuna

SEED = 42

random.seed(SEED)
torch.manual_seed(SEED)
np.random.seed(SEED)

device = torch.device("cpu")

def compute_metrics_for_classification(y_test, y_test_pred):
    f1 = f1_score(y_test, y_test_pred, average='weighted')
    return f1

def train_model(X_train, y_train, X_valid, y_valid, params):
    scaler = StandardScaler()
    X_train_scaled = scaler.fit_transform(X_train)
    X_valid_scaled = scaler.transform(X_valid)
    
    model = LogisticRegression(random_state=SEED, **params)
    model.fit(X_train_scaled, y_train)
    return model, scaler

def predict(model, X, scaler):
    X_scaled = scaler.transform(X)
    y_pred = model.predict(X_scaled)
    return y_pred

def objective(trial):
    params = {
        'C': trial.suggest_loguniform('C', 1e-4, 1e4),
        'solver': trial.suggest_categorical('solver', ['newton-cg', 'lbfgs', 'liblinear', 'sag', 'saga']),
    }
    
    model, scaler = train_model(X_train, y_train, X_valid, y_valid, params)
    y_valid_pred = predict(model, X_valid, scaler)
    f1 = compute_metrics_for_classification(y_valid, y_valid_pred)
    return f1

if __name__ == '__main__':

    data_df = pd.read_csv('train.csv')
    data_df = data_df.dropna(subset=['Attrition'])
    for col in data_df.columns:
        if pd.api.types.is_object_dtype(data_df[col]):
            le = LabelEncoder()
            data_df[col] = le.fit_transform(data_df[col])

    # 处理数据并存储到 numpy 数组中
    X = data_df.drop(['Attrition'], axis=1)
    y = data_df['Attrition'].to_numpy()

    # 创建训练集和验证集的划分
    X_train, X_valid, y_train, y_valid = train_test_split(
        X, y, test_size=0.10, random_state=SEED)

    # 使用 Optuna 进行超参数优化
    study = optuna.create_study(direction='maximize')
    study.optimize(objective, n_trials=100)

    # 获取最佳超参数
    best_params = study.best_params

    # 使用最佳超参数重新训练模型
    best_model, scaler = train_model(X_train, y_train, X_valid, y_valid, best_params)

    # 在验证集上评估模型并打印结果
    y_valid_pred = predict(best_model, X_valid, scaler)
    f1 = compute_metrics_for_classification(y_valid, y_valid_pred)
    print("Final F1 Score on validation set: ", f1)

    X_submission = submission_df
    y_submission = predict(best_model, X_submission, scaler)
    submit_predictions_for_test_set(y_submission)
