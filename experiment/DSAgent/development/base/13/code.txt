import pandas as pd
from sklearn.metrics import f1_score
import numpy as np
import random
import torch
from sklearn.model_selection import train_test_split
from submission import submit_predictions_for_test_set
from sklearn.preprocessing import LabelEncoder
import torch.nn as nn
import torch.optim as optim

SEED = 42

random.seed(SEED)
torch.manual_seed(SEED)
np.random.seed(SEED)

device = torch.device("cuda" if torch.cuda.is_available() else "cpu")


def compute_metrics_for_classification(y_test, y_test_pred):
    f1 = f1_score(y_test, y_test_pred, average='weighted')
    return f1


def train_model(X_train, y_train, X_valid, y_valid):
    # Convert data to PyTorch tensors
    X_train = torch.tensor(X_train.values, dtype=torch.float32).to(device)
    y_train = torch.tensor(y_train, dtype=torch.float32).view(-1, 1).to(device)
    X_valid = torch.tensor(X_valid.values, dtype=torch.float32).to(device)
    y_valid = torch.tensor(y_valid, dtype=torch.float32).view(-1, 1).to(device)

    # Define the model
    model = nn.Sequential(
        nn.Linear(X_train.shape[1], 128),
        nn.ReLU(),
        nn.Linear(128, 64),
        nn.ReLU(),
        nn.Linear(64, 32),
        nn.ReLU(),
        nn.Linear(32, 1),
        nn.Sigmoid()
    ).to(device)

    # Define the loss function and optimizer
    criterion = nn.BCELoss()
    optimizer = optim.Adam(model.parameters(), lr=0.001)

    # Training loop
    model.train()
    for epoch in range(50):
        optimizer.zero_grad()
        outputs = model(X_train)
        loss = criterion(outputs, y_train)
        loss.backward()
        optimizer.step()

        # Validation
        model.eval()
        with torch.no_grad():
            valid_outputs = model(X_valid)
            valid_loss = criterion(valid_outputs, y_valid)
        model.train()

        print(f'Epoch [{epoch+1}/50], Loss: {loss.item():.4f}, Valid Loss: {valid_loss.item():.4f}')

    return model


def predict(model, X):
    X = torch.tensor(X.values, dtype=torch.float32).to(device)
    model.eval()
    with torch.no_grad():
        y_pred = model(X)
    y_pred = (y_pred > 0.5).float().cpu().numpy()
    return y_pred.flatten()


if __name__ == '__main__':

    data_df = pd.read_csv('train.csv')
    data_df = data_df.dropna(subset=['class'])
    for col in data_df.columns:
        if pd.api.types.is_object_dtype(data_df[col]):
            le = LabelEncoder()
            data_df[col] = le.fit_transform(data_df[col])

    # 处理数据并存储到 numpy 数组中
    X = data_df.drop(['class'], axis=1)
    y = data_df['class'].to_numpy()

    # 创建训练集和验证集的划分
    X_train, X_valid, y_train, y_valid = train_test_split(
        X, y, test_size=0.10, random_state=SEED)

    # 定义并训练模型
    model = train_model(X_train, y_train, X_valid, y_valid)

    # 在验证集上评估模型并打印结果
    y_valid_pred = predict(model, X_valid)
    f1 = compute_metrics_for_classification(y_valid, y_valid_pred)
    print("Final F1 Score on validation set: ", f1)

    # 提交测试集的预测结果
    submission_df = pd.read_csv('test.csv')
    submission_df = submission_df.dropna(subset=['class'])
    for col in submission_df.columns:
        if pd.api.types.is_object_dtype(submission_df[col]):
            le = LabelEncoder()
            submission_df[col] = le.fit_transform(submission_df[col])
    submission_df = submission_df.drop(['class'], axis=1)
    X_submission = submission_df
    y_submission = predict(model, X_submission)
    submit_predictions_for_test_set(y_submission)
