import pandas as pd
from sklearn.metrics import mean_squared_error
import numpy as np
import random

from sklearn.linear_model import LinearRegression
from submission import submit_predictions_for_test_set
from sklearn.preprocessing import LabelEncoder
SEED = 42

    rmse = mean_squared_error(y_test, y_test_pred, squared=False)
    return rmse

    # Define and train the model
    model = LinearRegression()
    model.fit(X_train, y_train)
    return model

    return y_pred

import pandas as pd
from sklearn.metrics import mean_squared_error

    train_data = lgb.Dataset(X_train, label=y_train)
    valid_data = lgb.Dataset(X_valid, label=y_valid, reference=train_data)

        'verbose': 0

    model = lgb.train(params, train_data, valid_sets=[valid_data], early_stopping_rounds=10, verbose_eval=10)
    return model

    return y_pred

import pandas as pd
from sklearn.metrics import mean_squared_error
import numpy as np
import random
import torch
from sklearn.model_selection import train_test_split
from submission import submit_predictions_for_test_set
from sklearn.preprocessing import LabelEncoder
import lightgbm as lgb

SEED = 42

random.seed(SEED)
torch.manual_seed(SEED)
np.random.seed(SEED)

device = torch.device("cpu")


def compute_metrics_for_regression(y_test, y_test_pred):
    mse = mean_squared_error(y_test, y_test_pred)
    rmse = np.sqrt(mse)
    return rmse


def train_model(X_train, y_train, X_valid, y_valid):
    # Define and train the LGBM model
    train_data = lgb.Dataset(X_train, label=y_train)
    valid_data = lgb.Dataset(X_valid, label=y_valid, reference=train_data)
    
    params = {
        'objective': 'regression',
        'metric': 'rmse',
        'boosting_type': 'gbdt',
        'num_leaves': 31,
        'learning_rate': 0.05,
        'feature_fraction': 0.9,
        'bagging_fraction': 0.8,
        'bagging_freq': 5,
        'verbose': 0
    }
    
    model = lgb.train(
        params, 
        train_data, 
        valid_sets=[valid_data], 
        early_stopping_rounds=10, 
        callbacks=[lgb.early_stopping(10), lgb.log_evaluation(10)]
    )
    return model


def predict(model, X):
    # Generate predictions using the trained model
    y_pred = model.predict(X, num_iteration=model.best_iteration)
    return y_pred


if __name__ == '__main__':

    data_df = data_df.dropna(subset=['Hardness'])

    # Process data and store in numpy arrays

    y = data_df.Hardness.to_numpy()

    # Create train and validation set splits
    X_train, X_valid, y_train, y_valid = train_test_split(

    # Evaluate the model on the validation set and print the result
    y_valid_pred = predict(model, X_valid)
    rmse = compute_metrics_for_regression(y_valid, y_valid_pred)
