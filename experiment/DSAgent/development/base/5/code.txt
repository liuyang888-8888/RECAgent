import pandas as pd
from sklearn.metrics import f1_score
import numpy as np
import random
import torch
from sklearn.model_selection import train_test_split, GridSearchCV
from submission import submit_predictions_for_test_set
from sklearn.preprocessing import LabelEncoder
from sklearn.ensemble import RandomForestClassifier

SEED = 42

random.seed(SEED)
torch.manual_seed(SEED)
np.random.seed(SEED)

device = torch.device("cpu")


def compute_metrics_for_classification(y_test, y_test_pred):
    f1 = f1_score(y_test, y_test_pred, average='weighted')
    return f1


def train_model(X_train, y_train, X_valid, y_valid):
    # Define the parameter grid for hyperparameter tuning
    param_grid = {
        'n_estimators': [100, 200],
        'max_depth': [None, 10, 20]
    }

    # Initialize the Random Forest model
    model = RandomForestClassifier(random_state=SEED)

    # Perform grid search to find the best hyperparameters
    grid_search = GridSearchCV(estimator=model, param_grid=param_grid, scoring='f1_weighted', cv=3, n_jobs=-1)
    grid_search.fit(X_train, y_train)

    # Get the best model
    best_model = grid_search.best_estimator_

    # Evaluate the best model on the validation set
    y_valid_pred = best_model.predict(X_valid)
    f1 = compute_metrics_for_classification(y_valid, y_valid_pred)
    print("Best Random Forest Model F1 Score on validation set: ", f1)

    return best_model


def predict(model, X):
    # Generate predictions using the trained model
    y_pred = model.predict(X)
    return y_pred
