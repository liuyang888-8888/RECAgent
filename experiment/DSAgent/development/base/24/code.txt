import pandas as pd
from sklearn.metrics import silhouette_score
import numpy as np
import random
import torch
from sklearn.model_selection import train_test_split
from sklearn.preprocessing import LabelEncoder
from sklearn.cluster import KMeans

SEED = 42

random.seed(SEED)
torch.manual_seed(SEED)
np.random.seed(SEED)

device = torch.device("cpu")

def compute_metrics_for_clustering(X, labels):
    # 计算轮廓系数
    silhouette_avg = silhouette_score(X, labels)
    return silhouette_avg

def find_optimal_k(X, k_range=(2, 11)):
    best_k = k_range[0]
    best_score = -1
    for k in range(*k_range):
        kmeans = KMeans(n_clusters=k, random_state=SEED)
        kmeans.fit(X)
        score = compute_metrics_for_clustering(X, kmeans.labels_)
        if score > best_score:
            best_score = score
            best_k = k
    return best_k, best_score

def train_model(X_train):
    # 确定最优的 k 值
    best_k, _ = find_optimal_k(X_train)
    
    # 使用最优的 k 值训练 K-Means 模型
    model = KMeans(n_clusters=best_k, random_state=SEED)
    model.fit(X_train)
    return model

    data_df = pd.read_csv('train.csv')

import pandas as pd
from sklearn.metrics import silhouette_score
import numpy as np
import random
import torch
from sklearn.model_selection import train_test_split
from sklearn.preprocessing import LabelEncoder, StandardScaler
from sklearn.cluster import KMeans
from sklearn.impute import SimpleImputer

SEED = 42

random.seed(SEED)
torch.manual_seed(SEED)
np.random.seed(SEED)

device = torch.device("cpu")

def compute_metrics_for_clustering(X, labels):
    # 计算轮廓系数
    silhouette_avg = silhouette_score(X, labels)
    return silhouette_avg

def find_optimal_k(X, k_range=(2, 11)):
    best_k = k_range[0]
    best_score = -1
    for k in range(*k_range):
        kmeans = KMeans(n_clusters=k, random_state=SEED)
        kmeans.fit(X)
        score = compute_metrics_for_clustering(X, kmeans.labels_)
        if score > best_score:
            best_score = score
            best_k = k
    return best_k, best_score

def train_model(X_train):
    # Scale the features
    scaler = StandardScaler()
    X_train_scaled = scaler.fit_transform(X_train)
    
    # Determine the optimal k value
    best_k, _ = find_optimal_k(X_train_scaled)
    
    # Train the K-Means model with the optimal k value
    model = KMeans(n_clusters=best_k, random_state=SEED)
    model.fit(X_train_scaled)
    
    return model

if __name__ == '__main__':

    data_df = pd.read_csv('train.csv')

    for col in data_df.columns:
        if pd.api.types.is_object_dtype(data_df[col]):
            le = LabelEncoder()
            data_df[col] = le.fit_transform(data_df[col])

    # 处理数据并存储到 numpy 数组中
    X = data_df.values

    # 处理缺失值
    imputer = SimpleImputer(strategy='mean')
    X = imputer.fit_transform(X)

    # Scale the features
    scaler = StandardScaler()
    X = scaler.fit_transform(X)

    # Define and train the model
    model = train_model(X)

    # Evaluate the model on the training set and print the silhouette score

    print("final silhouette_avg on trained set after scaling: ", silhouette_avg)
