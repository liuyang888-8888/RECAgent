import pandas as pd
from sklearn.metrics import f1_score
import numpy as np
import random
import torch
from sklearn.model_selection import train_test_split
from submission import submit_predictions_for_test_set
from sklearn.preprocessing import LabelEncoder, StandardScaler
from sklearn.linear_model import LogisticRegression
import optuna

SEED = 42

random.seed(SEED)
torch.manual_seed(SEED)
np.random.seed(SEED)

device = torch.device("cpu")


def compute_metrics_for_classification(y_test, y_test_pred):
    f1 = f1_score(y_test, y_test_pred, average='weighted')
    return f1


def objective(trial, X_train, y_train, X_valid, y_valid):
    # Define the hyperparameters to tune
    C = trial.suggest_loguniform('C', 1e-4, 1e4)
    solver = trial.suggest_categorical('solver', ['newton-cg', 'lbfgs', 'liblinear', 'sag', 'saga'])

    # Define and train the logistic regression model
    model = LogisticRegression(C=C, solver=solver, random_state=SEED)
    model.fit(X_train, y_train)

    # Evaluate the model on the validation set
    y_valid_pred = model.predict(X_valid)
    f1 = compute_metrics_for_classification(y_valid, y_valid_pred)
    return f1


def train_model(X_train, y_train, X_valid, y_valid):
    # Define the study for hyperparameter tuning
    study = optuna.create_study(direction='maximize')
    study.optimize(lambda trial: objective(trial, X_train, y_train, X_valid, y_valid), n_trials=100)

    # Get the best hyperparameters
    best_params = study.best_params
    best_C = best_params['C']
    best_solver = best_params['solver']

    # Train the final model with the best hyperparameters on the entire training set
    final_model = LogisticRegression(C=best_C, solver=best_solver, random_state=SEED)
    final_model.fit(X_train, y_train)
    return final_model


def predict(model, X):
    # Predict using the trained model
    y_pred = model.predict(X)
    return y_pred


if __name__ == '__main__':

    data_df = pd.read_csv('train.csv')
    data_df = data_df.dropna(subset=['smoking'])
    for col in data_df.columns:
        if pd.api.types.is_object_dtype(data_df[col]):
            le = LabelEncoder()
            data_df[col] = le.fit_transform(data_df[col])

    # Normalize the input features
    scaler = StandardScaler()
    X = data_df.drop(['smoking'], axis=1)
    X = scaler.fit_transform(X)
    y = data_df['smoking'].to_numpy()

    # Create train and validation splits
    X_train, X_valid, y_train, y_valid = train_test_split(
        X, y, test_size=0.10, random_state=SEED)

    # Define and train the model
    model = train_model(X_train, y_train, X_valid, y_valid)

    # Evaluate the model on the validation set
    y_valid_pred = predict(model, X_valid)
    f1 = compute_metrics_for_classification(y_valid, y_valid_pred)
    print("Final F1 Score on validation set: ", f1)

    # Submit predictions for the test set
    submission_df = pd.read_csv('test.csv')
    submission_df = submission_df.dropna(subset=['smoking'])
    for col in submission_df.columns:
        if pd.api.types.is_object_dtype(submission_df[col]):
            le = LabelEncoder()
            submission_df[col] = le.fit_transform(submission_df[col])
    submission_df = submission_df.drop(['smoking'], axis=1)
    X_submission = scaler.transform(submission_df)
    y_submission = predict(model, X_submission)
    submit_predictions_for_test_set(y_submission)
