import pandas as pd
from sklearn.metrics import silhouette_score
import numpy as np
import random
import torch
from sklearn.model_selection import train_test_split
from submission import submit_predictions_for_test_set
from sklearn.preprocessing import LabelEncoder
from sklearn.cluster import KMeans

SEED = 42

random.seed(SEED)
torch.manual_seed(SEED)
np.random.seed(SEED)

device = torch.device("cpu")

def compute_metrics_for_clustering(X, labels):
    # 计算轮廓系数
    silhouette_avg = silhouette_score(X, labels)
    return silhouette_avg

def train_model(X_train, n_clusters, init_method):
    # Define and train the K-Means model
    model = KMeans(n_clusters=n_clusters, init=init_method, random_state=SEED)
    model.fit(X_train)
    return model

if __name__ == '__main__':

    data_df = pd.read_csv('train.csv')

    for col in data_df.columns:
        if pd.api.types.is_object_dtype(data_df[col]):
            le = LabelEncoder()
            data_df[col] = le.fit_transform(data_df[col])

    # 处理数据并存储到 numpy 数组中
    X = data_df

    best_silhouette_avg = -1
    best_n_clusters = None
    best_init_method = None

    # Try different combinations of n_clusters and init_method
    for n_clusters in [2, 3, 4]:
        for init_method in ['k-means++', 'random']:
            # 定义并训练模型
            model = train_model(X, n_clusters, init_method)

            # 在训练集上评估模型并打印结果
            silhouette_avg = compute_metrics_for_clustering(X, model.labels_)
            print(f"n_clusters={n_clusters}, init_method={init_method}, silhouette_avg={silhouette_avg}")

            # 更新最佳配置
            if silhouette_avg > best_silhouette_avg:
                best_silhouette_avg = silhouette_avg
                best_n_clusters = n_clusters
                best_init_method = init_method

    print(f"Best configuration: n_clusters={best_n_clusters}, init_method={best_init_method}, silhouette_avg={best_silhouette_avg}")

    # 使用最佳配置重新训练模型
    best_model = train_model(X, best_n_clusters, best_init_method)

    # 提交结果
    submit_predictions_for_test_set(best_model.labels_)
