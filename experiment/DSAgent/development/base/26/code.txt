import pandas as pd
from sklearn.metrics import mean_squared_error
import numpy as np
import random
import torch
from pmdarima import auto_arima
from sklearn.model_selection import train_test_split
from submission import submit_predictions_for_test_set
from sklearn.preprocessing import LabelEncoder
SEED = 42

random.seed(SEED)
torch.manual_seed(SEED)
np.random.seed(SEED)

device = torch.device("cpu")


def compute_metrics_for_timeseries(y_fits, y):
    mse = mean_squared_error(y, y_fits)
    return mse


def train_model(X_train):
    # Define and fit the ARIMA model using auto_arima
    model = auto_arima(X_train, seasonal=False, stepwise=True, suppress_warnings=True)
    return model


def predict(model, forecast_steps):
    # Predict the model
    forecast = model.predict(n_periods=forecast_steps)
    return forecast

import pandas as pd
from sklearn.metrics import mean_squared_error
import numpy as np
import random
import torch
from sklearn.model_selection import train_test_split
from submission import submit_predictions_for_test_set
from sklearn.preprocessing import LabelEncoder
from sklearn.linear_model import LinearRegression

SEED = 42

random.seed(SEED)
torch.manual_seed(SEED)
np.random.seed(SEED)

device = torch.device("cpu")


def compute_metrics_for_timeseries(y_fits, y):
    mse = mean_squared_error(y, y_fits)
    return mse


def train_model(X_train):
    # Define and train the model
    X = np.arange(len(X_train)).reshape(-1, 1)
    y = X_train.values
    model = LinearRegression()
    model.fit(X, y)
    return model


def predict(model, forecast_steps):
    # Predict the model
    future_indices = np.arange(len(model.coef_), len(model.coef_) + forecast_steps).reshape(-1, 1)
    forecast = model.predict(future_indices)
    return forecast


if __name__ == '__main__':

    data_df = pd.read_csv('train.csv')
    data_df['date'] = pd.to_datetime(data_df['date'])
    data_df.set_index('date', inplace=True)

    # 对数据按日期排序
    data_df.sort_index(inplace=True)

    # 模型训练
    model = train_model(data_df["Average"])

    # 打印模型评估结果
    mse = compute_metrics_for_timeseries(model.predict(np.arange(len(data_df["Average"])).reshape(-1, 1)), data_df["Average"])
    print("final MSE on trained set: ", mse)

    # 提交预测结果
    # 往后预测 3 步
    forecast_steps = 3
    y_submission = predict(model, forecast_steps=forecast_steps)
    submit_predictions_for_test_set(forecast_steps, y_submission)
