import pandas as pd

# Load the data
train_data_path = r'E:\csv\electricity_sales_train.csv'
train_df = pd.read_csv(train_data_path)

# Copy the DataFrame to avoid modifying the original data
train_df_copy = train_df.copy()

# Display the first few rows to understand the data
print(train_df_copy.head())

# Check data types of each column
print(train_df_copy.dtypes)

# Identify categorical and numerical columns
categorical_cols = train_df_copy.select_dtypes(include=['object']).columns
numerical_cols = train_df_copy.select_dtypes(include=['int64', 'float64']).columns

# Print the identified columns
print("Categorical Columns:", categorical_cols)
print("Numerical Columns:", numerical_cols)

# Apply appropriate preprocessing for categorical and numerical columns
# For example, label encoding for categorical columns and scaling for numerical columns
from sklearn.preprocessing import LabelEncoder, StandardScaler

# Initialize encoders and scalers
label_encoders = {col: LabelEncoder() for col in categorical_cols}
scalers = {col: StandardScaler() for col in numerical_cols}

# Apply label encoding to categorical columns
for col in categorical_cols:
    train_df_copy[col] = label_encoders[col].fit_transform(train_df_copy[col])

# Apply standard scaling to numerical columns
for col in numerical_cols:
    train_df_copy[col] = scalers[col].fit_transform(train_df_copy[[col]])

# Display the preprocessed data
print(train_df_copy.head())

from statsmodels.tsa.arima.model import ARIMA
import numpy as np

# Since the dataset is now indexed by date and we are dealing with time series data,
# we will use the ARIMA model to forecast the 'Sales' column.

# Convert the sales data to a time series
sales_series = train_df_preprocessed['Sales']

# Fit the ARIMA model
# Here we choose an ARIMA(1,1,1) model as a starting point
# p=1: AR term
# d=1: Differencing order
# q=1: MA term
model = ARIMA(sales_series, order=(1, 1, 1))
arima_result = model.fit()

# Print the summary of the model
print(arima_result.summary())
