# Import necessary libraries for preprocessing
from sklearn.preprocessing import StandardScaler, OrdinalEncoder
import numpy as np

# Copy the DataFrame to avoid modifying the original data
train_df_preprocessed = train_df_copy.copy()

# Handling missing values
# For simplicity, we will fill missing values with the median for numerical columns and mode for categorical columns
for col in numerical_cols:
    train_df_preprocessed[col].fillna(train_df_preprocessed[col].median(), inplace=True)

for col in categorical_cols:
    train_df_preprocessed[col].fillna(train_df_preprocessed[col].mode()[0], inplace=True)

# Scaling numerical features
scaler = StandardScaler()
train_df_preprocessed[numerical_cols] = scaler.fit_transform(train_df_preprocessed[numerical_cols])

# Encoding categorical variables
ordinal_encoder = OrdinalEncoder()
train_df_preprocessed[categorical_cols] = ordinal_encoder.fit_transform(train_df_preprocessed[categorical_cols])

# Ensure the label column remains unchanged
train_df_preprocessed['NObeyesdad'] = train_df_copy['NObeyesdad']

# Display the first few rows of the preprocessed DataFrame
print(train_df_preprocessed.head())

from sklearn.cluster import MeanShift, estimate_bandwidth

# Using the preprocessed data from the previous step
X = train_df_preprocessed.drop(columns=['NObeyesdad'])

# Estimate the bandwidth for the Mean Shift algorithm
bandwidth = estimate_bandwidth(X, quantile=0.03)

# Initialize and train the Mean Shift model
mean_shift = MeanShift(bandwidth=bandwidth, max_iter=500)
mean_shift.fit(X)

# Add the cluster labels to the original dataframe
train_df_preprocessed['Cluster'] = mean_shift.labels_

# Display the first few rows of the dataframe with the cluster labels
print(train_df_preprocessed.head())
