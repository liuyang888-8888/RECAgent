import pandas as pd

# Load the dataset
train_df = pd.read_csv(r'E:\csv\census_income_train.csv')

# Create a copy of the DataFrame for preprocessing
train_df_copy = train_df.copy()

# Display the first few rows of the DataFrame to understand its structure
print(train_df_copy.head())

# Check the data types of each column
print(train_df_copy.dtypes)

# Identify categorical and numerical columns
categorical_columns = train_df_copy.select_dtypes(include=['object']).columns
numerical_columns = train_df_copy.select_dtypes(include=['int64', 'float64']).columns

# Print the identified columns
print("Categorical Columns:", categorical_columns)
print("Numerical Columns:", numerical_columns)

# Apply label encoding to categorical columns
from sklearn.preprocessing import LabelEncoder
label_encoder = LabelEncoder()
for col in categorical_columns:
    train_df_copy[col] = label_encoder.fit_transform(train_df_copy[col])

# Apply scaling to numerical columns (excluding the label column if it's numerical)
from sklearn.preprocessing import StandardScaler
scaler = StandardScaler()
if 'label_column' in numerical_columns:  # Replace 'label_column' with the actual label column name
    numerical_columns = numerical_columns.drop('label_column')
train_df_copy[numerical_columns] = scaler.fit_transform(train_df_copy[numerical_columns])

# Display the preprocessed DataFrame
print(train_df_copy.head())

from sklearn.cluster import DBSCAN
import numpy as np

# Since 'income' is the target column, we exclude it from the features for clustering
features_for_clustering = train_df_copy.drop(columns=['income'])

# Apply DBSCAN clustering
dbscan = DBSCAN(eps=0.3, min_samples=10)  # Adjust eps and min_samples based on your dataset
clusters = dbscan.fit_predict(features_for_clustering)

# Add the cluster labels back to the DataFrame
train_df_copy['cluster'] = clusters

# Display the first few rows of the DataFrame with the cluster labels
print(train_df_copy.head())
