# Preprocessing the data
from sklearn.model_selection import train_test_split
from sklearn.preprocessing import StandardScaler, OrdinalEncoder

# Copy the datasets to avoid modifying the original data
train_df_processed = train_df_copy.copy()
test_df_processed = test_df_copy.copy()

# Handling missing values
# Assuming that missing values are represented as NaN
train_df_processed.fillna(train_df_processed.median(), inplace=True)
test_df_processed.fillna(test_df_processed.median(), inplace=True)

# Encoding categorical variables
# Since all columns are numeric, we will assume that 'cp', 'slope', 'ca', 'thal' are categorical
categorical_columns = ['cp', 'slope', 'ca', 'thal']

# Using OrdinalEncoder to encode categorical variables
ordinal_encoder = OrdinalEncoder()
train_df_processed[categorical_columns] = ordinal_encoder.fit_transform(train_df_processed[categorical_columns])
test_df_processed[categorical_columns] = ordinal_encoder.transform(test_df_processed[categorical_columns])

# Splitting the data into features (X) and target (y)
X_train = train_df_processed.drop(columns=['target'])
y_train = train_df_processed['target']
X_test = test_df_processed.drop(columns=['target'])
y_test = test_df_processed['target']

# Scaling numerical features
scaler = StandardScaler()
X_train_scaled = scaler.fit_transform(X_train)
X_test_scaled = scaler.transform(X_test)

# Converting scaled features back to DataFrame
X_train_scaled_df = pd.DataFrame(X_train_scaled, columns=X_train.columns)
X_test_scaled_df = pd.DataFrame(X_test_scaled, columns=X_test.columns)

# Displaying the first few rows of the processed data
print("Processed Training Data:")
display(X_train_scaled_df.head())
print("Processed Testing Data:")
display(X_test_scaled_df.head())
