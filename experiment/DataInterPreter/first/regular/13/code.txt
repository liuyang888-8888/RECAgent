from sklearn.preprocessing import LabelEncoder

# Copy the datasets to avoid modifying the original data
train_df_preprocessed = train_df_copy.copy()
test_df_preprocessed = test_df_copy.copy()

# Initialize the LabelEncoder
label_encoder = LabelEncoder()

# Encode categorical columns
for column in categorical_columns:
    # Fit the encoder on the training data and transform both training and testing data
    train_df_preprocessed[column] = label_encoder.fit_transform(train_df_preprocessed[column])
    test_df_preprocessed[column] = label_encoder.transform(test_df_preprocessed[column])

# Split the data into features (X) and labels (y)
X_train = train_df_preprocessed.drop(columns=['class'])
y_train = train_df_preprocessed['class']
X_test = test_df_preprocessed.drop(columns=['class'])
y_test = test_df_preprocessed['class']

# Display the first few rows of the preprocessed data
print("Preprocessed Training Data Features:")

print("\nPreprocessed Training Data Labels:")
print(y_train.head())
print("\nPreprocessed Testing Data Features:")
print(X_test.head())
print("\nPreprocessed Testing Data Labels:")
print(y_test.head())

from sklearn.ensemble import AdaBoostClassifier
from sklearn.metrics import accuracy_score, classification_report

# Initialize the AdaBoost classifier
ada_clf = AdaBoostClassifier(n_estimators=100, learning_rate=0.1, random_state=42)

# Train the model
ada_clf.fit(X_train, y_train)

# Predict on the training set
y_train_pred = ada_clf.predict(X_train)
print("Training Accuracy:", accuracy_score(y_train, y_train_pred))
print("Training Classification Report:\n", classification_report(y_train, y_train_pred))

# Predict on the testing set
y_test_pred = ada_clf.predict(X_test)
print("Testing Accuracy:", accuracy_score(y_test, y_test_pred))
print("Testing Classification Report:\n", classification_report(y_test, y_test_pred))
