import statsmodels.api as sm
from statsmodels.tsa.vector_ar.vecm import coint_johansen
from statsmodels.tsa.stattools import adfuller
import matplotlib.pyplot as plt

# Check for stationarity using the Augmented Dickey-Fuller test
def check_stationarity(series):
    result = adfuller(series)
    print(f'ADF Statistic: {result[0]}')
    print(f'p-value: {result[1]}')
    for key, value in result[4].items():
        print('Critical Values:')
        print(f'   {key}, {value}')
    if result[1] <= 0.05:
        print("The series is stationary.")
    else:
        print("The series is non-stationary.")

# Apply differencing if necessary to make the series stationary
def make_stationary(df):
    stationary_df = df.copy()
    for col in df.columns:
        if col != 'Date':
            check_stationarity(df[col])
            if adfuller(df[col])[1] > 0.05:
                stationary_df[col] = df[col].diff().dropna()
                check_stationarity(stationary_df[col])
    return stationary_df.dropna()

# Make the data stationary
stationary_df = make_stationary(train_df_preprocessed[['Open', 'Close', 'High', 'Low']])

# Perform Johansen cointegration test
def johansen_cointegration_test(df):
    result = coint_johansen(df, det_order=0, k_ar_diff=1)
    print("Johansen Cointegration Test Results:")
    print(f"Trace statistic: {result.lr1}")
    print(f"Critical values (90%, 95%, 99%): {result.cvt}")
    print(f"Max eigenvalue statistic: {result.lr2}")
    print(f"Critical values (90%, 95%, 99%): {result.cvm}")

johansen_cointegration_test(stationary_df)

# Fit the VAR model
model = sm.tsa.VAR(stationary_df)
lags = model.select_order(maxlags=12)
print(lags.summary())

# Select the optimal lag order
optimal_lag = lags.selected_orders['aic']
var_model = model.fit(optimal_lag)
print(var_model.summary())

# Plot the results
fig, axes = plt.subplots(nrows=2, ncols=2, figsize=(16, 8))
for i, ax in enumerate(axes.flatten()):
    var_model.plot_forecast(steps=10, ax=ax, plot_insample=True, series=i)
plt.tight_layout()
plt.show()
