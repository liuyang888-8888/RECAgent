import pandas as pd

# Load the datasets
train_df = pd.read_csv('E:\\csv\\mohs_hardness_train.csv')
test_df = pd.read_csv('E:\\csv\\mohs_hardness_test.csv')

# Copy the DataFrames to avoid modifying the original datasets
train_df_copy = train_df.copy()
test_df_copy = test_df.copy()

# Display the first few rows of the datasets to understand the structure
print("Train DataFrame:")
print(train_df_copy.head())
print("\nTest DataFrame:")
print(test_df_copy.head())

# Check data types of each column
print("\nData types in Train DataFrame:")
print(train_df_copy.dtypes)
print("\nData types in Test DataFrame:")
print(test_df_copy.dtypes)

# Identify categorical and numerical columns
categorical_columns = train_df_copy.select_dtypes(include=['object']).columns
numerical_columns = train_df_copy.select_dtypes(include=['int64', 'float64']).columns

# Print the identified columns
print("\nCategorical Columns:")
print(categorical_columns)
print("\nNumerical Columns:")
print(numerical_columns)

# Apply appropriate preprocessing for categorical and numerical columns
# For categorical columns, we can use label encoding instead of one-hot encoding
from sklearn.preprocessing import LabelEncoder

label_encoder = LabelEncoder()

for col in categorical_columns:
    train_df_copy[col] = label_encoder.fit_transform(train_df_copy[col])
    test_df_copy[col] = label_encoder.transform(test_df_copy[col])

# For numerical columns, we can apply scaling if necessary
from sklearn.preprocessing import StandardScaler

scaler = StandardScaler()

for col in numerical_columns:
    if col != 'label':  # Ensure the label column is not scaled
        train_df_copy[col] = scaler.fit_transform(train_df_copy[[col]])
        test_df_copy[col] = scaler.transform(test_df_copy[[col]])

# Display the preprocessed datasets
print("\nPreprocessed Train DataFrame:")
print(train_df_copy.head())
print("\nPreprocessed Test DataFrame:")
print(test_df_copy.head())

# Evaluate the model
