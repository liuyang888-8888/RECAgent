from sklearn.preprocessing import LabelEncoder, StandardScaler
import numpy as np

# Copy the DataFrame
train_df_processed = train_df_copy.copy()

# Handling missing values
# For numerical columns, fill with median
for col in numerical_columns:
    if train_df_processed[col].isnull().sum() > 0:
        train_df_processed[col].fillna(train_df_processed[col].median(), inplace=True)

# For categorical columns, fill with mode
for col in categorical_columns:
    if train_df_processed[col].isnull().sum() > 0:
        train_df_processed[col].fillna(train_df_processed[col].mode()[0], inplace=True)

# Encoding categorical variables
label_encoders = {}
for col in categorical_columns:
    le = LabelEncoder()
    train_df_processed[col] = le.fit_transform(train_df_processed[col])
    label_encoders[col] = le

# Scaling numerical features
scalers = {}
for col in numerical_columns:
    if col not in ['PassengerId', 'Survived']:
        scaler = StandardScaler()
        train_df_processed[col] = scaler.fit_transform(train_df_processed[[col]])
        scalers[col] = scaler

# Display the first few rows of the processed DataFrame
print(train_df_processed.head())
