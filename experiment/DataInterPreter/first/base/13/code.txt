from sklearn.preprocessing import LabelEncoder

# Copy the datasets to avoid modifying the original data
train_df_processed = train_df_copy.copy()
test_df_processed = test_df_copy.copy()

# Initialize the LabelEncoder
label_encoder = LabelEncoder()

# Encode categorical variables
for column in categorical_columns:
    # Fit the encoder on the training set and transform both training and testing sets
    train_df_processed[column] = label_encoder.fit_transform(train_df_processed[column].astype(str))
    test_df_processed[column] = label_encoder.transform(test_df_processed[column].astype(str))

# Split the datasets into features (X) and target (y)
X_train = train_df_processed.drop(columns=['class'])
y_train = train_df_processed['class']
X_test = test_df_processed.drop(columns=['class'])
y_test = test_df_processed['class']

# Display the first few rows of the processed datasets

print(y_train.head())
print("\nProcessed Testing Data Features (X_test):")
print(X_test.head())
print("\nProcessed Testing Data Target (y_test):")
print(y_test.head())

from xgboost import XGBClassifier
from sklearn.metrics import accuracy_score, classification_report

# Initialize the XGBoost classifier with some hyperparameters
xgb_model = XGBClassifier(
    n_estimators=1000,
    max_depth=6,
    learning_rate=0.05,
    subsample=0.8,
    colsample_bytree=0.8,
    use_label_encoder=False,
    eval_metric='logloss'
)

# Train the model
xgb_model.fit(X_train, y_train)

# Predict on the training set
y_train_pred = xgb_model.predict(X_train)
print("Training Accuracy:", accuracy_score(y_train, y_train_pred))
print("Training Classification Report:\n", classification_report(y_train, y_train_pred))

# Predict on the testing set
y_test_pred = xgb_model.predict(X_test)
print("Testing Accuracy:", accuracy_score(y_test, y_test_pred))
print("Testing Classification Report:\n", classification_report(y_test, y_test_pred))
