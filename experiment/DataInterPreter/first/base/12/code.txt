test_df = pd.read_csv('path_to_test_dataset.csv')

# Copy the DataFrames to avoid modifying the original datasets
train_df_copy = train_df.copy()
test_df_copy = test_df.copy()

# Check data types of each column
print("Train dataset column data types:")
print(train_df_copy.dtypes)
print("\nTest dataset column data types:")
print(test_df_copy.dtypes)

# Assuming all columns except the label column are numerical and need scaling
# Identify the label column (assuming it's named 'label')
label_column = 'label'

# Select columns to scale (all columns except the label column)
columns_to_scale = [col for col in train_df_copy.columns if col != label_column]

# Initialize the scaler
scaler = StandardScaler()

# Fit the scaler on the training data and transform both training and testing data
train_df_copy[columns_to_scale] = scaler.fit_transform(train_df_copy[columns_to_scale])
test_df_copy[columns_to_scale] = scaler.transform(test_df_copy[columns_to_scale])

# Verify the preprocessing
print("\nPreprocessed Train dataset head:")
print(train_df_copy.head())
print("\nPreprocessed Test dataset head:")
print(test_df_copy.head())
