import pandas as pd

# Load the datasets
train_df = pd.read_csv(r'E:\csv\wild_blueberry_train.csv')
test_df = pd.read_csv(r'E:\csv\wild_blueberry_test.csv')

# Create copies of the datasets to work on
train_df_copy = train_df.copy()
test_df_copy = test_df.copy()

# Function to preprocess the data
def preprocess_data(df):
    # Monitor data types per column
    print(df.dtypes)
    
    # Identify categorical and numerical columns
    categorical_cols = df.select_dtypes(include=['object']).columns
    numerical_cols = df.select_dtypes(include=['int64', 'float64']).columns
    
    # Handle categorical data (using label encoding as an alternative to one-hot encoding)
    for col in categorical_cols:
        df[col] = df[col].astype('category').cat.codes
    
    # Handle numerical data (scaling if necessary, but not the label column)
    from sklearn.preprocessing import StandardScaler
    scaler = StandardScaler()
    if 'yield' in numerical_cols:  # Assuming 'yield' is the label column
        numerical_cols = numerical_cols.drop('yield')
    df[numerical_cols] = scaler.fit_transform(df[numerical_cols])
    
    return df

# Preprocess the training and testing datasets
train_df_processed = preprocess_data(train_df_copy)
test_df_processed = preprocess_data(test_df_copy)

# Display the first few rows of the processed datasets
train_df_processed.head(), test_df_processed.head()
