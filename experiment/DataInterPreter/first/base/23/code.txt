import numpy as np
from sklearn.preprocessing import StandardScaler, OrdinalEncoder

# Copy the DataFrame to avoid modifying the original data
train_df_preprocessed = train_df_copy.copy()

# Handling missing values
# For numerical columns, fill with the mean
for col in numerical_columns:
    train_df_preprocessed[col].fillna(train_df_preprocessed[col].mean(), inplace=True)

# For categorical columns, fill with the most frequent value
for col in categorical_columns:
    train_df_preprocessed[col].fillna(train_df_preprocessed[col].mode()[0], inplace=True)

# Encoding categorical variables
ordinal_encoder = OrdinalEncoder()
train_df_preprocessed[categorical_columns] = ordinal_encoder.fit_transform(train_df_preprocessed[categorical_columns])

# Normalizing numerical features

from sklearn.cluster import KMeans
from sklearn.metrics import silhouette_score

# Selecting the features for clustering
X = train_df_selected.values

# Training the KMeans model
kmeans = KMeans(n_clusters=3, random_state=42)  # Assuming 3 clusters based on the NObeyesdad categories
kmeans.fit(X)

# Adding the cluster labels to the dataframe
train_df_selected['Cluster'] = kmeans.labels_

# Evaluating the model using Silhouette Score
silhouette_avg = silhouette_score(X, kmeans.labels_)
print(f'Silhouette Score: {silhouette_avg}')

# Displaying the first few rows of the dataframe with the cluster labels
