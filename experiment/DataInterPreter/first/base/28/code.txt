from statsmodels.tsa.statespace.sarimax import SARIMAX
import numpy as np

# Convert the 'Date' column to be the index
train_set = train_set.set_index('Date')
validation_set = validation_set.set_index('Date')

# Define the SARIMA model
model = SARIMAX(train_set['Sales'], order=(1, 1, 1), seasonal_order=(1, 1, 1, 12))

# Fit the model
model_fit = model.fit(disp=False)

# Summary of the model
print(model_fit.summary())

# Forecasting on the validation set
forecast = model_fit.forecast(steps=len(validation_set))
forecast = pd.DataFrame(forecast, index=validation_set.index, columns=['Forecast'])

# Calculate RMSE
from sklearn.metrics import mean_squared_error
rmse = np.sqrt(mean_squared_error(validation_set['Sales'], forecast['Forecast']))
print(f"RMSE: {rmse}")

# Plotting the results
import matplotlib.pyplot as plt
plt.figure(figsize=(10, 6))
plt.plot(train_set['Sales'], label='Train')
plt.plot(validation_set['Sales'], label='Validation')
plt.plot(forecast['Forecast'], label='Forecast')
plt.legend()
plt.show()
