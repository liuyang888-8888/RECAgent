import pandas as pd

# Load the datasets
train_df = pd.read_csv('E:\\csv\\mohs_hardness_train.csv')
test_df = pd.read_csv('E:\\csv\\mohs_hardness_test.csv')

# Copy the DataFrames to ensure we don't modify the original data
train_df_copy = train_df.copy()
test_df_copy = test_df.copy()

# Function to preprocess the data
def preprocess_data(df):
    # Monitor data types per column
    print("Data types before preprocessing:")
    print(df.dtypes)
    
    # Identify categorical and numerical columns
    categorical_cols = df.select_dtypes(include=['object']).columns
    numerical_cols = df.select_dtypes(include=['int64', 'float64']).columns
    
    # Apply appropriate methods for categorical data
    for col in categorical_cols:
        if col != 'label':  # Ensure we do not touch the label column
            df[col] = df[col].astype('category').cat.codes
    
    # Apply appropriate methods for numerical data
    for col in numerical_cols:
        if col != 'label':  # Ensure we do not touch the label column
            df[col] = (df[col] - df[col].mean()) / df[col].std()
    
    # Monitor data types after preprocessing
    print("Data types after preprocessing:")
    print(df.dtypes)
    
    return df

# Preprocess the training and testing datasets
train_df_preprocessed = preprocess_data(train_df_copy)
test_df_preprocessed = preprocess_data(test_df_copy)

# Display the first few rows of the preprocessed datasets
print("Preprocessed Training Data:")
print(train_df_preprocessed.head())
print("Preprocessed Testing Data:")
print(test_df_preprocessed.head())

def handle_missing_values(df):
    # Check for missing values
    print("Missing values before handling:")
    print(df.isnull().sum())
    
    # Fill missing values with median for numerical columns
    numerical_cols = df.select_dtypes(include=['int64', 'float64']).columns
    for col in numerical_cols:
        if col != 'label':
            df[col].fillna(df[col].median(), inplace=True)
    
    # Check for missing values after handling
    print("Missing values after handling:")
    print(df.isnull().sum())
    return df

from sklearn.model_selection import train_test_split, GridSearchCV
from xgboost import XGBRegressor
from sklearn.metrics import mean_squared_error, r2_score

# Splitting the training data into training and validation sets
X_train_val, X_val, y_train_val, y_val = train_test_split(X_train, y_train, test_size=0.2, random_state=42)

# Initialize the XGBoost Regressor
xgb_model = XGBRegressor(objective='reg:squarederror', random_state=42)

# Define the parameter grid for Grid Search
param_grid = {
    'n_estimators': [100, 200, 300],
    'max_depth': [3, 5, 7],
    'learning_rate': [0.01, 0.1, 0.2],
    'subsample': [0.8, 1.0],
    'colsample_bytree': [0.8, 1.0]
}

# Perform Grid Search to find the best parameters
grid_search = GridSearchCV(estimator=xgb_model, param_grid=param_grid, cv=3, scoring='neg_mean_squared_error', n_jobs=-1)
grid_search.fit(X_train_val, y_train_val)

# Get the best model
best_model = grid_search.best_estimator_

# Evaluate the best model on the validation set
y_val_pred = best_model.predict(X_val)
val_mse = mean_squared_error(y_val, y_val_pred)
val_r2 = r2_score(y_val, y_val_pred)

print(f"Best Model Parameters: {grid_search.best_params_}")
print(f"Validation MSE: {val_mse}")
print(f"Validation R2 Score: {val_r2}")

# Train the best model on the entire training set
best_model.fit(X_train, y_train)

# Predict on the test set
y_test_pred = best_model.predict(X_test)

# Evaluate the model on the test set
test_mse = mean_squared_error(y_test, y_test_pred)
test_r2 = r2_score(y_test, y_test_pred)

print(f"Test MSE: {test_mse}")
print(f"Test R2 Score: {test_r2}")
