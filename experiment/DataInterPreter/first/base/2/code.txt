import pandas as pd

# Load the datasets
train_df = pd.read_csv('E:\\csv\\cirrhosis_patient_train.csv')
test_df = pd.read_csv('E:\\csv\\cirrhosis_patient_test.csv')

# Copy the DataFrames to ensure we don't modify the original data
train_df_copy = train_df.copy()
test_df_copy = test_df.copy()

# Display the first few rows of the datasets to understand the structure
print("Train DataFrame:")
print(train_df_copy.head())
print("\nTest DataFrame:")
print(test_df_copy.head())

# Check data types of each column
print("\nData types in Train DataFrame:")
print(train_df_copy.dtypes)
print("\nData types in Test DataFrame:")
print(test_df_copy.dtypes)

# Identify categorical and numerical columns
categorical_columns = train_df_copy.select_dtypes(include=['object']).columns
numerical_columns = train_df_copy.select_dtypes(include=['int64', 'float64']).columns

# Print the identified columns
print("\nCategorical Columns:")
print(categorical_columns)
print("\nNumerical Columns:")
print(numerical_columns)

# Apply appropriate preprocessing for categorical columns
for col in categorical_columns:
    # Use label encoding for categorical columns
    unique_values = train_df_copy[col].unique()
    value_map = {value: idx for idx, value in enumerate(unique_values)}
    train_df_copy[col] = train_df_copy[col].map(value_map)
    test_df_copy[col] = test_df_copy[col].map(value_map)

# Apply appropriate preprocessing for numerical columns
from sklearn.preprocessing import StandardScaler

scaler = StandardScaler()
train_df_copy[numerical_columns] = scaler.fit_transform(train_df_copy[numerical_columns])
test_df_copy[numerical_columns] = scaler.transform(test_df_copy[numerical_columns])

# Verify the preprocessing
print("\nPreprocessed Train DataFrame:")
print(train_df_copy.head())
print("\nPreprocessed Test DataFrame:")
print(test_df_copy.head())

# Copy the DataFrames to ensure we don't modify the original data
train_df_processed = train_df_copy.copy()
test_df_processed = test_df_copy.copy()

# Handling missing values
# For numerical columns, fill with median
numerical_columns = ['id', 'N_Days', 'Age', 'Bilirubin', 'Cholesterol', 'Albumin', 'Copper', 'Alk_Phos', 'SGOT', 'Tryglicerides', 'Platelets', 'Prothrombin', 'Stage']
for col in numerical_columns:
    median_value = train_df_processed[col].median()
    train_df_processed[col].fillna(median_value, inplace=True)
    test_df_processed[col].fillna(median_value, inplace=True)

# For categorical columns, fill with mode
categorical_columns = ['Drug', 'Sex', 'Ascites', 'Hepatomegaly', 'Spiders', 'Edema', 'Status']
for col in categorical_columns:
    mode_value = train_df_processed[col].mode()[0]
    train_df_processed[col].fillna(mode_value, inplace=True)
    test_df_processed[col].fillna(mode_value, inplace=True)

# Splitting the data into features (X) and target (y)
X_train = train_df_processed.drop(columns=['Status'])
y_train = train_df_processed['Status']

X_test = test_df_processed.drop(columns=['Status'])
y_test = test_df_processed['Status']

# Display the first few rows of the processed DataFrames
print("Processed Train DataFrame:")
print(X_train.head())
print("\nProcessed Test DataFrame:")
print(X_test.head())

from xgboost import XGBClassifier
from sklearn.metrics import accuracy_score, classification_report

# Initialize the XGBoost classifier with some hyperparameters
xgb_model = XGBClassifier(
    n_estimators=1000,
    max_depth=7,
    learning_rate=0.05,
    subsample=0.8,
    colsample_bytree=0.8,
    use_label_encoder=False,
    eval_metric='mlogloss'
)

# Fit the model on the training data
xgb_model.fit(X_train, y_train)

# Predict the labels for the test set
y_pred = xgb_model.predict(X_test)

# Evaluate the model
accuracy = accuracy_score(y_test, y_pred)
report = classification_report(y_test, y_pred)

print(f"Model Accuracy: {accuracy}")
print(report)
