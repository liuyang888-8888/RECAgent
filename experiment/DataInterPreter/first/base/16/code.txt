for col in categorical_columns:
    train_df_copy[col] = label_encoder.fit_transform(train_df_copy[col])
    test_df_copy[col] = label_encoder.transform(test_df_copy[col])

# For numerical columns, we can apply scaling if necessary

scaler = MinMaxScaler()

for col in numerical_columns:
    if col != 'label':  # Ensure the label column is not scaled
        train_df_copy[[col]] = scaler.fit_transform(train_df_copy[[col]])

from xgboost import XGBRegressor
from sklearn.metrics import mean_squared_error, r2_score

# Initialize the XGBoost Regressor with some hyperparameters
xgb_model = XGBRegressor(
    n_estimators=1000,
    max_depth=7,
    learning_rate=0.05,
    subsample=0.8,
    colsample_bytree=0.8,
    random_state=42
)

# Fit the model on the training data
xgb_model.fit(X_train, y_train)

# Predict on the training set
y_train_pred = xgb_model.predict(X_train)

# Predict on the test set
y_test_pred = xgb_model.predict(X_test)

# Evaluate the model
train_mse = mean_squared_error(y_train, y_train_pred)
test_mse = mean_squared_error(y_test, y_test_pred)
train_r2 = r2_score(y_train, y_train_pred)
test_r2 = r2_score(y_test, y_test_pred)

print(f"Training MSE: {train_mse:.4f}, Training R2: {train_r2:.4f}")
print(f"Test MSE: {test_mse:.4f}, Test R2: {test_r2:.4f}")
