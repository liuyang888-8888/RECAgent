from xgboost import XGBRegressor
from sklearn.metrics import mean_squared_error, r2_score

# Convert datetime to numeric for XGBoost
train_X['date'] = train_X['date'].map(pd.Timestamp.timestamp)
test_X['date'] = test_X['date'].map(pd.Timestamp.timestamp)

# Initialize the model
model = XGBRegressor(objective='reg:squarederror', n_estimators=1000, max_depth=7, learning_rate=0.05, subsample=0.8, colsample_bytree=0.8)

# Train the model
model.fit(train_X, train_y)

# Make predictions
train_predictions = model.predict(train_X)
test_predictions = model.predict(test_X)

# Evaluate the model
train_rmse = mean_squared_error(train_y, train_predictions, squared=False)
test_rmse = mean_squared_error(test_y, test_predictions, squared=False)
train_r2 = r2_score(train_y, train_predictions)
test_r2 = r2_score(test_y, test_predictions)

print(f"Train RMSE: {train_rmse}")
print(f"Test RMSE: {test_rmse}")
print(f"Train R2 Score: {train_r2}")
print(f"Test R2 Score: {test_r2}")
