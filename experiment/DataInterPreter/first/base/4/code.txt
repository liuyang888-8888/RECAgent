import pandas as pd

# Load the datasets
train_df = pd.read_csv(r'E:\csv\employee_attrition_train.csv')
test_df = pd.read_csv(r'E:\csv\employee_attrition_test.csv')

# Create copies of the datasets to work on
train_df_copy = train_df.copy()
test_df_copy = test_df.copy()

# Function to preprocess the data
def preprocess_data(df):
    # Identify categorical and numerical columns
    categorical_cols = df.select_dtypes(include=['object']).columns
    numerical_cols = df.select_dtypes(include=['int64', 'float64']).columns
    
    # Handle categorical data using label encoding
    from sklearn.preprocessing import LabelEncoder
    le = LabelEncoder()
    for col in categorical_cols:
        df[col] = le.fit_transform(df[col])
    
    # Handle numerical data: no scaling for now, just ensure they are numeric
    for col in numerical_cols:
        df[col] = pd.to_numeric(df[col], errors='coerce')
    
    # Fill missing values with median for numerical columns
    for col in numerical_cols:
        df[col].fillna(df[col].median(), inplace=True)
    
    return df

# Apply preprocessing to both train and test datasets
train_df_processed = preprocess_data(train_df_copy)
test_df_processed = preprocess_data(test_df_copy)
