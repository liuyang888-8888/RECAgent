from sklearn.cluster import MeanShift, estimate_bandwidth

# Using the selected features dataframe for clustering
X_cluster = train_df_selected.values

# Estimate the bandwidth for the Mean Shift algorithm
bandwidth = estimate_bandwidth(X_cluster, quantile=0.2)

# Initialize and fit the Mean Shift model
ms = MeanShift(bandwidth=bandwidth, bin_seeding=True)
ms.fit(X_cluster)

# Get the cluster labels
cluster_labels = ms.labels_

# Add the cluster labels to the original dataframe
train_df_selected['Cluster'] = cluster_labels

# Display the first few rows of the dataframe with cluster labels
print(train_df_selected.head())
