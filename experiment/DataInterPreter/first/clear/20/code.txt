from sklearn.preprocessing import StandardScaler, OrdinalEncoder
import numpy as np

# Copy the datasets to avoid modifying the original data
train_df_preprocessed = train_df_copy.copy()
test_df_preprocessed = test_df_copy.copy()

# Check for missing values and handle them
def handle_missing_values(df):
    # Fill missing values with the median for numerical columns
    for col in df.select_dtypes(include=[np.number]).columns:
        if df[col].isnull().sum() > 0:
            df[col].fillna(df[col].median(), inplace=True)
    return df

train_df_preprocessed = handle_missing_values(train_df_preprocessed)
test_df_preprocessed = handle_missing_values(test_df_preprocessed)

# Encode categorical variables using Ordinal Encoding
def encode_categorical_variables(df):
    # Identify categorical columns (if any)
    categorical_cols = ['season', 'weathersit']
    encoder = OrdinalEncoder()
    df[categorical_cols] = encoder.fit_transform(df[categorical_cols])
    return df

train_df_preprocessed = encode_categorical_variables(train_df_preprocessed)
test_df_preprocessed = encode_categorical_variables(test_df_preprocessed)

# Scale numerical features (excluding the label column 'cnt')
def scale_numerical_features(df):
    scaler = StandardScaler()
    numerical_cols = df.select_dtypes(include=[np.number]).drop(columns=['cnt']).columns
    df[numerical_cols] = scaler.fit_transform(df[numerical_cols])
    return df

train_df_preprocessed = scale_numerical_features(train_df_preprocessed)
test_df_preprocessed = scale_numerical_features(test_df_preprocessed)

# Display the first few rows of the preprocessed data
print("Preprocessed Training Data:")
display(train_df_preprocessed.head())
print("Preprocessed Testing Data:")
display(test_df_preprocessed.head())

from xgboost import XGBRegressor
from sklearn.metrics import mean_squared_error, r2_score

# Initialize the XGBoost Regressor with specified parameters
xgb_model = XGBRegressor(n_estimators=8, learning_rate=0.01, objective='reg:squarederror')

# Train the model
xgb_model.fit(X_train, y_train)

# Make predictions on the training set to evaluate the model
y_train_pred = xgb_model.predict(X_train)
train_rmse = np.sqrt(mean_squared_error(y_train, y_train_pred))
train_r2 = r2_score(y_train, y_train_pred)

# Make predictions on the test set to evaluate the model
y_test_pred = xgb_model.predict(X_test)
test_rmse = np.sqrt(mean_squared_error(y_test, y_test_pred))
test_r2 = r2_score(y_test, y_test_pred)

# Print the evaluation metrics
print(f"Training RMSE: {train_rmse}, R2 Score: {train_r2}")
print(f"Testing RMSE: {test_rmse}, R2 Score: {test_r2}")
