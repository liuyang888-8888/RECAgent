from sklearn.preprocessing import LabelEncoder

# Handling missing values
def handle_missing_values(df):
    # Fill missing numeric values with median
    numeric_cols = df.select_dtypes(include=['number']).columns
    df[numeric_cols] = df[numeric_cols].fillna(df[numeric_cols].median())
    
    # Fill missing categorical values with mode
    categorical_cols = df.select_dtypes(include=['object']).columns
    df[categorical_cols] = df[categorical_cols].fillna(df[categorical_cols].mode().iloc[0])
    return df

# Encoding categorical variables
def encode_categorical_variables(df):
    le = LabelEncoder()
    categorical_cols = df.select_dtypes(include=['object']).columns
    for col in categorical_cols:
        if col != 'Status':  # Exclude the label column from encoding
            df[col] = le.fit_transform(df[col])
    return df

# Apply preprocessing to both training and testing datasets
train_df_processed = handle_missing_values(train_df_copy.copy())
train_df_processed = encode_categorical_variables(train_df_processed)
