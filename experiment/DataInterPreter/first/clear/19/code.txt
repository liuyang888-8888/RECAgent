from sklearn.preprocessing import LabelEncoder, StandardScaler
import numpy as np

# Handling missing values
def handle_missing_values(df):
    df.fillna(df.median(), inplace=True)  # Fill numerical missing values with median
    df.fillna(df.mode().iloc[0], inplace=True)  # Fill categorical missing values with mode
    return df

# Encoding categorical variables
def encode_categorical_variables(df, categorical_columns):
    label_encoder = LabelEncoder()
    for col in categorical_columns:
        df[col] = label_encoder.fit_transform(df[col])
    return df

# Scaling numerical features
def scale_numerical_features(df, numerical_columns):
    scaler = StandardScaler()
    df[numerical_columns] = scaler.fit_transform(df[numerical_columns])
    return df

# Apply preprocessing to both training and testing datasets
train_df_preprocessed = train_df_copy.copy()
test_df_preprocessed = test_df_copy.copy()

# Handle missing values
train_df_preprocessed = handle_missing_values(train_df_preprocessed)
test_df_preprocessed = handle_missing_values(test_df_preprocessed)

# Encode categorical variables
train_df_preprocessed = encode_categorical_variables(train_df_preprocessed, categorical_columns)
test_df_preprocessed = encode_categorical_variables(test_df_preprocessed, categorical_columns)

# Scale numerical features
# Exclude the label columns (G1, G2, G3) from scaling
numerical_columns_to_scale = [col for col in numerical_columns if col not in ['G1', 'G2', 'G3']]
train_df_preprocessed = scale_numerical_features(train_df_preprocessed, numerical_columns_to_scale)
test_df_preprocessed = scale_numerical_features(test_df_preprocessed, numerical_columns_to_scale)

# Display the first few rows of the preprocessed datasets
