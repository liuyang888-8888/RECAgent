from statsmodels.tsa.vector_ar.vecm import coint_johansen
from statsmodels.tsa.vector_ar.var_model import VAR
import numpy as np

# Selecting the 'High' column for the VAR model
data_for_var = train_high

# Check for stationarity and perform differencing if necessary
def check_stationarity(data):
    result = coint_johansen(data, det_order=0, k_ar_diff=1)
    return result.lr1[0] > result.cvt[0][1]

# Differencing the data if it is not stationary
if not check_stationarity(data_for_var):
    data_for_var = data_for_var.diff().dropna()

# Fit the VAR model
model = VAR(data_for_var)
results = model.fit(ic='bic')  # Using BIC for optimal lag selection

# Print the summary of the model
print(results.summary())
