import pandas as pd

# Load the data
train_data = pd.read_csv(r'E:\csv\monster_train.csv')
test_data = pd.read_csv(r'E:\csv\monster_test.csv')

# Copy the DataFrames to avoid modifying the original data
train_data_copy = train_data.copy()
test_data_copy = test_data.copy()

# Display the first few rows of the train and test data to understand the structure
print("Train Data:")
print(train_data_copy.head())
print("\nTest Data:")
print(test_data_copy.head())

# Check the data types of each column
print("\nData Types of Train Data:")
print(train_data_copy.dtypes)
print("\nData Types of Test Data:")
print(test_data_copy.dtypes)

# Identify categorical and numerical columns
categorical_columns = train_data_copy.select_dtypes(include=['object']).columns
numerical_columns = train_data_copy.select_dtypes(include=['int64', 'float64']).columns

# Print the identified columns
print("\nCategorical Columns:")
print(categorical_columns)
print("\nNumerical Columns:")
print(numerical_columns)

# Apply appropriate preprocessing methods
# For categorical columns, use label encoding instead of one-hot encoding
from sklearn.preprocessing import LabelEncoder

label_encoder = LabelEncoder()

for col in categorical_columns:
    train_data_copy[col] = label_encoder.fit_transform(train_data_copy[col])
    test_data_copy[col] = label_encoder.transform(test_data_copy[col])

# For numerical columns, apply scaling if necessary
from sklearn.preprocessing import StandardScaler

scaler = StandardScaler()

for col in numerical_columns:
    if col != 'label':  # Ensure the label column is not scaled
        train_data_copy[col] = scaler.fit_transform(train_data_copy[[col]])
        test_data_copy[col] = scaler.transform(test_data_copy[[col]])

# Display the preprocessed data
print("\nPreprocessed Train Data:")
print(train_data_copy.head())

# Copy the original DataFrame to avoid modifying the original data
monster_data_copy = monster_data.copy()

# Handling missing values
# Check for missing values in each column
missing_values = monster_data_copy.isnull().sum()
print("Missing values before handling:\n", missing_values)

# Fill missing values with appropriate strategies
# For numeric columns, fill with the mean
numeric_columns = ['bone_length', 'rotting_flesh', 'hair_length']
for col in numeric_columns:
    monster_data_copy[col].fillna(monster_data_copy[col].mean(), inplace=True)

# For categorical columns, fill with the mode
categorical_columns = ['color', 'type']
for col in categorical_columns:
    monster_data_copy[col].fillna(monster_data_copy[col].mode()[0], inplace=True)

# Re-check for missing values
missing_values_after = monster_data_copy.isnull().sum()
print("Missing values after handling:\n", missing_values_after)

# Encoding categorical variables
# Use LabelEncoder for categorical columns
label_encoders = {}
for col in categorical_columns:
    le = LabelEncoder()
    monster_data_copy[col] = le.fit_transform(monster_data_copy[col])
    label_encoders[col] = le

# Standardize numeric columns (excluding 'id' as it's not a feature)

monster_data_copy[numeric_columns[1:]] = scaler.fit_transform(monster_data_copy[numeric_columns[1:]])
