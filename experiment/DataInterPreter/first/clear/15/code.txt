import pandas as pd

# Load the datasets
train_df = pd.read_csv(r'E:\csv\crab_age_train.csv')
test_df = pd.read_csv(r'E:\csv\crab_age_test.csv')

# Copy the DataFrames to avoid modifying the original datasets
train_df_copy = train_df.copy()
test_df_copy = test_df.copy()

# Check data types and apply appropriate methods
print("Original Data Types - Train:")
print(train_df_copy.dtypes)
print("\nOriginal Data Types - Test:")
print(test_df_copy.dtypes)

# Identify categorical and numerical columns
categorical_columns = train_df_copy.select_dtypes(include=['object']).columns
numerical_columns = train_df_copy.select_dtypes(include=['int64', 'float64']).columns

# Apply necessary preprocessing steps
for col in numerical_columns:
    if col != 'label':  # Ensure not to modify the label column
        # Example: Scaling numerical columns (if needed)
        mean_val = train_df_copy[col].mean()
        std_val = train_df_copy[col].std()
        train_df_copy[col] = (train_df_copy[col] - mean_val) / std_val
        test_df_copy[col] = (test_df_copy[col] - mean_val) / std_val

for col in categorical_columns:
    # Example: Label encoding for categorical columns (if needed)
    unique_values = train_df_copy[col].unique()
    value_map = {value: idx for idx, value in enumerate(unique_values)}
    train_df_copy[col] = train_df_copy[col].map(value_map)
    test_df_copy[col] = test_df_copy[col].map(value_map)

# Verify the changes
print("\nProcessed Data Types - Train:")
print(train_df_copy.dtypes)
print("\nProcessed Data Types - Test:")
print(test_df_copy.dtypes)

from sklearn.model_selection import train_test_split
from sklearn.metrics import mean_squared_error, r2_score
from xgboost import XGBRegressor

# Splitting the processed training data into features and target
X_train = train_df_copy.drop(columns=['id', 'Age'])
y_train = train_df_copy['Age']

# Splitting the processed testing data into features and target
X_test = test_df_copy.drop(columns=['id', 'Age'])
y_test = test_df_copy['Age']

# Defining the XGBoost Regressor with specified parameters
xgb_model = XGBRegressor(
    n_estimators=1000,
    max_depth=7,
    learning_rate=0.05,
    subsample=0.8,
    colsample_bytree=0.8,
    random_state=42
)

# Training the model
xgb_model.fit(X_train, y_train)

# Making predictions
y_pred_train = xgb_model.predict(X_train)
y_pred_test = xgb_model.predict(X_test)

# Evaluating the model
mse_train = mean_squared_error(y_train, y_pred_train)
r2_train = r2_score(y_train, y_pred_train)
mse_test = mean_squared_error(y_test, y_pred_test)
r2_test = r2_score(y_test, y_pred_test)

print(f"Training MSE: {mse_train:.4f}, R2 Score: {r2_train:.4f}")
print(f"Testing MSE: {mse_test:.4f}, R2 Score: {r2_test:.4f}")
