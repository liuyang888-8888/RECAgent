import pandas as pd

# Load the training data

# Perform data preprocessing steps
# Example: Convert date column to datetime type (if applicable)
if 'Date' in train_df_copy.columns:
    train_df_copy['Date'] = pd.to_datetime(train_df_copy['Date'])
    test_df_copy['Date'] = pd.to_datetime(test_df_copy['Date'])

# Example: Extract year, month, and day from the date column
if 'Date' in train_df_copy.columns:
    train_df_copy['Year'] = train_df_copy['Date'].dt.year
    train_df_copy['Month'] = train_df_copy['Date'].dt.month
    train_df_copy['Day'] = train_df_copy['Date'].dt.day
    
    test_df_copy['Year'] = test_df_copy['Date'].dt.year
    test_df_copy['Month'] = test_df_copy['Date'].dt.month
    test_df_copy['Day'] = test_df_copy['Date'].dt.day

# Example: Handle missing values (if any)
# For simplicity, we'll fill missing values with the median for numerical columns
numerical_columns = train_df_copy.select_dtypes(include=['number']).columns
for col in numerical_columns:
    if train_df_copy[col].isnull().any():

        train_df_copy[col].fillna(median_value, inplace=True)
        test_df_copy[col].fillna(median_value, inplace=True)

print("\nPreprocessed Training Data Preview:")
print(train_df_copy.head())

print("\nPreprocessed Test Data Preview:")

from statsmodels.tsa.arima.model import ARIMA
import numpy as np

# Selecting the target variable (Close price) for the ARIMA model
target_column = 'Close'
train_target = train_df_copy[target_column]

# Fitting the ARIMA model
# Here we use (p, d, q) = (5, 1, 0) as a starting point for the ARIMA parameters
# p: The number of lag observations included in the model.
# d: The number of times that the raw observations are differenced.
# q: The size of the moving average window.
model = ARIMA(train_target, order=(5, 1, 0))
model_fit = model.fit()

# Summary of the model
print(model_fit.summary())
