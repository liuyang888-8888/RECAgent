import pandas as pd

# Load the datasets
train_df = pd.read_csv('E:\\csv\\employee_attrition_train.csv')

# Handling missing values
train_df_copy.fillna(train_df_copy.median(), inplace=True)
test_df_copy.fillna(test_df_copy.median(), inplace=True)

# Ensure that the label column is not altered
label_column = 'Attrition'
if label_column in train_df_copy.columns:
    train_labels = train_df_copy[label_column].copy()
    train_df_copy.drop(columns=[label_column], inplace=True)
if label_column in test_df_copy.columns:
    test_labels = test_df_copy[label_column].copy()
    test_df_copy.drop(columns=[label_column], inplace=True)

# Encode categorical variables
categorical_columns = ['Department', 'EducationField', 'Gender', 'JobRole', 'MaritalStatus', 'Over18', 'OverTime']
label_encoder = LabelEncoder()
for col in categorical_columns:
    train_df_copy[col] = label_encoder.fit_transform(train_df_copy[col])
    test_df_copy[col] = label_encoder.transform(test_df_copy[col])

# Scale numerical features
numerical_columns = [col for col in train_df_copy.columns if col not in categorical_columns]
scaler = MinMaxScaler()
train_df_copy[numerical_columns] = scaler.fit_transform(train_df_copy[numerical_columns])
test_df_copy[numerical_columns] = scaler.transform(test_df_copy[numerical_columns])

# Reattach the label column
train_df_copy[label_column] = train_labels
test_df_copy[label_column] = test_labels

# Display the first few rows of the preprocessed datasets
print("\nPreprocessed Train dataset head:")
print(train_df_copy.head())
print("\nPreprocessed Test dataset head:")
print(test_df_copy.head())
