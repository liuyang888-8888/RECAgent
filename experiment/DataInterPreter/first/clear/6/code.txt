test_df_processed = test_df_copy.copy()

# Check for missing values in both training and testing sets
print("Missing values in Training Set:")
print(train_df_processed.isnull().sum())
print("\nMissing values in Testing Set:")
print(test_df_processed.isnull().sum())

# Handle missing values if any (for this example, we will fill with the median for numeric columns)
numeric_columns = ['gravity', 'ph', 'osmo', 'cond', 'urea', 'calc']

for col in numeric_columns:
    if train_df_processed[col].isnull().any():
        median_value = train_df_processed[col].median()
        train_df_processed[col].fillna(median_value, inplace=True)
        test_df_processed[col].fillna(median_value, inplace=True)

print("\nAfter handling missing values:")
print("Missing values in Training Set:")
print(train_df_processed.isnull().sum())
print("\nMissing values in Testing Set:")
print(test_df_processed.isnull().sum())

from sklearn.tree import DecisionTreeClassifier
from sklearn.metrics import accuracy_score, classification_report

# Initialize the Decision Tree Classifier
dt_classifier = DecisionTreeClassifier(random_state=42, max_depth=5, min_samples_split=10)

# Train the model
dt_classifier.fit(X_train, y_train)

# Predict on the training set
y_train_pred = dt_classifier.predict(X_train)
print("Training Accuracy:", accuracy_score(y_train, y_train_pred))
print("Training Classification Report:\n", classification_report(y_train, y_train_pred))

# Predict on the testing set
y_test_pred = dt_classifier.predict(X_test)
print("Testing Accuracy:", accuracy_score(y_test, y_test_pred))
print("Testing Classification Report:\n", classification_report(y_test, y_test_pred))
