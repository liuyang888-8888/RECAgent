# Perform one-hot encoding on specified columns
columns_to_encode = ['peripheral_pulse', 'mucous_membrane', 'capillary_refill_time']

# One-hot encode the specified columns for the training dataset
train_df_encoded = pd.get_dummies(train_df_copy, columns=columns_to_encode, drop_first=True)

# One-hot encode the specified columns for the test dataset
test_df_encoded = pd.get_dummies(test_df_copy, columns=columns_to_encode, drop_first=True)

# Ensure both datasets have the same columns after one-hot encoding
missing_cols = set(train_df_encoded.columns) - set(test_df_encoded.columns)
for c in missing_cols:
    test_df_encoded[c] = 0

# Reorder the columns in the test dataset to match the training dataset
test_df_encoded = test_df_encoded[train_df_encoded.columns]

# Display the first few rows of the encoded datasets
print("\nOne-Hot Encoded Train DataFrame:")
print(train_df_encoded.head())
print("\nOne-Hot Encoded Test DataFrame:")
print(test_df_encoded.head())

from lightgbm import LGBMClassifier
from sklearn.metrics import accuracy_score, classification_report

# Initialize the LightGBM classifier with specified parameters
lgbm_clf = LGBMClassifier(n_estimators=10, learning_rate=0.1)

# Train the model using the training data
lgbm_clf.fit(X_train, y_train)

# Predict the outcomes for the test data
y_pred = lgbm_clf.predict(X_test)

# Evaluate the model's performance
accuracy = accuracy_score(y_test, y_pred)
report = classification_report(y_test, y_pred)

print(f"Accuracy: {accuracy}")
print("Classification Report:")
print(report)
