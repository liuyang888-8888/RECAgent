import pandas as pd
from sklearn.preprocessing import LabelEncoder, StandardScaler

# Load the datasets
train_df = pd.read_csv(r'E:\csv\train.csv')
test_df = pd.read_csv(r'E:\csv\test.csv')

# Copy the DataFrames to ensure we don't modify the original data
train_df_copy = train_df.copy()
test_df_copy = test_df.copy()

# Monitor data types per column
print("Train dataset column types:")
print(train_df_copy.dtypes)
print("\nTest dataset column types:")
print(test_df_copy.dtypes)

# Identify categorical and numerical columns
categorical_cols = train_df_copy.select_dtypes(include=['object']).columns
numerical_cols = train_df_copy.select_dtypes(include=['int64', 'float64']).columns

# Encode categorical columns using LabelEncoder
label_encoder = LabelEncoder()
for col in categorical_cols:
    if col != 'label':  # Ensure not to encode the label column
        train_df_copy[col] = label_encoder.fit_transform(train_df_copy[col])
        test_df_copy[col] = label_encoder.transform(test_df_copy[col])

# Scale numerical columns using StandardScaler
scaler = StandardScaler()
train_df_copy[numerical_cols] = scaler.fit_transform(train_df_copy[numerical_cols])
test_df_copy[numerical_cols] = scaler.transform(test_df_copy[numerical_cols])

# Display the first few rows of the preprocessed datasets
print("\nPreprocessed Train Dataset:")
print(train_df_copy.head())
print("\nPreprocessed Test Dataset:")
print(test_df_copy.head())
