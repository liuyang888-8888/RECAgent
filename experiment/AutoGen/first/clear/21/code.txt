import pandas as pd
import numpy as np
from sklearn.ensemble import GradientBoostingRegressor
from sklearn.metrics import mean_squared_error

# 1. 导入数据
train_data_path = r'E:\Postgraduate\Code\SunnylnAgent\data\csv\forest_fires_train.csv'
test_data_path = r'E:\Postgraduate\Code\SunnylnAgent\data\csv\forest_fires_test.csv'

train_df = pd.read_csv(train_data_path)
test_df = pd.read_csv(test_data_path)

# 2. 数据预处理
# 检查缺失值
print("训练集缺失值情况：", train_df.isnull().sum())
print("测试集缺失值情况：", test_df.isnull().sum())

# 对分类变量进行独热编码
train_df = pd.get_dummies(train_df, columns=['month', 'day'])
test_df = pd.get_dummies(test_df, columns=['month', 'day'])

# 确保训练集和测试集的特征列一致
train_columns = set(train_df.columns)
test_columns = set(test_df.columns)

# 找出训练集和测试集中缺失的特征列
missing_in_test = train_columns - test_columns
missing_in_train = test_columns - train_columns

# 在测试集中添加缺失的特征列，并将值设置为0
for col in missing_in_test:
    if col != 'area':
        test_df[col] = 0

# 在训练集中添加缺失的特征列，并将值设置为0
for col in missing_in_train:
    if col != 'area':
        train_df[col] = 0

# 重新排序列，确保训练集和测试集的特征列顺序一致
train_df = train_df[train_df.columns]
test_df = test_df[train_df.columns]

# 3. 准备特征和目标变量
X_train = train_df.drop(columns=['area'])
y_train = train_df['area']

X_test = test_df.drop(columns=['area'])
y_test = test_df['area']

# 4. 构建模型
model = GradientBoostingRegressor(n_estimators=10, learning_rate=0.3, random_state=42)

# 5. 训练模型
model.fit(X_train, y_train)

# 6. 评估模型
y_pred = model.predict(X_test)
mse = mean_squared_error(y_test, y_pred)
print(f"测试集上的均方误差（MSE）: {mse}")
