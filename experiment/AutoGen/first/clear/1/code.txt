import pandas as pd
import numpy as np
from sklearn.ensemble import RandomForestClassifier
from sklearn.metrics import f1_score
from sklearn.preprocessing import OneHotEncoder

# 1. 加载数据
train_path = r'E:\Postgraduate\Code\SunnylnAgent\data\csv\bank_customer_churn_train.csv'
test_path = r'E:\Postgraduate\Code\SunnylnAgent\data\csv\bank_customer_churn_test.csv'

test_data = pd.read_csv(test_path)

# 2. 数据预处理
# 独热编码 Geography 和 Gender
encoder = OneHotEncoder(sparse=False, drop='first')
categorical_features = ['Geography', 'Gender']

# 拼接独热编码后的特征
train_encoded = encoder.fit_transform(train_data[categorical_features])
test_encoded = encoder.transform(test_data[categorical_features])

train_data_encoded = pd.concat([train_data.drop(categorical_features, axis=1), pd.DataFrame(train_encoded)], axis=1)
test_data_encoded = pd.concat([test_data.drop(categorical_features, axis=1), pd.DataFrame(test_encoded)], axis=1)

# 分离特征和标签
X_train = train_data_encoded.drop('Exited', axis=1)
y_train = train_data_encoded['Exited']

X_test = test_data_encoded.drop('Exited', axis=1)
y_test = test_data_encoded['Exited']

# 3. 训练随机森林模型
rf_model = RandomForestClassifier(n_estimators=10, random_state=42)
rf_model.fit(X_train, y_train)

# 4. 评估模型
y_pred = rf_model.predict(X_test)
f1 = f1_score(y_test, y_pred, average='weighted')

print(f'加权平均 F1 分数: {f1}')

import pandas as pd
import numpy as np
from sklearn.ensemble import RandomForestClassifier
from sklearn.metrics import f1_score

# 1. 加载数据
train_path = r'E:\Postgraduate\Code\SunnylnAgent\data\csv\bank_customer_churn_train.csv'
test_path = r'E:\Postgraduate\Code\SunnylnAgent\data\csv\bank_customer_churn_test.csv'

train_data = pd.read_csv(train_path)
test_data = pd.read_csv(test_path)

# 2. 数据预处理
# 独热编码 Geography 和 Gender
train_data = pd.get_dummies(train_data, columns=['Geography', 'Gender'], drop_first=True)
test_data = pd.get_dummies(test_data, columns=['Geography', 'Gender'], drop_first=True)

# 检查并转换非数值列
for col in train_data.columns:
    if train_data[col].dtype == 'object':
        train_data[col] = pd.to_numeric(train_data[col], errors='coerce')
        test_data[col] = pd.to_numeric(test_data[col], errors='coerce')

# 填充缺失值
train_data.fillna(0, inplace=True)
test_data.fillna(0, inplace=True)

# 分离特征和标签
X_train = train_data.drop('Exited', axis=1)
y_train = train_data['Exited']

X_test = test_data.drop('Exited', axis=1)
y_test = test_data['Exited']

# 3. 训练随机森林模型
rf_model = RandomForestClassifier(n_estimators=10, random_state=42)
rf_model.fit(X_train, y_train)

# 4. 评估模型
y_pred = rf_model.predict(X_test)
f1 = f1_score(y_test, y_pred, average='weighted')

print(f'加权平均 F1 分数: {f1}')
