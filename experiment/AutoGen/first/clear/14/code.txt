import pandas as pd
import lightgbm as lgb
from sklearn.metrics import mean_squared_error
import os

# 设置数据路径
data_path = r'E:\Postgraduate\Code\SunnylnAgent\data\csv'

# 加载训练集和测试集
train_data = pd.read_csv(os.path.join(data_path, 'abalone_train.csv'))
test_data = pd.read_csv(os.path.join(data_path, 'abalone_test.csv'))

# 分离特征和标签
X_train = train_data.drop(columns=['Rings'])
y_train = train_data['Rings']
X_test = test_data.drop(columns=['Rings'])
y_test = test_data['Rings']

# 转换为 LightGBM 数据集
lgb_train = lgb.Dataset(X_train, y_train)
lgb_test = lgb.Dataset(X_test, y_test, reference=lgb_train)

# 设置 LightGBM 参数
params = {
    'boosting_type': 'rf',
    'num_trees': 20,
    'objective': 'regression',
    'metric': 'mse',
    'feature_fraction': 0.6,
    'bagging_fraction': 0.8,
    'bagging_freq': 1,
    'verbose': 0
}

# 训练模型
gbm = lgb.train(params, lgb_train, num_boost_round=20, valid_sets=lgb_test, early_stopping_rounds=5)

# 预测
y_pred = gbm.predict(X_test, num_iteration=gbm.best_iteration)

# 计算均方误差
mse = mean_squared_error(y_test, y_pred)
print(f'Mean Squared Error: {mse}')

# 保存预测结果
output_path = os.path.join(data_path, 'abalone_predictions.csv')
pd.DataFrame({'Rings': y_pred}).to_csv(output_path, index=False)
