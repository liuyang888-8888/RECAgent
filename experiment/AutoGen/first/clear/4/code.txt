import pandas as pd
from sklearn.tree import DecisionTreeClassifier
from sklearn.metrics import f1_score
from sklearn.preprocessing import LabelEncoder

# 1. 导入必要的库

# 2. 加载数据
train_path = r'E:\Postgraduate\Code\SunnylnAgent\data\csv\employee_attrition_train.csv'
test_path = r'E:\Postgraduate\Code\SunnylnAgent\data\csv\employee_attrition_test.csv'

train_df = pd.read_csv(train_path)
test_df = pd.read_csv(test_path)

# 3. 数据预处理
# 检查并处理缺失值
print("训练集缺失值情况：\n", train_df.isnull().sum())
print("测试集缺失值情况：\n", test_df.isnull().sum())

# 填充缺失值
numeric_columns = train_df.select_dtypes(include=['number']).columns
categorical_columns = train_df.select_dtypes(include=['object']).columns

train_df[numeric_columns] = train_df[numeric_columns].fillna(train_df[numeric_columns].median())
test_df[numeric_columns] = test_df[numeric_columns].fillna(test_df[numeric_columns].median())

for column in categorical_columns:
    train_df[column] = train_df[column].fillna(train_df[column].mode()[0])
    test_df[column] = test_df[column].fillna(test_df[column].mode()[0])

# 将分类变量进行编码
label_encoders = {}
for column in categorical_columns:
    le = LabelEncoder()
    train_df[column] = le.fit_transform(train_df[column])
    test_df[column] = le.transform(test_df[column])
    label_encoders[column] = le

# 分离特征和目标变量
X_train = train_df.drop(columns=['Attrition'])
y_train = train_df['Attrition']
X_test = test_df.drop(columns=['Attrition'])
y_test = test_df['Attrition']

# 4. 构建决策树模型
dt_model = DecisionTreeClassifier(min_samples_leaf=3, min_samples_split=5, max_depth=50)
dt_model.fit(X_train, y_train)

# 5. 预测和评估
y_pred = dt_model.predict(X_test)
f1 = f1_score(y_test, y_pred, average='weighted')
print(f"测试集上的加权平均 F1 分数: {f1}")

# 6. 保存预测结果
test_df['Attrition_Prediction'] = y_pred
test_df.to_csv(r'E:\Postgraduate\Code\SunnylnAgent\data\csv\employee_attrition_test_predictions.csv', index=False)
