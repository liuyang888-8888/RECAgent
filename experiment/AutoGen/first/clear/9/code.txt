import pandas as pd
from sklearn.ensemble import RandomForestClassifier
from sklearn.metrics import f1_score
from sklearn.impute import SimpleImputer

# 加载数据
train_data_path = r'E:\Postgraduate\Code\SunnylnAgent\data\csv\heart_diseases_train.csv'
test_data_path = r'E:\Postgraduate\Code\SunnylnAgent\data\csv\heart_diseases_test.csv'

train_df = pd.read_csv(train_data_path)
test_df = pd.read_csv(test_data_path)

# 查看数据
print("Train data shape:", train_df.shape)
print("Test data shape:", test_df.shape)

# 处理缺失值
imputer = SimpleImputer(strategy='mean')

# 填充训练集中的缺失值
train_df_imputed = pd.DataFrame(imputer.fit_transform(train_df), columns=train_df.columns)

# 填充测试集中的缺失值
test_df_imputed = pd.DataFrame(imputer.transform(test_df), columns=test_df.columns)

# 分离特征和目标变量
X_train = train_df_imputed.drop(columns=['target'])
y_train = train_df_imputed['target']

X_test = test_df_imputed.drop(columns=['target'])
y_test = test_df_imputed['target']

# 构建随机森林分类模型
rf_classifier = RandomForestClassifier(n_estimators=10, random_state=42)

# 训练模型
rf_classifier.fit(X_train, y_train)

# 预测测试集
y_pred = rf_classifier.predict(X_test)

# 计算F1分数
f1 = f1_score(y_test, y_pred, average='weighted')

print("F1 Score on test set:", f1)
