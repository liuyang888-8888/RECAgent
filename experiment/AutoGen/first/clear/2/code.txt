import pandas as pd
from sklearn.svm import SVC
from sklearn.metrics import f1_score
from sklearn.preprocessing import LabelEncoder

# 1. 导入必要的库

# 2. 加载数据
train_data_path = r'E:\Postgraduate\Code\SunnylnAgent\data\csv\cirrhosis_patient_train.csv'
test_data_path = r'E:\Postgraduate\Code\SunnylnAgent\data\csv\cirrhosis_patient_test.csv'

train_df = pd.read_csv(train_data_path)
test_df = pd.read_csv(test_data_path)

# 3. 检查数据列名
print("训练集列名:", train_df.columns)
print("测试集列名:", test_df.columns)

# 4. 数据预处理
# 检查并处理缺失值
if train_df.isnull().sum().sum() > 0:
    train_df = train_df.dropna()
if test_df.isnull().sum().sum() > 0:
    test_df = test_df.dropna()

# 将特征和标签分开
# 假设标签列名为 'Status'
label_column = 'Status'
if label_column not in train_df.columns:
    raise ValueError(f"列名 '{label_column}' 未在训练集中找到")
if label_column not in test_df.columns:
    raise ValueError(f"列名 '{label_column}' 未在测试集中找到")

# 对分类变量进行编码
categorical_columns = ['Drug', 'Sex', 'Ascites', 'Hepatomegaly', 'Spiders', 'Edema']
label_encoders = {}

for column in categorical_columns:
    le = LabelEncoder()
    train_df[column] = le.fit_transform(train_df[column])
    test_df[column] = le.transform(test_df[column])
    label_encoders[column] = le

X_train = train_df.drop(columns=[label_column])
y_train = train_df[label_column]

X_test = test_df.drop(columns=[label_column])
y_test = test_df[label_column]

# 5. 构建 SVM 模型
svm_model = SVC(kernel='linear', max_iter=200)

# 6. 训练模型
svm_model.fit(X_train, y_train)

# 7. 预测
y_pred = svm_model.predict(X_test)

# 8. 评估模型
f1 = f1_score(y_test, y_pred, average='weighted')
print(f'加权平均 F1 分数: {f1}')
