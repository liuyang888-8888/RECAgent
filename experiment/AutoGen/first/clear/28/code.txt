import pandas as pd
import numpy as np
from statsmodels.tsa.arima.model import ARIMA
from sklearn.metrics import mean_squared_error

# 1. 导入数据
train_data_path = r'E:\Postgraduate\Code\SunnylnAgent\data\csv\electricity_sales_train.csv'
test_data_path = r'E:\Postgraduate\Code\SunnylnAgent\data\csv\electricity_sales_test.csv'

train_df = pd.read_csv(train_data_path)
test_df = pd.read_csv(test_data_path)

# 2. 数据预处理
# 假设数据集中的时间列名为 'date'，销量列名为 'sales'
train_df['date'] = pd.to_datetime(train_df['date'])
test_df['date'] = pd.to_datetime(test_df['date'])

train_df.set_index('date', inplace=True)
test_df.set_index('date', inplace=True)

# 3. 构建和训练 ARIMA 模型
# 设置自回归项数（p）为1
model = ARIMA(train_df['sales'], order=(1, 1, 0))
model_fit = model.fit()

# 4. 评估模型
# 在训练集上计算均方误差（MSE）
train_predictions = model_fit.fittedvalues
train_mse = mean_squared_error(train_df['sales'], train_predictions)
print(f'Training MSE: {train_mse}')

# 在测试集上进行预测
test_predictions = model_fit.forecast(steps=len(test_df))
test_mse = mean_squared_error(test_df['sales'], test_predictions)
print(f'Testing MSE: {test_mse}')

# 输出训练集上的均方误差
print(f'Training MSE: {train_mse}')
