import pandas as pd
from sklearn.tree import DecisionTreeRegressor
from sklearn.metrics import mean_squared_error
from sklearn.model_selection import train_test_split
from sklearn.preprocessing import LabelEncoder

# 1. 加载数据
train_data_path = r'E:\Postgraduate\Code\SunnylnAgent\data\csv\crab_age_train.csv'
test_data_path = r'E:\Postgraduate\Code\SunnylnAgent\data\csv\crab_age_test.csv'

train_df = pd.read_csv(train_data_path)
test_df = pd.read_csv(test_data_path)

# 2. 数据预处理
# 检查数据是否有缺失值
print("训练数据缺失值情况：", train_df.isnull().sum())
print("测试数据缺失值情况：", test_df.isnull().sum())

    raise ValueError("训练数据中没有 'age' 或 'Age' 列")

    raise ValueError("测试数据中没有 'age' 或 'Age' 列")

test_df = test_df.rename(columns={'Age': 'age'})

# 3. 数据抽样

y_train = train_sample_df['age']
