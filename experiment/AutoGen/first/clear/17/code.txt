import pandas as pd
from sklearn.neighbors import KNeighborsRegressor
from sklearn.metrics import mean_squared_error

# 读取训练集和测试集
train_data_path = r'E:\Postgraduate\Code\SunnylnAgent\data\csv\mohs_hardness_train.csv'

train_df = pd.read_csv(train_data_path)
test_df = pd.read_csv(test_data_path)

# 分离特征和目标变量
X_train = train_df.drop(columns=['hardness'])
y_train = train_df['hardness']

X_test = test_df.drop(columns=['hardness'])
y_test = test_df['hardness']

# 构建 KNN 模型
k = 5
knn = KNeighborsRegressor(n_neighbors=k)

# 训练模型
knn.fit(X_train, y_train)

# 在测试集上进行预测
y_pred = knn.predict(X_test)

# 计算均方误差
mse = mean_squared_error(y_test, y_pred)

# 输出均方误差
print(f"Mean Squared Error (MSE): {mse}")

# 保存预测结果
test_df['predicted_hardness'] = y_pred
test_df.to_csv(r'E:\Postgraduate\Code\SunnylnAgent\data\csv\mohs_hardness_test_predictions.csv', index=False)
