import pandas as pd
from sklearn.linear_model import LogisticRegression
from sklearn.preprocessing import OneHotEncoder
from sklearn.metrics import f1_score
import joblib

# 1. 加载数据
train_data_path = r'E:\Postgraduate\Code\SunnylnAgent\data\csv\crime_category_train.csv'
test_data_path = r'E:\Postgraduate\Code\SunnylnAgent\data\csv\crime_category_test.csv'

train_df = pd.read_csv(train_data_path)
test_df = pd.read_csv(test_data_path)

# 2. 数据预处理
# 对 pdDistrict 字段进行独热编码
encoder = OneHotEncoder(sparse=False, handle_unknown='ignore')
pdDistrict_encoded_train = encoder.fit_transform(train_df[['pdDistrict']])
pdDistrict_encoded_test = encoder.transform(test_df[['pdDistrict']])

# 将独热编码后的数据与原始数据合并
train_df_encoded = pd.concat([train_df.drop('pdDistrict', axis=1), pd.DataFrame(pdDistrict_encoded_train, columns=encoder.get_feature_names_out(['pdDistrict']))], axis=1)
test_df_encoded = pd.concat([test_df.drop('pdDistrict', axis=1), pd.DataFrame(pdDistrict_encoded_test, columns=encoder.get_feature_names_out(['pdDistrict']))], axis=1)

# 分离特征和标签
X_train = train_df_encoded.drop('category', axis=1)
y_train = train_df_encoded['category']
X_test = test_df_encoded.drop('category', axis=1)
y_test = test_df_encoded['category']

# 3. 构建模型
model = LogisticRegression(penalty='l2', C=1.5, multi_class='multinomial', solver='lbfgs', max_iter=1000)

# 4. 训练模型
model.fit(X_train, y_train)

# 5. 评估模型
y_pred = model.predict(X_test)
f1 = f1_score(y_test, y_pred, average='weighted')
print(f'F1 Score (weighted): {f1}')

# 6. 保存模型
model_path = r'E:\Postgraduate\Code\SunnylnAgent\data\csv\logistic_regression_model.pkl'
joblib.dump(model, model_path)
