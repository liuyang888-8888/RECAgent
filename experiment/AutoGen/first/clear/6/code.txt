import pandas as pd
from sklearn.tree import DecisionTreeClassifier
from sklearn.metrics import f1_score

# 读取数据
train_data_path = r'E:\Postgraduate\Code\SunnylnAgent\data\csv\kidney_stone_train.csv'
test_data_path = r'E:\Postgraduate\Code\SunnylnAgent\data\csv\kidney_stone_test.csv'

train_df = pd.read_csv(train_data_path)
test_df = pd.read_csv(test_data_path)

# 检查数据
print("训练集前几行：")
print(train_df.head())
print("测试集前几行：")
print(test_df.head())

# 分离特征和标签
X_train = train_df.drop(columns=['target'])
y_train = train_df['target']

X_test = test_df.drop(columns=['target'])
y_test = test_df['target']

# 检查是否有缺失值
print("训练集缺失值情况：")
