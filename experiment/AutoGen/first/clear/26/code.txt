df = df.sort_values('date')

# 提取 CO2 浓度列
co2_series = df['co2']

# 定义 ARIMA 模型参数
p, d, q = 2, 1, 1

# 训练 ARIMA 模型
model = ARIMA(co2_series, order=(p, d, q))
model_fit = model.fit()

# 获取拟合值
fitted_values = model_fit.fittedvalues

# 计算训练集上的均方误差（MSE）

print(f'Training Set MSE: {mse}')
