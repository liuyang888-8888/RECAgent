import pandas as pd
import numpy as np
from statsmodels.tsa.arima.model import ARIMA
from sklearn.metrics import mean_squared_error

# 1. 加载数据
train_data_path = r'E:\Postgraduate\Code\SunnylnAgent\data\csv\nvidia_share_price_train.csv'
test_data_path = r'E:\Postgraduate\Code\SunnylnAgent\data\csv\nvidia_share_price_test.csv'

train_df = pd.read_csv(train_data_path)
test_df = pd.read_csv(test_data_path)

# 2. 数据预处理
# 假设日期列名为 'Date'，股价列名为 'High'
train_df['Date'] = pd.to_datetime(train_df['Date'])
test_df['Date'] = pd.to_datetime(test_df['Date'])

train_df.set_index('Date', inplace=True)
test_df.set_index('Date', inplace=True)

# 3. 构建 ARIMA 模型
# 选择合适的 ARIMA 参数 (p, d, q)
# 这里我们选择 (5, 1, 0) 作为示例参数
model = ARIMA(train_df['High'], order=(5, 1, 0))

# 4. 模型拟合
model_fit = model.fit()

# 5. 预测
# 在训练集上进行预测
train_predictions = model_fit.fittedvalues

# 在测试集上进行预测
test_predictions = model_fit.forecast(steps=len(test_df))

# 6. 评估
# 计算训练集上的均方误差（MSE）
train_mse = mean_squared_error(train_df['High'], train_predictions)
print(f'Training MSE: {train_mse}')

# 计算测试集上的均方误差（MSE）
