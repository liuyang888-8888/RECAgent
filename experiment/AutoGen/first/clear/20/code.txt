import pandas as pd
from sklearn.ensemble import GradientBoostingRegressor
from sklearn.metrics import mean_squared_error
import joblib

# 读取数据
train_data_path = r'E:\Postgraduate\Code\SunnylnAgent\data\csv\bike_sharing_train.csv'
test_data_path = r'E:\Postgraduate\Code\SunnylnAgent\data\csv\bike_sharing_test.csv'

train_df = pd.read_csv(train_data_path)
test_df = pd.read_csv(test_data_path)

# 查看数据列
print("训练集列名:", train_df.columns)
print("测试集列名:", test_df.columns)

# 选择特征和目标变量
features = ['season', 'yr', 'mnth', 'hr', 'holiday', 'weekday', 'workingday', 'weathersit', 'temp', 'atemp', 'hum', 'windspeed']
target = 'cnt'

# 准备训练数据
X_train = train_df[features]
y_train = train_df[target]

# 准备测试数据
X_test = test_df[features]
y_test = test_df[target]

# 构建模型
gbdt = GradientBoostingRegressor(n_estimators=8, learning_rate=0.01, random_state=42)

# 训练模型
gbdt.fit(X_train, y_train)

# 在测试集上进行预测
y_pred = gbdt.predict(X_test)

# 计算均方误差
mse = mean_squared_error(y_test, y_pred)
print(f"测试集上的均方误差 (MSE): {mse}")

# 保存预测结果
test_df['predicted_cnt'] = y_pred
output_path = r'E:\Postgraduate\Code\SunnylnAgent\data\csv\bike_sharing_test_predictions.csv'
test_df.to_csv(output_path, index=False)

# 保存模型
model_path = r'E:\Postgraduate\Code\SunnylnAgent\data\csv\gbdt_bike_sharing_model.pkl'
joblib.dump(gbdt, model_path)

print("模型训练完成，预测结果已保存。")
