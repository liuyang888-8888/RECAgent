import pandas as pd
import numpy as np
from statsmodels.tsa.arima.model import ARIMA
from sklearn.metrics import mean_squared_error

# 1. 导入数据
train_path = r'E:\Postgraduate\Code\SunnylnAgent\data\csv\sp500_index_train.csv'
test_path = r'E:\Postgraduate\Code\SunnylnAgent\data\csv\sp500_index_test.csv'

train_data = pd.read_csv(train_path, parse_dates=['Date'], index_col='Date')
test_data = pd.read_csv(test_path, parse_dates=['Date'], index_col='Date')

# 2. 数据预处理
# 确保数据按时间顺序排列
train_data = train_data.sort_index()
test_data = test_data.sort_index()

# 检查并处理缺失值
train_data = train_data.dropna()
test_data = test_data.dropna()

# 3. 构建 ARIMA 模型
# 选择合适的 ARIMA 参数 (p, d, q)
# 这里我们使用 (1, 1, 1) 作为示例参数
model = ARIMA(train_data['Close'], order=(1, 1, 1))
model_fit = model.fit()

# 4. 评估模型
# 在训练集上计算均方误差（MSE）
train_predictions = model_fit.fittedvalues
train_mse = mean_squared_error(train_data['Close'], train_predictions)
print(f'Training MSE: {train_mse}')

# 在测试集上进行预测
test_predictions = model_fit.forecast(steps=len(test_data))
test_mse = mean_squared_error(test_data['Close'], test_predictions)
print(f'Testing MSE: {test_mse}')
