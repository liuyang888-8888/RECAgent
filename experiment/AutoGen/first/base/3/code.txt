import pandas as pd
from sklearn.ensemble import RandomForestClassifier
from sklearn.metrics import f1_score
from sklearn.preprocessing import LabelEncoder

# 1. 导入必要的库

# 2. 加载数据
train_path = r'E:\Postgraduate\Code\SunnylnAgent\data\csv\crime_category_train.csv'
test_path = r'E:\Postgraduate\Code\SunnylnAgent\data\csv\crime_category_test.csv'

train_data = pd.read_csv(train_path)
test_data = pd.read_csv(test_path)

# 检查列名
print("Train data columns:", train_data.columns)
print("Test data columns:", test_data.columns)

# 3. 数据预处理
# 检查并处理缺失值
if train_data.isnull().sum().sum() > 0:
    train_data = train_data.dropna()
if test_data.isnull().sum().sum() > 0:
    test_data = test_data.dropna()

# 确保 'category' 列存在
if 'category' not in train_data.columns or 'category' not in test_data.columns:
    raise ValueError("The 'category' column is missing in the dataset.")

# 将类别标签编码为数值
label_encoder = LabelEncoder()
train_data['category'] = label_encoder.fit_transform(train_data['category'])
test_data['category'] = label_encoder.transform(test_data['category'])

# 对分类特征进行独热编码
train_data = pd.get_dummies(train_data, columns=['dayOfWeek', 'pdDistrict'])
test_data = pd.get_dummies(test_data, columns=['dayOfWeek', 'pdDistrict'])

# 确保训练集和测试集的列一致
missing_cols = set(train_data.columns) - set(test_data.columns)
for c in missing_cols:
    test_data[c] = 0
test_data = test_data[train_data.columns]

# 分离特征和标签
X_train = train_data.drop(columns=['category'])
y_train = train_data['category']
X_test = test_data.drop(columns=['category'])
y_test = test_data['category']

# 4. 训练模型
model = RandomForestClassifier(random_state=42)
model.fit(X_train, y_train)

# 5. 评估模型
y_pred = model.predict(X_test)
f1 = f1_score(y_test, y_pred, average='weighted')
print(f'Weighted F1 Score: {f1}')
