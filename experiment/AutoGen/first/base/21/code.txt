import pandas as pd
from sklearn.linear_model import LinearRegression
from sklearn.metrics import mean_squared_error

# 1. 加载数据
train_data_path = r'E:\Postgraduate\Code\SunnylnAgent\data\csv\forest_fires_train.csv'
test_data_path = r'E:\Postgraduate\Code\SunnylnAgent\data\csv\forest_fires_test.csv'

train_df = pd.read_csv(train_data_path)
test_df = pd.read_csv(test_data_path)

# 2. 数据预处理
# 检查是否有缺失值
print("训练集缺失值情况：", train_df.isnull().sum())
print("测试集缺失值情况：", test_df.isnull().sum())

# 3. 特征选择和编码
# 选择用于预测的特征
features = ['X', 'Y', 'month', 'day', 'FFMC', 'DMC', 'DC', 'ISI', 'temp', 'RH', 'wind', 'rain']
target = 'area'

# 对分类特征进行独热编码
train_df = pd.get_dummies(train_df, columns=['month', 'day'])
test_df = pd.get_dummies(test_df, columns=['month', 'day'])

# 确保训练集和测试集的特征列一致
train_columns = train_df.columns.difference([target])
test_columns = test_df.columns.difference([target])

# 找出训练集和测试集的特征列差异
missing_in_test = set(train_columns) - set(test_columns)
missing_in_train = set(test_columns) - set(train_columns)

# 在测试集中添加缺失的特征列，并设置为0
for col in missing_in_test:
    test_df[col] = 0

# 在训练集中添加缺失的特征列，并设置为0
for col in missing_in_train:
    train_df[col] = 0

# 确保特征列顺序一致
train_df = train_df[train_columns.union([target])]
test_df = test_df[train_columns.union([target])]

X_train = train_df[train_columns]
y_train = train_df[target]

X_test = test_df[train_columns]
y_test = test_df[target]

# 4. 模型训练
model = LinearRegression()
model.fit(X_train, y_train)

# 5. 模型评估
y_pred = model.predict(X_test)
