import pandas as pd
import numpy as np
from statsmodels.tsa.arima.model import ARIMA
from sklearn.metrics import mean_squared_error

# 1. 导入数据
train_data_path = r'E:\Postgraduate\Code\SunnylnAgent\data\csv\nvidia_share_price_train.csv'
test_data_path = r'E:\Postgraduate\Code\SunnylnAgent\data\csv\nvidia_share_price_test.csv'

train_df = pd.read_csv(train_data_path)
test_df = pd.read_csv(test_data_path)

# 2. 数据预处理
# 确保日期列是 datetime 类型
train_df['Date'] = pd.to_datetime(train_df['Date'])
test_df['Date'] = pd.to_datetime(test_df['Date'])

# 设置日期列为索引
train_df.set_index('Date', inplace=True)
test_df.set_index('Date', inplace=True)

# 确保日期索引是单调递增的
train_df = train_df.sort_index()
test_df = test_df.sort_index()

# 设置日期索引的频率
train_df = train_df.asfreq('B')  # 'B' 表示工作日频率
test_df = test_df.asfreq('B')

# 处理缺失值
train_df = train_df.interpolate(method='linear')  # 线性插值处理缺失值
test_df = test_df.interpolate(method='linear')

# 选择每日最高价作为目标变量
train_series = train_df['High']
test_series = test_df['High']

# 3. 构建 ARIMA 模型
# 这里我们选择 (5,1,0) 作为 ARIMA 模型的参数，你可以根据实际情况调整
model = ARIMA(train_series, order=(5, 1, 0))
model_fit = model.fit()

# 4. 预测
train_predictions = model_fit.predict(start=train_series.index[0], end=train_series.index[-1])

# 确保预测结果中没有 NaN 值
train_predictions = train_predictions.fillna(method='ffill')  # 前向填充 NaN 值

# 5. 计算均方误差
mse = mean_squared_error(train_series, train_predictions)
print(f'训练集上的均方误差 (MSE): {mse}')
