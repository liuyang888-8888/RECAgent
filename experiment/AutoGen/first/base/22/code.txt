import pandas as pd
from sklearn.cluster import KMeans
from sklearn.preprocessing import StandardScaler
from sklearn.metrics import silhouette_score

# 读取数据
data_path = r'E:\Postgraduate\Code\SunnylnAgent\data\csv\monster_train.csv'
data = pd.read_csv(data_path)

# 查看数据前几行
print(data.head())

# 检查数据是否有缺失值
print(data.isnull().sum())

# 选择特征列
features = data.columns.drop('type')  # 假设 'type' 是目标列，不用于聚类

# 对非数值特征进行独热编码
data_encoded = pd.get_dummies(data[features], drop_first=True)

# 标准化特征
scaler = StandardScaler()
scaled_data = scaler.fit_transform(data_encoded)

# 使用 KMeans 进行聚类
kmeans = KMeans(n_clusters=3, random_state=42)  # 假设我们选择 3 个聚类
kmeans.fit(scaled_data)

# 获取聚类标签
labels = kmeans.labels_

# 计算轮廓系数
silhouette_avg = silhouette_score(scaled_data, labels)
print(f'轮廓系数: {silhouette_avg}')
