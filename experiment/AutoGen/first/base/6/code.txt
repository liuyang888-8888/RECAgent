import pandas as pd
from sklearn.linear_model import LogisticRegression
from sklearn.metrics import f1_score
from sklearn.impute import SimpleImputer

# 1. 导入必要的库

# 2. 加载数据
train_path = r'E:\Postgraduate\Code\SunnylnAgent\data\csv\kidney_stone_train.csv'
test_path = r'E:\Postgraduate\Code\SunnylnAgent\data\csv\kidney_stone_test.csv'

train_data = pd.read_csv(train_path)
test_data = pd.read_csv(test_path)

# 3. 数据预处理
# 检查缺失值

print("测试集缺失值情况：\n", test_data.isnull().sum())

# 将目标变量转换为二分类标签
train_data['target'] = train_data['target'].map({'Yes': 1, 'No': 0})

# 4. 特征选择
features = train_data.columns[:-1]  # 假设最后一列是目标变量
X_train = train_data[features]
y_train = train_data['target']

X_test = test_data[features]
y_test = test_data['target']

# 5. 模型训练

# 6. 模型评估

print(f"测试集上的 F1 分数: {f1}")
