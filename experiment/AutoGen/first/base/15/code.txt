import pandas as pd
from sklearn.linear_model import LinearRegression
from sklearn.metrics import mean_squared_error

# 加载数据
train_data_path = r'E:\Postgraduate\Code\SunnylnAgent\data\csv\crab_age_train.csv'
test_data_path = r'E:\Postgraduate\Code\SunnylnAgent\data\csv\crab_age_test.csv'

train_df = pd.read_csv(train_data_path)
test_df = pd.read_csv(test_data_path)

# 打印列名
print("Train data columns:", train_df.columns)
print("Test data columns:", test_df.columns)

# 对 'Sex' 列进行 one-hot 编码
train_df = pd.get_dummies(train_df, columns=['Sex'])
test_df = pd.get_dummies(test_df, columns=['Sex'])

# 分离特征和目标变量
X_train = train_df.drop(columns=['Age'])
y_train = train_df['Age']

X_test = test_df.drop(columns=['Age'])
y_test = test_df['Age']

# 确保训练集和测试集的特征列一致
X_test = X_test.reindex(columns=X_train.columns, fill_value=0)

# 训练线性回归模型
model = LinearRegression()
model.fit(X_train, y_train)

# 预测测试集
y_pred = model.predict(X_test)

# 计算均方误差
mse = mean_squared_error(y_test, y_pred)
print(f'Test MSE: {mse}')
