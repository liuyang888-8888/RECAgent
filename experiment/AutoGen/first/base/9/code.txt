import pandas as pd
from sklearn.ensemble import RandomForestClassifier
from sklearn.metrics import f1_score

# 1. 导入必要的库

# 2. 加载数据
train_data_path = r'E:\Postgraduate\Code\SunnylnAgent\data\csv\heart_diseases_train.csv'
test_data_path = r'E:\Postgraduate\Code\SunnylnAgent\data\csv\heart_diseases_test.csv'

train_df = pd.read_csv(train_data_path)
test_df = pd.read_csv(test_data_path)

# 3. 数据预处理
# 检查缺失值
print("训练集缺失值情况：", train_df.isnull().sum())
print("测试集缺失值情况：", test_df.isnull().sum())

# 使用中位数填充缺失值
train_df.fillna(train_df.median(), inplace=True)
test_df.fillna(test_df.median(), inplace=True)

# 将数据集分为特征和标签
X_train = train_df.drop(columns=['target'])
y_train = train_df['target']

X_test = test_df.drop(columns=['target'])
y_test = test_df['target']

# 4. 构建模型
model = RandomForestClassifier(random_state=42)

# 5. 训练模型
model.fit(X_train, y_train)

# 6. 评估模型
y_pred = model.predict(X_test)
f1 = f1_score(y_test, y_pred, average='weighted')
print(f"测试集上的加权平均 F1 分数: {f1}")
