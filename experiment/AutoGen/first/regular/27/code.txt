import pandas as pd
import numpy as np
from statsmodels.tsa.vector_ar.var_model import VAR
from sklearn.metrics import mean_squared_error

# 1. 导入数据
data_path = r'E:\Postgraduate\Code\SunnylnAgent\data\csv\bitcoin_price_train.csv'
data = pd.read_csv(data_path)

# 2. 数据预处理
# 检查缺失值
print(data.isnull().sum())

# 选择特征
# 假设我们选择 'High', 'Low', 'Open', 'Close' 作为特征
features = ['High', 'Low', 'Open', 'Close']
data = data[features]

# 3. 构建 VAR 模型
# 将数据转换为时间序列格式
data.index = pd.to_datetime(data.index)

# 拟合 VAR 模型
model = VAR(data)
lags = model.select_order(maxlags=10)
print(lags.summary())

# 选择最佳滞后阶数
best_lag = lags.selected_orders['aic']
var_model = model.fit(best_lag)

# 4. 计算训练集上的均方误差
# 获取预测值
predictions = var_model.fittedvalues

# 计算均方误差
mse = mean_squared_error(data[best_lag:], predictions)
print(f'Training Set MSE: {mse}')

# 5. 报告结果
print(f'Training Set MSE: {mse}')
