import pandas as pd
import numpy as np
from sklearn.ensemble import GradientBoostingRegressor
from sklearn.metrics import mean_squared_error
from sklearn.preprocessing import OneHotEncoder
from sklearn.compose import ColumnTransformer
from sklearn.pipeline import Pipeline

# 1. 导入数据
train_data_path = r'E:\Postgraduate\Code\SunnylnAgent\data\csv\forest_fires_train.csv'
test_data_path = r'E:\Postgraduate\Code\SunnylnAgent\data\csv\forest_fires_test.csv'

train_df = pd.read_csv(train_data_path)
test_df = pd.read_csv(test_data_path)

# 2. 数据预处理
# 检查缺失值
print("训练集缺失值情况：", train_df.isnull().sum())
print("测试集缺失值情况：", test_df.isnull().sum())

# 分离特征和目标变量
X_train = train_df.drop(columns=['area'])
y_train = train_df['area']

X_test = test_df.drop(columns=['area'])
y_test = test_df['area']

# 3. 特征编码
# 确定需要编码的特征
categorical_features = ['month', 'day']
numerical_features = X_train.columns.difference(categorical_features)

# 创建预处理步骤
preprocessor = ColumnTransformer(
    transformers=[
        ('cat', OneHotEncoder(), categorical_features),
        ('num', 'passthrough', numerical_features)
    ])

# 4. 创建模型管道
model = Pipeline(steps=[
    ('preprocessor', preprocessor),
    ('regressor', GradientBoostingRegressor())
])

# 5. 训练模型
model.fit(X_train, y_train)

# 6. 评估模型
y_pred = model.predict(X_test)
mse = mean_squared_error(y_test, y_pred)
print(f"测试集上的均方误差（MSE）: {mse}")
