import pandas as pd
from sklearn.tree import DecisionTreeRegressor
from sklearn.metrics import mean_squared_error
from sklearn.preprocessing import LabelEncoder

# 1. 加载数据
train_data_path = r'E:\Postgraduate\Code\SunnylnAgent\data\csv\crab_age_train.csv'
test_data_path = r'E:\Postgraduate\Code\SunnylnAgent\data\csv\crab_age_test.csv'

train_df = pd.read_csv(train_data_path)
test_df = pd.read_csv(test_data_path)

# 2. 数据预处理
# 检查缺失值
print("训练集缺失值情况：", train_df.isnull().sum())
print("测试集缺失值情况：", test_df.isnull().sum())

# 对性别进行编码
label_encoder = LabelEncoder()
train_df['Sex'] = label_encoder.fit_transform(train_df['Sex'])
test_df['Sex'] = label_encoder.transform(test_df['Sex'])

# 分离特征和目标变量
X_train = train_df.drop(columns=['Age'])
y_train = train_df['Age']
X_test = test_df.drop(columns=['Age'])
y_test = test_df['Age']

# 3. 构建决策树模型
model = DecisionTreeRegressor(random_state=42)

# 4. 训练模型
model.fit(X_train, y_train)

# 5. 评估模型
y_pred = model.predict(X_test)
mse = mean_squared_error(y_test, y_pred)
print(f"测试集上的均方误差（MSE）: {mse}")
