import pandas as pd
from sklearn.svm import SVC
from sklearn.metrics import f1_score
from sklearn.preprocessing import StandardScaler

# 读取数据
train_data = pd.read_csv(r'E:\Postgraduate\Code\SunnylnAgent\data\csv\cirrhosis_patient_train.csv')
test_data = pd.read_csv(r'E:\Postgraduate\Code\SunnylnAgent\data\csv\cirrhosis_patient_test.csv')

# 查看数据的前几行
print(train_data.head())
print(test_data.head())

# 分离特征和标签
X_train = train_data.drop(columns=['Status'])
y_train = train_data['Status']

X_test = test_data.drop(columns=['Status'])
y_test = test_data['Status']

# 对分类变量进行独热编码
X_train_encoded = pd.get_dummies(X_train, drop_first=True)
X_test_encoded = pd.get_dummies(X_test, drop_first=True)

# 确保训练集和测试集的列一致
X_test_encoded = X_test_encoded.reindex(columns=X_train_encoded.columns, fill_value=0)

# 标准化特征
scaler = StandardScaler()
X_train_scaled = scaler.fit_transform(X_train_encoded)
X_test_scaled = scaler.transform(X_test_encoded)

# 构建 SVM 模型
svm_model = SVC(kernel='linear', probability=True)
svm_model.fit(X_train_scaled, y_train)

# 预测
y_pred = svm_model.predict(X_test_scaled)

# 计算 F1 分数
f1 = f1_score(y_test, y_pred, average='weighted')
print(f'加权平均 F1 分数: {f1}')
