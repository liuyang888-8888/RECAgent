import pandas as pd
from sklearn.tree import DecisionTreeClassifier
from sklearn.metrics import f1_score
from sklearn.preprocessing import LabelEncoder

# 1. 导入必要的库

# 2. 加载数据
train_path = r'E:\Postgraduate\Code\SunnylnAgent\data\csv\employee_attrition_train.csv'
test_path = r'E:\Postgraduate\Code\SunnylnAgent\data\csv\employee_attrition_test.csv'

train_data = pd.read_csv(train_path)
test_data = pd.read_csv(test_path)

# 3. 数据预处理
# 检查并处理缺失值
print("训练集缺失值情况：\n", train_data.isnull().sum())
print("测试集缺失值情况：\n", test_data.isnull().sum())

# 假设没有缺失值，如果有缺失值需要进行处理

# 将分类变量进行编码
label_encoders = {}

for column in train_data.select_dtypes(include=['object']).columns:
    le = LabelEncoder()
    train_data[column] = le.fit_transform(train_data[column])
    test_data[column] = le.transform(test_data[column])
    label_encoders[column] = le

# 分离特征和目标变量
X_train = train_data.drop(columns=['Attrition'])
y_train = train_data['Attrition']

X_test = test_data.drop(columns=['Attrition'])
y_test = test_data['Attrition']

# 4. 构建决策树模型
model = DecisionTreeClassifier()

# 5. 训练模型
model.fit(X_train, y_train)

# 6. 评估模型
y_pred = model.predict(X_test)
f1 = f1_score(y_test, y_pred, average='weighted')
print(f"测试集上的加权平均 F1 分数: {f1}")
