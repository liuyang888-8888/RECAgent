import pandas as pd
from sklearn.linear_model import LogisticRegression
from sklearn.metrics import f1_score
from sklearn.preprocessing import LabelEncoder

# 定义数据路径
train_path = r'E:\Postgraduate\Code\SunnylnAgent\data\csv\crime_category_train.csv'
test_path = r'E:\Postgraduate\Code\SunnylnAgent\data\csv\crime_category_test.csv'

# 加载数据
train_df = pd.read_csv(train_path)
test_df = pd.read_csv(test_path)

# 检查数据列名
print("Train data columns:", train_df.columns)
print("Test data columns:", test_df.columns)

# 检查并处理缺失值
if train_df.isnull().sum().sum() > 0:
    train_df = train_df.dropna()
if test_df.isnull().sum().sum() > 0:
    test_df = test_df.dropna()

# 分离特征和标签
if 'category' in train_df.columns and 'category' in test_df.columns:
    X_train = train_df.drop(columns=['category'])
    y_train = train_df['category']
    X_test = test_df.drop(columns=['category'])
    y_test = test_df['category']
else:
    raise ValueError("The 'category' column is not found in the data.")

# 对类别特征进行 one-hot 编码
X_train = pd.get_dummies(X_train, columns=['dayOfWeek', 'pdDistrict'])
X_test = pd.get_dummies(X_test, columns=['dayOfWeek', 'pdDistrict'])

# 确保训练集和测试集的列一致
missing_cols = set(X_train.columns) - set(X_test.columns)
for c in missing_cols:
    X_test[c] = 0
X_test = X_test[X_train.columns]

# 将类别标签编码为数值
label_encoder = LabelEncoder()
y_train = label_encoder.fit_transform(y_train)
y_test = label_encoder.transform(y_test)

# 构建逻辑回归模型
model = LogisticRegression(multi_class='multinomial', solver='lbfgs', max_iter=1000)
model.fit(X_train, y_train)

# 在测试集上预测
y_pred = model.predict(X_test)

# 计算 F1 分数
f1 = f1_score(y_test, y_pred, average='weighted')
print(f'Weighted F1 Score: {f1}')
