import pandas as pd
import numpy as np
from sklearn.ensemble import AdaBoostRegressor
from sklearn.metrics import mean_squared_error
from sklearn.preprocessing import LabelEncoder

# 1. 导入数据
train_path = r'E:\Postgraduate\Code\SunnylnAgent\data\csv\student_performance_train.csv'
test_path = r'E:\Postgraduate\Code\SunnylnAgent\data\csv\student_performance_test.csv'

train_data = pd.read_csv(train_path)
test_data = pd.read_csv(test_path)

# 2. 数据预处理
# 检查缺失值
print("训练数据缺失值情况：", train_data.isnull().sum())
print("测试数据缺失值情况：", test_data.isnull().sum())

# 处理分类变量
label_encoders = {}
for column in train_data.select_dtypes(include=['object']).columns:
    le = LabelEncoder()
    train_data[column] = le.fit_transform(train_data[column])
    test_data[column] = le.transform(test_data[column])
    label_encoders[column] = le

# 3. 准备特征和目标变量
X_train = train_data.drop(columns=['G3'])
y_train = train_data['G3']
X_test = test_data.drop(columns=['G3'])
y_test = test_data['G3']

# 4. 构建 AdaBoost 回归模型
ada_reg = AdaBoostRegressor(random_state=42)

# 5. 训练模型
ada_reg.fit(X_train, y_train)

# 6. 评估模型
y_pred = ada_reg.predict(X_test)
mse = mean_squared_error(y_test, y_pred)
print(f"测试集上的均方误差（MSE）: {mse}")
