import pandas as pd
import numpy as np
from sklearn.cluster import MeanShift
from sklearn.metrics import silhouette_score
from sklearn.preprocessing import StandardScaler, OneHotEncoder
from sklearn.compose import ColumnTransformer
from sklearn.pipeline import Pipeline

# 1. 导入数据
data_path = r'E:\Postgraduate\Code\SunnylnAgent\data\csv\obesity_cvd_risk_train.csv'
data = pd.read_csv(data_path)

# 2. 数据预处理
# 检查缺失值
print("Missing values:\n", data.isnull().sum())

# 假设没有缺失值，如果有缺失值需要进行处理
# data.fillna(data.mean(), inplace=True)  # 用均值填充缺失值

# 将分类变量进行编码
# 假设 'NObeyesdad' 列是分类变量
data['NObeyesdad'] = data['NObeyesdad'].astype('category').cat.codes

# 选择特征列
features = data.drop(columns=['NObeyesdad'])

# 确定分类变量和数值变量
categorical_features = ['Gender', 'family_history_with_overweight', 'FAVC', 'CAEC', 'SMOKE', 'SCC', 'CALC', 'MTRANS']
numerical_features = [col for col in features.columns if col not in categorical_features]

# 创建预处理管道
preprocessor = ColumnTransformer(
    transformers=[
        ('num', StandardScaler(), numerical_features),
        ('cat', OneHotEncoder(), categorical_features)
    ])

# 3. 训练 Mean Shift 聚类模型
pipeline = Pipeline(steps=[
    ('preprocessor', preprocessor),
    ('mean_shift', MeanShift())
])

# 拟合模型
pipeline.fit(features)

# 获取聚类结果
clusters = pipeline.named_steps['mean_shift'].labels_

# 4. 计算轮廓系数
features_transformed = preprocessor.transform(features)
silhouette_avg = silhouette_score(features_transformed, clusters)
print("Silhouette Coefficient: ", silhouette_avg)
