import pandas as pd
import numpy as np
from sklearn.ensemble import GradientBoostingRegressor
from sklearn.metrics import mean_squared_error

# 1. 导入数据
train_path = r'E:\Postgraduate\Code\SunnylnAgent\data\csv\bike_sharing_train.csv'
test_path = r'E:\Postgraduate\Code\SunnylnAgent\data\csv\bike_sharing_test.csv'

train_data = pd.read_csv(train_path)
test_data = pd.read_csv(test_path)

# 2. 数据预处理
# 检查缺失值
print("训练集缺失值情况：", train_data.isnull().sum())
print("测试集缺失值情况：", test_data.isnull().sum())

# 假设没有缺失值，如果有缺失值需要进行处理

# 3. 准备特征和目标变量
X_train = train_data.drop(columns=['cnt'])
y_train = train_data['cnt']

X_test = test_data.drop(columns=['cnt'])
y_test = test_data['cnt']

# 4. 构建模型
model = GradientBoostingRegressor()

# 5. 训练模型
model.fit(X_train, y_train)

# 6. 评估模型
y_pred = model.predict(X_test)
mse = mean_squared_error(y_test, y_pred)
print(f"测试集上的均方误差（MSE）: {mse}")
