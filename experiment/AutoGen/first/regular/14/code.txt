import pandas as pd
import lightgbm as lgb
from sklearn.metrics import mean_squared_error

# 读取数据
train_data_path = r'E:\Postgraduate\Code\SunnylnAgent\data\csv\abalone_train.csv'
test_data_path = r'E:\Postgraduate\Code\SunnylnAgent\data\csv\abalone_test.csv'

train_df = pd.read_csv(train_data_path)

# 分离特征和标签
X_train = train_df.drop(columns=['Age'])

X_test = test_df.drop(columns=['Age'])
y_test = test_df['Age']

# 转换数据为 LightGBM 的 Dataset 格式
train_data = lgb.Dataset(X_train, label=y_train)
test_data = lgb.Dataset(X_test, label=y_test, reference=train_data)

# 设置 LightGBM 参数

    'objective': 'regression',
    'metric': 'mse',
    'boosting_type': 'gbdt',
    'num_leaves': 31,
    'learning_rate': 0.05,
    'feature_fraction': 0.9,
    'bagging_fraction': 0.8,
    'bagging_freq': 5,
    'verbose': 0
}

# 训练模型
num_rounds = 100
bst = lgb.train(params, train_data, num_rounds, valid_sets=[test_data], early_stopping_rounds=10)

# 预测
y_pred = bst.predict(X_test, num_iteration=bst.best_iteration)

# 计算均方误差
mse = mean_squared_error(y_test, y_pred)
print(f'Mean Squared Error on test set: {mse}')
