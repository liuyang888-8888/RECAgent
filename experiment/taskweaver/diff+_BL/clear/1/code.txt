import pandas as pd
from sklearn.preprocessing import OneHotEncoder

# Load the datasets
train_data_path = 'E:\\Postgraduate\\Code\\SunnylnAgent\\data\\csv\\bank_customer_churn_train.csv'
test_data_path = 'E:\\Postgraduate\\Code\\SunnylnAgent\\data\\csv\\bank_customer_churn_test.csv'
train_df = pd.read_csv(train_data_path)
test_df = pd.read_csv(test_data_path)

# One-hot encode 'Geography' and 'Gender'
categorical_features = ['Geography', 'Gender']
encoder = OneHotEncoder()

# Fit the encoder on the training set and transform both sets
encoded_train = encoder.fit_transform(train_df[categorical_features]).toarray()
encoded_test = encoder.transform(test_df[categorical_features]).toarray()

# Create a DataFrame of the encoded variables
encoded_train_df = pd.DataFrame(encoded_train, columns=encoder.get_feature_names_out(categorical_features))
encoded_test_df = pd.DataFrame(encoded_test, columns=encoder.get_feature_names_out(categorical_features))

# Drop the original categorical columns and concatenate the encoded ones
train_df = train_df.drop(columns=categorical_features).join(encoded_train_df)
test_df = test_df.drop(columns=categorical_features).join(encoded_test_df)
