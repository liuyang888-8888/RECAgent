import pandas as pd
from sklearn.ensemble import GradientBoostingRegressor
from sklearn.metrics import mean_squared_error

# Load the datasets
train_data_path = 'E:\\Postgraduate\\Code\\SunnylnAgent\\data\\csv\\bike_sharing_train.csv'
test_data_path = 'E:\\Postgraduate\\Code\\SunnylnAgent\\data\\csv\\bike_sharing_test.csv'

train_df = pd.read_csv(train_data_path)
test_df = pd.read_csv(test_data_path)

# Define features and target
features = train_df.drop('cnt', axis=1)
target = train_df['cnt']

# Initialize the Gradient Boosting Regressor
model = GradientBoostingRegressor(n_estimators=8, learning_rate=0.01, random_state=42)

# Train the model
model.fit(features, target)

# Predict on the test set
predictions = model.predict(test_df.drop('cnt', axis=1))

# Calculate Mean Squared Error
mse = mean_squared_error(test_df['cnt'], predictions)

# Save the predictions
predictions, mse
