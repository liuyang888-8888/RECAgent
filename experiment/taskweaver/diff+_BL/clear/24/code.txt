import pandas as pd

# Load the dataset
file_path = 'E:\\Postgraduate\\Code\\SunnylnAgent\\data\\csv\\titanic_train.csv'
titanic_df = pd.read_csv(file_path)

# Check for missing values
missing_values = titanic_df.isnull().sum()

# Fill missing values with mean/mode
for column in missing_values[missing_values > 0].index:
    if titanic_df[column].dtype == 'float64' or titanic_df[column].dtype == 'int64':
        titanic_df[column].fillna(titanic_df[column].mean(), inplace=True)
    else:
        titanic_df[column].fillna(titanic_df[column].mode()[0], inplace=True)

# Verify the changes
missing_values_after_fill = titanic_df.isnull().sum()

# Result variables
titanic_df, missing_values, missing_values_after_fill
