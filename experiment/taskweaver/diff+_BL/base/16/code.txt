import pandas as pd
from sklearn.preprocessing import OneHotEncoder
from sklearn.model_selection import train_test_split

# Load the datasets
train_data = pd.read_csv(r'E:\Postgraduate\Code\SunnylnAgent\data\csv\media_campaign_cost_train.csv')
test_data = pd.read_csv(r'E:\Postgraduate\Code\SunnylnAgent\data\csv\media_campaign_cost_test.csv')

# Check for missing values
if train_data.isnull().sum().sum() > 0:
    train_data.fillna(train_data.mean(), inplace=True)
if test_data.isnull().sum().sum() > 0:
    test_data.fillna(test_data.mean(), inplace=True)

# Separate features and target
X_train = train_data.drop(columns=['cost'])
y_train = train_data['cost']
X_test = test_data.drop(columns=['cost'])
y_test = test_data['cost']

# Encode categorical variables
encoder = OneHotEncoder()
categorical_features = X_train.select_dtypes(include=['object']).columns

X_train_encoded = encoder.fit_transform(X_train[categorical_features])
X_test_encoded = encoder.transform(X_test[categorical_features])

X_train = pd.concat([X_train.drop(columns=categorical_features), pd.DataFrame(X_train_encoded.toarray(), columns=encoder.get_feature_names_out(categorical_features))], axis=1)
X_test = pd.concat([X_test.drop(columns=categorical_features), pd.DataFrame(X_test_encoded.toarray(), columns=encoder.get_feature_names_out(categorical_features))], axis=1)
