import pandas as pd
from sklearn.preprocessing import LabelEncoder

# Load the datasets
train_data_path = 'E:\\Postgraduate\\Code\\SunnylnAgent\\data\\csv\\employee_attrition_train.csv'
test_data_path = 'E:\\Postgraduate\\Code\\SunnylnAgent\\data\\csv\\employee_attrition_test.csv'

train_df = pd.read_csv(train_data_path)
test_df = pd.read_csv(test_data_path)

# Feature selection - select relevant features for the model
features = [col for col in train_df.columns if col != 'Attrition']

# Encode categorical variables
label_encoders = {}
for feature in features:
    if train_df[feature].dtype == 'object':
        le = LabelEncoder()
        train_df[feature] = le.fit_transform(train_df[feature])
        test_df[feature] = le.transform(test_df[feature])
        label_encoders[feature] = le

# Encode target variable
le_target = LabelEncoder()
train_df['Attrition'] = le_target.fit_transform(train_df['Attrition'])
test_df['Attrition'] = le_target.transform(test_df['Attrition'])

# Prepare the final datasets
X_train = train_df[features]
y_train = train_df['Attrition']
X_test = test_df[features]
y_test = test_df['Attrition']
